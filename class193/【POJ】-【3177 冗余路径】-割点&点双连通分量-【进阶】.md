### 题目描述
给定一个有桥的连通图，通过加边使其成为边双连通图，求最少需要添加的边数。

### 笔试/面试考察点分析
- 考察边双连通分量的概念和应用
- Tarjan算法求桥的实现
- 缩点技巧的应用
- 时间复杂度分析（O(n+m)）
- 结论推导：需要添加的边数为(cnt+1)/2，其中cnt为缩点后度数为1的节点数

### 解题思路
1. 使用Tarjan算法找到所有桥
2. 将每个边双连通分量缩为一个点
3. 统计缩点后度数为1的节点数cnt
4. 最少需要添加的边数为(cnt+1)/2

### 代码实现
```cpp
#include <cstring>
#include <algorithm>
#include <iostream>
#include <stack>
using namespace std;

const int N=5005;
const int M=20005;
int n, m;
int head[N], E=0;
int dfn[N], low[N], ts=0;
stack<int> st;
int id[N], ebcc_cnt=0; // 边双连通分量计数
bool is_bridge[M];
int d[N]; // 统计点的度数

struct Edge{int v, ne;}e[M];

inline void add(int a, int b){
    e[E].v=b;
    e[E].ne=head[a];
    head[a]=E++;
}

void tarjan(int u, int from){
    dfn[u]=low[u]=++ts;
    st.push(u);
    for(int i=head[u]; ~i; i=e[i].ne){
        int v=e[i].v;
        if(!dfn[v]){ // 如果是树枝边
            tarjan(v, i);
            low[u]=min(low[u], low[v]);
            if(dfn[u]<low[v]){ // 连续的较小的偶数和较大的奇数构成桥, (0, 1), (2,3), (4,5)
                is_bridge[i]=is_bridge[i^1]=true; 
            }
        }
        else if(i!=(from^1)){ // 如果不是反向边更新low[u]
            low[u]=min(low[u], dfn[v]);
        }
    }
    
    if(dfn[u]==low[u]){ 
        ++ebcc_cnt;
        int j;
        do{
            j=st.top(); st.pop(); id[j]=ebcc_cnt;
        }while(j!=u);
    }
}

int main(){
    cin>>n>>m;
    memset(head, -1, sizeof head);
    while(m--){
        int a, b;
        // 建立无向图
        cin>>a>>b;
        add(a, b);
        add(b, a);
    }
    
    tarjan(1, -1);
    for(int i=0; i<E; ++i){
        if(is_bridge[i]) d[id[e[i].v]]++;
    }
    
    // 统计缩点后度数为1的点
    // 根据结论计算, 需要添加边的数量为(cnt+1)/2
    int cnt=0;
    for(int i=1; i<=ebcc_cnt; ++i)
        if(d[i]==1) ++cnt;
    
    cout<<(cnt+1)/2<<endl;
    
    return 0;
}
```

### 代码逐行注释
```cpp
#include <cstring>
#include <algorithm>
#include <iostream>
#include <stack>
using namespace std;

const int N=5005;
const int M=20005;
int n, m;
int head[N], E=0; // 邻接表头
int dfn[N], low[N], ts=0; // 时间戳数组
stack<int> st; // 存储节点的栈
int id[N], ebcc_cnt=0; // 边双连通分量计数
bool is_bridge[M]; // 标记边是否为桥
int d[N]; // 统计缩点后的度数

struct Edge{int v, ne;}e[M]; // 边结构体

// 添加边函数
inline void add(int a, int b){
    e[E].v=b;
    e[E].ne=head[a];
    head[a]=E++;
}

// Tarjan算法求桥和边双连通分量
void tarjan(int u, int from){
    dfn[u]=low[u]=++ts; // 初始化时间戳
    st.push(u); // 将当前节点入栈
    for(int i=head[u]; ~i; i=e[i].ne){
        int v=e[i].v;
        if(!dfn[v]){ // 邻接节点未被访问
            tarjan(v, i); // 递归遍历
            low[u]=min(low[u], low[v]); // 更新low值
            if(dfn[u]<low[v]){ // 判断是否为桥
                is_bridge[i]=is_bridge[i^1]=true; 
            }
        }
        else if(i!=(from^1)){ // 不是反向边，更新low值
            low[u]=min(low[u], dfn[v]);
        }
    }
    
    if(dfn[u]==low[u]){ // 找到一个边双连通分量
        ++ebcc_cnt;
        int j;
        do{
            j=st.top(); st.pop();
            id[j]=ebcc_cnt; // 标记节点所属的边双连通分量
        }while(j!=u);
    }
}

int main(){
    cin>>n>>m;
    memset(head, -1, sizeof head); // 初始化邻接表
    while(m--){
        int a, b;
        cin>>a>>b;
        add(a, b);
        add(b, a);
    }
    
    tarjan(1, -1); // 从节点1开始遍历
    for(int i=0; i<E; ++i){
        if(is_bridge[i]) d[id[e[i].v]]++; // 统计缩点后的度数
    }
    
    int cnt=0;
    for(int i=1; i<=ebcc_cnt; ++i)
        if(d[i]==1) ++cnt; // 统计度数为1的节点数
    
    cout<<(cnt+1)/2<<endl; // 输出最少需要添加的边数
    
    return 0;
}
```

### 时间/空间复杂度分析
- 时间复杂度：O(n+m)，n为节点数，m为边数
- 空间复杂度：O(n+m)，存储邻接表和栈
