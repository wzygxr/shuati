### 题目描述
给定一个无向图，求最少需要安装多少个逃生装置，使得任意删除一个点后，每个连通块至少有一个逃生装置。

### 笔试/面试考察点分析
- 考察点双连通分量的应用
- 考察割点的核心作用
- 考察图的容错性设计
- 常见坑点：重边/自环处理、根节点特殊处理

### 解题思路
1. 使用Tarjan算法求割点和点双连通分量
2. 将每个点双连通分量缩为一个点
3. 构建缩点后的树结构
4. 统计树中叶子节点数量，即为最少逃生装置数量

### 完整代码实现
```cpp
#include <iostream>
#include <vector>
#include <stack>
#include <algorithm>
#include <cstring>
using namespace std;

const int MAXN = 1005; // 节点最大数量
const int MAXM = 2005; // 边最大数量

struct Edge {
    int to, next;
} edge[MAXM];

int head[MAXN], tot;
int dfn[MAXN], low[MAXN], timestamp;
bool is_cut[MAXN];
int dcc_cnt; // 点双连通分量数量
vector<int> dcc[MAXN]; // 存储每个点双连通分量的节点
stack<pair<int, int>> st; // 存储边的栈
int cut_cnt; // 割点总数

// 初始化
void init() {
    tot = 0;
    memset(head, -1, sizeof(head));
    memset(dfn, 0, sizeof(dfn));
    memset(low, 0, sizeof(low));
    memset(is_cut, false, sizeof(is_cut));
    timestamp = 0;
    dcc_cnt = 0;
    cut_cnt = 0;
    while (!st.empty()) st.pop();
    for (int i = 0; i < MAXN; i++) dcc[i].clear();
}

// 添加边
void add_edge(int u, int v) {
    edge[tot].to = v;
    edge[tot].next = head[u];
    head[u] = tot++;
}

// Tarjan算法求割点和点双连通分量
void tarjan(int u, int fa) {
    dfn[u] = low[u] = ++timestamp;
    int child = 0;
    for (int i = head[u]; i != -1; i = edge[i].next) {
        int v = edge[i].to;
        if (v == fa) continue;
        if (!dfn[v]) {
            st.push({u, v});
            child++;
            tarjan(v, u);
            low[u] = min(low[u], low[v]);
            // 割点判定
            if (fa == -1 && child >= 2) is_cut[u] = true;
            if (fa != -1 && low[v] >= dfn[u]) is_cut[u] = true;
            // 点双连通分量提取
            if (low[v] >= dfn[u]) {
                dcc_cnt++;
                while (true) {
                    pair<int, int> e = st.top();
                    st.pop();
                    if (dcc[dcc_cnt].empty() || dcc[dcc_cnt].back() != e.first) {
                        dcc[dcc_cnt].push_back(e.first);
                    }
                    if (dcc[dcc_cnt].back() != e.second) {
                        dcc[dcc_cnt].push_back(e.second);
                    }
                    if (e.first == u && e.second == v) break;
                }
            }
        } else if (dfn[v] < dfn[u]) {
            st.push({u, v});
            low[u] = min(low[u], dfn[v]);
        }
    }
}

// 统计割点总数
void count_cut(int n) {
    cut_cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (is_cut[i]) cut_cnt++;
    }
}

int main() {
    int n, m, case_num = 0;
    while (cin >> n >> m && n != 0) {
        init();
        for (int i = 0; i < m; i++) {
            int u, v;
            cin >> u >> v;
            add_edge(u, v);
            add_edge(v, u);
        }
        // 处理每个连通块
        for (int i = 1; i <= n; i++) {
            if (!dfn[i]) {
                tarjan(i, -1);
                // 处理根节点所在的点双连通分量
                if (!st.empty()) {
                    dcc_cnt++;
                    while (!st.empty()) {
                        pair<int, int> e = st.top();
                        st.pop();
                        if (dcc[dcc_cnt].empty() || dcc[dcc_cnt].back() != e.first) {
                            dcc[dcc_cnt].push_back(e.first);
                        }
                        if (dcc[dcc_cnt].back() != e.second) {
                            dcc[dcc_cnt].push_back(e.second);
                        }
                    }
                }
            }
        }
        // 统计割点
        count_cut(n);
        // 统计每个点双连通分量的割点数量
        int leaf = 0;
        for (int i = 1; i <= dcc_cnt; i++) {
            int cnt = 0;
            for (int u : dcc[i]) {
                if (is_cut[u]) cnt++;
            }
            if (cnt == 1) leaf++;
        }
        // 计算最少逃生装置数量
        long long ans = (leaf + 1) / 2;
        cout << "Case " << ++case_num << ": " << ans << endl;
    }
    return 0;
}
```

### 代码逐行注释
```cpp
#include <iostream>
#include <vector>
#include <stack>
#include <algorithm>
#include <cstring>
using namespace std;

const int MAXN = 1005; // 节点最大数量，适配题目数据范围
const int MAXM = 2005; // 边最大数量，无向图双向存储

struct Edge {
    int to, next;
} edge[MAXM]; // 边结构体

int head[MAXN], tot; // 邻接表头指针、边计数
int dfn[MAXN], low[MAXN], timestamp; // 时间戳数组
bool is_cut[MAXN]; // 标记是否为割点
int dcc_cnt; // 点双连通分量数量
vector<int> dcc[MAXN]; // 存储每个点双连通分量的节点
stack<pair<int, int>> st; // 存储边的栈，用于点双连通分量提取
int cut_cnt; // 割点总数

// 初始化函数
void init() {
    tot = 0;
    memset(head, -1, sizeof(head)); // 邻接表头初始化
    memset(dfn, 0, sizeof(dfn)); // 时间戳数组初始化
    memset(low, 0, sizeof(low)); // 可回溯最早时间数组初始化
    memset(is_cut, false, sizeof(is_cut)); // 割点标记数组初始化
    timestamp = 0; // 时间戳计数器重置
    dcc_cnt = 0; // 点双连通分量数量重置
    cut_cnt = 0; // 割点总数重置
    while (!st.empty()) st.pop(); // 清空边栈
    for (int i = 0; i < MAXN; i++) dcc[i].clear(); // 清空点双连通分量存储
}

// 添加边函数
void add_edge(int u, int v) {
    edge[tot].to = v; // 边的目标节点
    edge[tot].next = head[u]; // 边的下一条边指针
    head[u] = tot++; // 更新邻接表头指针
}

// Tarjan算法求割点和点双连通分量
// u：当前节点，fa：父节点
void tarjan(int u, int fa) {
    dfn[u] = low[u] = ++timestamp; // 初始化时间戳
    int child = 0; // 子节点计数
    for (int i = head[u]; i != -1; i = edge[i].next) { // 遍历邻接边
        int v = edge[i].to; // 邻接节点
        if (v == fa) continue; // 跳过父节点
        if (!dfn[v]) { // 邻接节点未被访问过
            st.push({u, v}); // 将边入栈
            child++; // 子节点计数+1
            tarjan(v, u); // 递归遍历子节点
            low[u] = min(low[u], low[v]); // 回溯更新low[u]
            // 割点判定
            if (fa == -1 && child >= 2) is_cut[u] = true; // 根节点割点判定
            if (fa != -1 && low[v] >= dfn[u]) is_cut[u] = true; // 非根节点割点判定
            // 点双连通分量提取
            if (low[v] >= dfn[u]) { // 当low[v] >= dfn[u]时，弹出边栈至当前边
                dcc_cnt++; // 点双连通分量数量+1
                while (true) {
                    pair<int, int> e = st.top(); // 取栈顶边
                    st.pop(); // 弹出栈顶边
                    // 将边的两个端点加入点双连通分量
                    if (dcc[dcc_cnt].empty() || dcc[dcc_cnt].back() != e.first) {
                        dcc[dcc_cnt].push_back(e.first);
                    }
                    if (dcc[dcc_cnt].back() != e.second) {
                        dcc[dcc_cnt].push_back(e.second);
                    }
                    if (e.first == u && e.second == v) break; // 弹出到当前边为止
                }
            }
        } else if (dfn[v] < dfn[u]) { // 邻接节点已被访问过，且是祖先节点
            st.push({u, v}); // 将边入栈
            low[u] = min(low[u], dfn[v]); // 更新low[u]
        }
    }
}

// 统计割点总数
void count_cut(int n) {
    cut_cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (is_cut[i]) cut_cnt++;
    }
}

int main() {
    int n, m, case_num = 0;
    while (cin >> n >> m && n != 0) { // 多组数据输入
        init(); // 初始化
        for (int i = 0; i < m; i++) { // 读取边数据
            int u, v;
            cin >> u >> v;
            add_edge(u, v); // 添加正向边
            add_edge(v, u); // 添加反向边（无向图）
        }
        // 处理每个连通块
        for (int i = 1; i <= n; i++) {
            if (!dfn[i]) { // 未访问过的节点作为连通块根节点
                tarjan(i, -1); // 根节点父节点设为-1
                // 处理根节点所在的点双连通分量
                if (!st.empty()) { // 边栈中剩余边构成根节点所在的点双连通分量
                    dcc_cnt++;
                    while (!st.empty()) {
                        pair<int, int> e = st.top();
                        st.pop();
                        if (dcc[dcc_cnt].empty() || dcc[dcc_cnt].back() != e.first) {
                            dcc[dcc_cnt].push_back(e.first);
                        }
                        if (dcc[dcc_cnt].back() != e.second) {
                            dcc[dcc_cnt].push_back(e.second);
                        }
                    }
                }
            }
        }
        // 统计割点总数
        count_cut(n);
        // 统计每个点双连通分量的割点数量
        int leaf = 0;
        for (int i = 1; i <= dcc_cnt; i++) {
            int cnt = 0;
            for (int u : dcc[i]) {
                if (is_cut[u]) cnt++;
            }
            if (cnt == 1) leaf++; // 割点数量为1的点双连通分量即为叶子节点
        }
        // 计算最少逃生装置数量
        long long ans = (leaf + 1) / 2; // 叶子节点数量+1除以2向上取整
        cout << "Case " << ++case_num << ": " << ans << endl;
    }
    return 0;
}
```

### 时间/空间复杂度分析
- 时间复杂度：O(V + E)，其中V为节点数，E为边数
- 空间复杂度：O(V + E)，用于存储邻接表和点双连通分量
