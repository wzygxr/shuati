# 树上代数与组合数学难题

## 1. 树上群论应用

### 问题1：树的自同构计数

**题目描述**：
计算一棵树的自同构数量，即树的同构映射数量。

**解法**：
使用树哈希+动态规划。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
#include <set>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
long long subtree_hash[MAXN];
int sz[MAXN];
long long count_automorphisms[MAXN];

// 计算阶乘
long long fact[MAXN];

long long power(long long base, long long exp, long long mod) {
    long long result = 1;
    base %= mod;
    while (exp > 0) {
        if (exp & 1) result = (result * base) % mod;
        base = (base * base) % mod;
        exp >>= 1;
    }
    return result;
}

long long mod_inverse(long long a, long long mod) {
    return power(a, mod - 2, mod);
}

// 使用质数作为哈希基数
const long long BASE = 37;

// 用于子树哈希的结构
struct SubtreeInfo {
    long long hash;
    int size;
    long long automorphisms;  // 自同构数量
    
    bool operator<(const SubtreeInfo& other) const {
        if (hash != other.hash) return hash < other.hash;
        if (size != other.size) return size < other.size;
        return automorphisms < other.automorphisms;
    }
};

map<SubtreeInfo, int> hash_count;
int hash_id = 0;

long long dfs_hash(int u, int parent) {
    vector<SubtreeInfo> children_info;
    
    for (int v : adj[u]) {
        if (v != parent) {
            long long child_hash = dfs_hash(v, u);
            SubtreeInfo info;
            info.hash = child_hash;
            info.size = sz[v];
            info.automorphisms = count_automorphisms[v];
            children_info.push_back(info);
        }
    }
    
    // 对子树信息排序，确保同构的树得到相同的哈希值
    sort(children_info.begin(), children_info.end());
    
    // 计算当前子树的哈希值
    long long hash_value = 1;
    for (const auto& info : children_info) {
        hash_value = (hash_value * BASE + info.hash) % MOD;
    }
    
    // 计算自同构数量
    long long auto_count = 1;
    int n = children_info.size();
    
    // 对于相同子树的排列组合
    for (int i = 0; i < n; ) {
        int j = i;
        while (j < n && children_info[j].hash == children_info[i].hash) j++;
        
        int count = j - i;
        auto_count = (auto_count * fact[count]) % MOD;  // 相同子树的排列数
        
        // 每个相同子树的自同构数
        for (int k = i; k < j; k++) {
            auto_count = (auto_count * children_info[k].automorphisms) % MOD;
        }
        
        i = j;
    }
    
    sz[u] = 1;
    for (int v : adj[u]) {
        if (v != parent) sz[u] += sz[v];
    }
    
    subtree_hash[u] = hash_value;
    count_automorphisms[u] = auto_count;
    
    return hash_value;
}

int main() {
    int n;
    cin >> n;
    
    // 预计算阶乘
    fact[0] = 1;
    for (int i = 1; i <= n; i++) {
        fact[i] = (fact[i-1] * i) % MOD;
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_hash(1, 0);
    
    cout << "树的自同构数量: " << count_automorphisms[1] << endl;
    
    return 0;
}
```

## 2. 树上多项式运算

### 问题2：树上路径多项式乘积

**题目描述**：
在一棵树上，每个节点有一个多项式的系数，查询u到v路径上所有节点对应的多项式乘积。

**解法**：
使用树链剖分+多项式乘法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <vector>
using namespace std;

const int MAXN = 1e4 + 5;
const int MOD = 998244353;  // 原根为3的质数

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
vector<int> coeffs[MAXN];  // 每个节点的多项式系数
int cnt = 0;

// 快速数论变换（NTT）
const int G = 3;  // 原根

long long power(long long base, long long exp, long long mod) {
    long long result = 1;
    base %= mod;
    while (exp > 0) {
        if (exp & 1) result = (result * base) % mod;
        base = (base * base) % mod;
        exp >>= 1;
    }
    return result;
}

void ntt(vector<long long>& a, bool invert) {
    int n = a.size();
    
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    
    for (int len = 2; len <= n; len <<= 1) {
        long long wlen = invert ? power(power(G, (MOD - 1) / len, MOD), MOD - 2, MOD) 
                                : power(G, (MOD - 1) / len, MOD);
        
        for (int i = 0; i < n; i += len) {
            long long w = 1;
            for (int j = 0; j < len / 2; j++) {
                long long u = a[i + j];
                long long v = (a[i + j + len / 2] * w) % MOD;
                a[i + j] = (u + v) % MOD;
                a[i + j + len / 2] = (u - v + MOD) % MOD;
                w = (w * wlen) % MOD;
            }
        }
    }
    
    if (invert) {
        long long n_inv = power(n, MOD - 2, MOD);
        for (int i = 0; i < n; i++) {
            a[i] = (a[i] * n_inv) % MOD;
        }
    }
}

vector<long long> multiply(vector<long long> a, vector<long long> b) {
    int n = 1;
    while (n < a.size() + b.size()) n <<= 1;
    
    a.resize(n);
    b.resize(n);
    
    ntt(a, false);
    ntt(b, false);
    
    for (int i = 0; i < n; i++) {
        a[i] = (a[i] * b[i]) % MOD;
    }
    
    ntt(a, true);
    
    // 移除前导零
    while (a.size() > 1 && a.back() == 0) a.pop_back();
    
    return a;
}

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

// 简化版：只计算路径上的多项式乘积
vector<long long> multiplyPathPolynomials(int u, int v) {
    vector<long long> result = {coeffs[1][0]};  // 初始化为第一个节点的多项式
    
    // 获取路径上的节点
    vector<int> path_nodes;
    
    int temp_u = u, temp_v = v;
    
    while (top[temp_u] != top[temp_v]) {
        if (depth[top[temp_u]] < depth[top[temp_v]]) swap(temp_u, temp_v);
        
        for (int i = id[top[temp_u]]; i <= id[temp_u]; i++) {
            if (rev[i]) {
                path_nodes.push_back(rev[i]);
            }
        }
        
        temp_u = parent[top[temp_u]];
    }
    
    if (depth[temp_u] > depth[temp_v]) swap(temp_u, temp_v);
    
    for (int i = id[temp_u]; i <= id[temp_v]; i++) {
        if (rev[i]) {
            path_nodes.push_back(rev[i]);
        }
    }
    
    // 计算乘积
    for (int node : path_nodes) {
        vector<long long> node_poly;
        for (int c : coeffs[node]) {
            node_poly.push_back(c);
        }
        result = multiply(result, node_poly);
    }
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        int deg;
        cin >> deg;
        coeffs[i].resize(deg + 1);
        for (int j = 0; j <= deg; j++) {
            cin >> coeffs[i][j];
        }
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        vector<long long> result = multiplyPathPolynomials(u, v);
        
        cout << "乘积多项式: ";
        for (int j = 0; j < result.size(); j++) {
            cout << result[j] << " ";
        }
        cout << endl;
    }
    
    return 0;
}
```

## 3. 树上组合计数

### 问题3：树上独立集计数

**题目描述**：
计算一棵树上所有独立集的数量（选择节点使得没有两个相邻节点都被选中）。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中u节点是否被选中的独立集数量

void dfs(int u, int parent) {
    dp[u][0] = 1;  // u不选
    dp[u][1] = 1;  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不选：v可选可不选
            dp[u][0] = (dp[u][0] * (dp[v][0] + dp[v][1])) % MOD;
            
            // u选：v不能选
            dp[u][1] = (dp[u][1] * dp[v][0]) % MOD;
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0);
    
    long long result = (dp[1][0] + dp[1][1]) % MOD;
    cout << "独立集数量: " << result << endl;
    
    return 0;
}
```

## 4. 树上生成函数

### 问题4：树上路径长度生成函数

**题目描述**：
计算一棵树上所有路径长度的生成函数，即计算系数序列{a_k}，其中a_k表示长度为k的路径数量。

**解法**：
使用点分治+FFT。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <complex>
using namespace std;

const int MAXN = 1e5 + 5;
const double PI = acos(-1);

vector<int> adj[MAXN];
bool vis[MAXN];
int sz[MAXN];
long long path_count[MAXN * 2];  // 路径长度计数

struct Complex {
    double x, y;
    Complex(double _x = 0, double _y = 0) : x(_x), y(_y) {}
    Complex operator+(const Complex& other) const {
        return Complex(x + other.x, y + other.y);
    }
    Complex operator-(const Complex& other) const {
        return Complex(x - other.x, y - other.y);
    }
    Complex operator*(const Complex& other) const {
        return Complex(x * other.x - y * other.y, x * other.y + y * other.x);
    }
};

void fft(vector<Complex>& a, int invert) {
    int n = a.size();
    
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    
    for (int len = 2; len <= n; len <<= 1) {
        double angle = 2 * PI / len * (invert ? -1 : 1);
        Complex wlen(cos(angle), sin(angle));
        
        for (int i = 0; i < n; i += len) {
            Complex w(1, 0);
            for (int j = 0; j < len / 2; j++) {
                Complex u = a[i + j];
                Complex v = a[i + j + len / 2] * w;
                a[i + j] = u + v;
                a[i + j + len / 2] = u - v;
                w = w * wlen;
            }
        }
    }
    
    if (invert) {
        for (int i = 0; i < n; i++) {
            a[i].x /= n;
        }
    }
}

vector<long long> multiply(vector<long long> a, vector<long long> b) {
    vector<Complex> fa, fb;
    
    int n = 1;
    while (n < a.size() + b.size()) n <<= 1;
    
    fa.resize(n);
    fb.resize(n);
    
    for (int i = 0; i < a.size(); i++) fa[i] = Complex(a[i], 0);
    for (int i = 0; i < b.size(); i++) fb[i] = Complex(b[i], 0);
    
    fft(fa, 1);
    fft(fb, 1);
    
    for (int i = 0; i < n; i++) {
        fa[i] = fa[i] * fb[i];
    }
    
    fft(fa, -1);
    
    vector<long long> result(n);
    for (int i = 0; i < n; i++) {
        result[i] = (long long)(fa[i].x + 0.5);
    }
    
    return result;
}

int getRoot(int u, int parent, int total) {
    sz[u] = 1;
    int max_part = 0;
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            int child_root = getRoot(v, u, total);
            if (child_root) return child_root;
            sz[u] += sz[v];
            max_part = max(max_part, sz[v]);
        }
    }
    
    max_part = max(max_part, total - sz[u]);
    if (max_part * 2 <= total) return u;
    return 0;
}

void getDistances(int u, int parent, int dist, vector<int>& distances) {
    distances.push_back(dist);
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getDistances(v, u, dist + 1, distances);
        }
    }
}

void calc(int u) {
    // 计算通过u的路径
    vector<int> subtree_distances = {0};  // u到自己的距离为0
    vector<long long> temp_count(MAXN * 2, 0);
    temp_count[0] = 1;
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            vector<int> distances;
            getDistances(v, u, 1, distances);
            
            // 将当前子树的距离与之前的结果合并
            vector<long long> current_count(MAXN * 2, 0);
            for (int d : distances) {
                current_count[d]++;
            }
            
            // 与之前的计数合并
            vector<long long> product = multiply(vector<long long>(temp_count.begin(), 
                                temp_count.begin() + subtree_distances.size()), 
                                current_count);
            
            for (int i = 0; i < product.size(); i++) {
                path_count[i] += product[i];
            }
            
            // 更新subtree_distances
            for (int d : distances) {
                subtree_distances.push_back(d);
            }
        }
    }
}

void solve(int u) {
    vis[u] = true;
    calc(u);
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            int new_root = getRoot(v, u, sz[v]);
            solve(new_root);
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    int root = getRoot(1, 0, n);
    solve(root);
    
    cout << "路径长度生成函数系数: ";
    for (int i = 0; i < n; i++) {
        if (path_count[i] > 0) {
            cout << "长度" << i << "的路径数: " << path_count[i] << " ";
        }
    }
    cout << endl;
    
    return 0;
}
```

## 5. 树上线性代数

### 问题5：基尔霍夫矩阵树定理应用

**题目描述**：
计算一个树的生成树数量（对于树来说就是1），并扩展到更复杂的图结构。

**解法**：
使用基尔霍夫矩阵树定理。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const int MAXN = 105;
const double EPS = 1e-9;

double matrix[MAXN][MAXN];

// 计算矩阵的行列式（高斯消元法）
double determinant(int n) {
    double det = 1;
    
    for (int i = 0; i < n; i++) {
        // 找主元
        int max_row = i;
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > abs(matrix[max_row][i])) {
                max_row = j;
            }
        }
        
        if (max_row != i) {
            for (int j = 0; j < n; j++) {
                swap(matrix[i][j], matrix[max_row][j]);
            }
            det = -det;
        }
        
        if (abs(matrix[i][i]) < EPS) {
            return 0;
        }
        
        det *= matrix[i][i];
        
        // 消元
        for (int j = i + 1; j < n; j++) {
            double ratio = matrix[j][i] / matrix[i][i];
            for (int k = i; k < n; k++) {
                matrix[j][k] -= ratio * matrix[i][k];
            }
        }
    }
    
    return det;
}

int main() {
    int n, m;
    cin >> n >> m;
    
    // 构建邻接矩阵和度数矩阵
    vector<vector<int>> adj_matrix(n, vector<int>(n, 0));
    vector<int> degree(n, 0);
    
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        u--; v--;  // 转换为0索引
        adj_matrix[u][v] = adj_matrix[v][u] = 1;
        degree[u]++;
        degree[v]++;
    }
    
    // 构建拉普拉斯矩阵（基尔霍夫矩阵）
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) {
                matrix[i][j] = degree[i];
            } else {
                matrix[i][j] = -adj_matrix[i][j];
            }
        }
    }
    
    // 删除最后一行和最后一列，计算(n-1)阶子式的行列式
    n--;
    double result = determinant(n);
    
    cout << "生成树数量: " << (long long)(result + 0.5) << endl;
    
    return 0;
}
```

## 6. 树上概率组合

### 问题6：树上随机游走返回概率

**题目描述**：
在一棵树上，从节点u开始随机游走，求在k步内返回节点u的概率。

**解法**：
使用动态规划。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e3 + 5;
const int MAXK = 1e3 + 5;

vector<int> adj[MAXN];
double dp[MAXK][MAXN];  // dp[step][node] 表示step步后到达node的概率
int degree[MAXN];

int main() {
    int n, k, start;
    cin >> n >> k >> start;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        degree[u]++;
        degree[v]++;
    }
    
    // 初始化
    dp[0][start] = 1.0;
    
    for (int step = 1; step <= k; step++) {
        for (int u = 1; u <= n; u++) {
            dp[step][u] = 0.0;
            
            for (int v : adj[u]) {
                dp[step][u] += dp[step-1][v] / degree[v];
            }
        }
    }
    
    cout << "k步内返回起始点的概率: " << dp[k][start] << endl;
    
    // 计算期望返回时间
    double expected_time = 0.0;
    double remaining_prob = 1.0;
    
    for (int step = 1; step <= k && remaining_prob > 1e-9; step++) {
        double return_prob = dp[step][start] * remaining_prob;
        expected_time += step * return_prob;
        remaining_prob -= return_prob;
    }
    
    cout << "期望返回时间: " << expected_time << endl;
    
    return 0;
}
```
