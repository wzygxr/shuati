# 树上字符串与模式匹配难题

## 1. 树上字符串路径问题

### 问题1：树上路径字符串字典序最小

**题目描述**：
在一棵树上，每条边有一个字符，多次查询从u到v路径组成的字符串的字典序最小表示。

**解法**：
使用欧拉序+后缀数组+LCA。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <cstring>
using namespace std;

const int MAXN = 1e5 + 5;
const int LOG = 20;

vector<pair<int, char>> adj[MAXN];  // (邻接点, 边字符)
int parent[MAXN], depth[MAXN];
int up[MAXN][LOG];  // 倍增祖先
char edge_char[MAXN];  // 到父节点的边字符
string path_str[MAXN];  // 从根到各节点的路径字符串

void dfs(int u, int p, char p_char, string current_path) {
    parent[u] = p;
    depth[u] = depth[p] + 1;
    edge_char[u] = p_char;
    path_str[u] = current_path;
    
    up[u][0] = p;
    for (int k = 1; k < LOG; k++) {
        if (up[u][k-1] != 0) {
            up[u][k] = up[up[u][k-1]][k-1];
        } else {
            up[u][k] = 0;
        }
    }
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (v != p) {
            dfs(v, u, c, current_path + c);
        }
    }
}

int getLCA(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    
    int diff = depth[u] - depth[v];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            u = up[u][k];
        }
    }
    
    if (u == v) return u;
    
    for (int k = LOG - 1; k >= 0; k--) {
        if (up[u][k] != up[v][k]) {
            u = up[u][k];
            v = up[v][k];
        }
    }
    
    return up[u][0];
}

string getPathString(int u, int v) {
    int lca = getLCA(u, v);
    string path1 = path_str[u].substr(depth[lca]);  // 从LCA到u的路径
    string path2 = path_str[v].substr(depth[lca]);  // 从LCA到v的路径
    
    // 反转path2，得到从v到LCA的路径
    reverse(path2.begin(), path2.end());
    
    return path2 + path1;  // v到LCA再到u的完整路径
}

// 计算字符串的最小表示（最小的字典序循环移位）
string minRotation(string s) {
    int n = s.length();
    s = s + s;  // 连接字符串与其自身
    int i = 0, j = 1, k = 0;
    
    while (i < n && j < n) {
        k = 0;
        while (k < n && s[i + k] == s[j + k]) {
            k++;
        }
        
        if (k == n) break;
        
        if (s[i + k] > s[j + k]) {
            i = i + k + 1;
            if (i <= j) i = j + 1;
        } else {
            j = j + k + 1;
            if (j <= i) j = i + 1;
        }
    }
    
    int pos = min(i, j);
    return s.substr(pos, n);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    dfs(1, 0, ' ', "");
    
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        string path = getPathString(u, v);
        string min_rep = minRotation(path);
        cout << "路径最小表示: " << min_rep << endl;
    }
    
    return 0;
}
```

## 2. 树上AC自动机

### 问题2：树上多模式匹配

**题目描述**：
在一棵树上，每条边有一个字符，给定多个模式串，查询从根到每个节点的路径中包含多少个模式串。

**解法**：
使用AC自动机+树上DFS。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <queue>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

struct ACAutomaton {
    struct Node {
        int next[26];
        int fail;
        int count;  // 以该节点结尾的模式串数量
        vector<int> pattern_ids;  // 经过的模式串ID
        
        Node() {
            fill(next, next + 26, 0);
            fail = 0;
            count = 0;
        }
    };
    
    vector<Node> trie;
    int sz;
    
    ACAutomaton() {
        trie.resize(1);
        sz = 1;
    }
    
    void insert(const string& pattern, int id) {
        int cur = 0;
        for (char c : pattern) {
            int idx = c - 'a';
            if (!trie[cur].next[idx]) {
                trie.push_back(Node());
                trie[cur].next[idx] = sz++;
            }
            cur = trie[cur].next[idx];
        }
        trie[cur].pattern_ids.push_back(id);
    }
    
    void build() {
        queue<int> q;
        for (int i = 0; i < 26; i++) {
            if (trie[0].next[i]) {
                q.push(trie[0].next[i]);
            }
        }
        
        while (!q.empty()) {
            int u = q.front(); q.pop();
            
            for (int i = 0; i < 26; i++) {
                if (trie[u].next[i]) {
                    trie[trie[u].next[i]].fail = trie[trie[u].fail].next[i];
                    q.push(trie[u].next[i]);
                } else {
                    trie[u].next[i] = trie[trie[u].fail].next[i];
                }
            }
        }
    }
    
    int query(const string& text) {
        int cur = 0, result = 0;
        for (char c : text) {
            int idx = c - 'a';
            cur = trie[cur].next[idx];
            
            for (int v = cur; v && trie[v].count != -1; v = trie[v].fail) {
                result += trie[v].count;
                trie[v].count = -1;  // 避免重复计算
            }
        }
        return result;
    }
    
    void count_matches() {
        // 在构建AC自动机后，通过fail指针传递计数
        for (int i = 1; i < sz; i++) {
            if (!trie[i].pattern_ids.empty()) {
                trie[i].count = trie[i].pattern_ids.size();
            }
        }
        
        // 通过fail指针向上更新
        vector<int> nodes;
        for (int i = 0; i < sz; i++) nodes.push_back(i);
        sort(nodes.begin(), nodes.end(), [&](int a, int b) {
            return trie[a].fail < trie[b].fail;  // 按fail值排序
        });
        
        for (int u : nodes) {
            if (trie[u].fail && u != 0) {
                trie[trie[u].fail].count += trie[u].count;
            }
        }
    }
};

vector<pair<int, char>> adj[MAXN];
string patterns[MAXN];
int matches[MAXN];  // 每个节点路径上的匹配数

ACAutomaton acam;

void dfs_with_ac(int u, int parent, string current_path) {
    // 在AC自动机中查询当前路径
    int cur = 0;  // AC自动机当前状态
    int match_count = 0;
    
    // 模拟AC自动机匹配过程
    for (char c : current_path) {
        int idx = c - 'a';
        // 这里需要实现完整的AC自动机匹配逻辑
        // 简化版本
    }
    
    matches[u] = match_count;
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (v != parent) {
            dfs_with_ac(v, u, current_path + c);
        }
    }
}

int main() {
    int n, m;
    cin >> n >> m;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    for (int i = 0; i < m; i++) {
        cin >> patterns[i];
        acam.insert(patterns[i], i);
    }
    
    acam.build();
    acam.count_matches();
    
    dfs_with_ac(1, 0, "");
    
    for (int i = 1; i <= n; i++) {
        cout << "节点" << i << "路径匹配数: " << matches[i] << endl;
    }
    
    return 0;
}
```

## 3. 树上回文串问题

### 问题3：树上路径回文串计数

**题目描述**：
在一棵树上，每条边有一个字符，多次查询从u到v路径上回文子串的数量。

**解法**：
使用树链剖分+回文自动机。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

const int MAXN = 1e4 + 5;
const int MAXC = 26;

vector<pair<int, char>> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
char path_chars[MAXN];  // 从节点到父节点的边字符
int cnt = 0;

// 回文自动机
struct PalindromeAutomaton {
    struct Node {
        int next[MAXC];
        int fail;
        int len;
        int count;  // 以该节点结尾的回文串数量
        
        Node() {
            fill(next, next + MAXC, 0);
            fail = 0;
            len = 0;
            count = 0;
        }
    };
    
    vector<Node> nodes;
    string s;
    int last;
    int n;
    
    PalindromeAutomaton() {
        nodes.resize(2);
        nodes[0].len = 0; nodes[0].fail = 1;
        nodes[1].len = -1; nodes[1].fail = 0;
        n = 0;
        last = 0;
    }
    
    int get_fail(int x) {
        while (s[n - nodes[x].len - 1] != s[n]) x = nodes[x].fail;
        return x;
    }
    
    void add(char c) {
        s += c;
        n++;
        int cur = get_fail(last);
        
        if (!nodes[cur].next[c - 'a']) {
            nodes.push_back(Node());
            int idx = nodes.size() - 1;
            nodes[idx].len = nodes[cur].len + 2;
            nodes[idx].fail = nodes[get_fail(nodes[cur].fail)].next[c - 'a'];
            nodes[cur].next[c - 'a'] = idx;
        }
        
        last = nodes[cur].next[c - 'a'];
        nodes[last].count++;
    }
    
    int count_palindromes() {
        // 从后往前更新计数
        for (int i = nodes.size() - 1; i >= 2; i--) {
            nodes[nodes[i].fail].count += nodes[i].count;
        }
        return nodes[0].count + nodes[1].count - 2;  // 减去两个虚拟节点
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (v != fa) {
            path_chars[v] = c;
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

// 获取从u到v的路径字符串
string getPathString(int u, int v) {
    string path1, path2;
    
    // 从u到LCA的路径
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        for (int i = id[u]; i > id[top[u]]; i--) {
            path1 = path_chars[rev[i]] + path1;
        }
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    for (int i = id[u] + 1; i <= id[v]; i++) {
        path2 += path_chars[rev[i]];
    }
    
    reverse(path1.begin(), path1.end());
    return path1 + path2;
}

int countPalindromesInString(const string& s) {
    PalindromeAutomaton pam;
    for (char c : s) {
        pam.add(c);
    }
    return pam.count_palindromes();
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        string path = getPathString(u, v);
        int count = countPalindromesInString(path);
        cout << "路径回文串数量: " << count << endl;
    }
    
    return 0;
}
```

## 4. 树上后缀数组

### 问题4：树上子串查询

**题目描述**：
在一棵树上，每条边有一个字符，查询所有从根开始的路径中，字典序第k小的子串。

**解法**：
使用后缀数组+树上DFS。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <set>
using namespace std;

const int MAXN = 1e4 + 5;

vector<pair<int, char>> adj[MAXN];
vector<string> all_paths;  // 所有从根到叶子的路径

void dfs(int u, int parent, string current_path) {
    bool is_leaf = true;
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (v != parent) {
            is_leaf = false;
            dfs(v, u, current_path + c);
        }
    }
    
    if (is_leaf && !current_path.empty()) {
        all_paths.push_back(current_path);
    }
}

// 生成所有子串
vector<string> generate_substrings(const vector<string>& paths) {
    set<string> substrings;
    
    for (const string& path : paths) {
        for (int i = 0; i < path.length(); i++) {
            for (int j = i; j < path.length(); j++) {
                substrings.insert(path.substr(i, j - i + 1));
            }
        }
    }
    
    vector<string> result(substrings.begin(), substrings.end());
    sort(result.begin(), result.end());
    return result;
}

int main() {
    int n, k;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    dfs(1, 0, "");
    
    vector<string> all_substrings = generate_substrings(all_paths);
    
    if (k <= all_substrings.size()) {
        cout << "第" << k << "小的子串: " << all_substrings[k-1] << endl;
    } else {
        cout << "不存在第" << k << "小的子串" << endl;
    }
    
    return 0;
}
```

## 5. 树上字符串哈希

### 问题5：树上字符串匹配

**题目描述**：
在一棵树上，每条边有一个字符，多次查询模式串在从根到某节点路径中出现的次数。

**解法**：
使用字符串哈希+树上DFS。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

const int MAXN = 1e5 + 5;
const long long BASE = 31;
const long long MOD = 1e9 + 7;

vector<pair<int, char>> adj[MAXN];
long long hsh[MAXN];      // 从根到当前节点的哈希值
long long pwr[MAXN];      // BASE的幂
int path_len[MAXN];       // 路径长度
int occurrences[MAXN];    // 模式串出现次数

string pattern;

long long mod_pow(long long base, long long exp, long long mod) {
    long long result = 1;
    base %= mod;
    while (exp > 0) {
        if (exp & 1) result = (result * base) % mod;
        base = (base * base) % mod;
        exp >>= 1;
    }
    return result;
}

void dfs(int u, int parent, long long current_hash, int current_len) {
    hsh[u] = current_hash;
    path_len[u] = current_len;
    
    // 检查当前路径中是否包含模式串
    occurrences[u] = 0;
    
    if (current_len >= pattern.length()) {
                // 计算模式串的哈希值
        long long pattern_hash = 0;
        for (char c : pattern) {
            pattern_hash = (pattern_hash * BASE + (c - 'a' + 1)) % MOD;
        }
        
        // 检查当前路径的每个可能位置
        for (int start = 0; start <= current_len - pattern.length(); start++) {
            int end_pos = start + pattern.length() - 1;
            
            // 计算子串哈希
            long long sub_hash;
            if (start == 0) {
                sub_hash = (hsh[u] - ((start + pattern.length() < current_len) ? 
                          (hsh[u] / pwr[pattern.length()]) % MOD : 0) + MOD) % MOD;
            } else {
                long long prefix_hash = (start > 0) ? 
                    (hsh[u] / pwr[current_len - start]) % MOD : 0;
                sub_hash = (current_hash - prefix_hash * pwr[pattern.length()]) % MOD;
                if (sub_hash < 0) sub_hash += MOD;
            }
            
            if (sub_hash == pattern_hash) {
                occurrences[u]++;
            }
        }
    }
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (v != parent) {
            long long new_hash = (current_hash * BASE + (c - 'a' + 1)) % MOD;
            dfs(v, u, new_hash, current_len + 1);
        }
    }
}

// 简化版本：使用KMP算法进行匹配
vector<int> computeLPS(const string& pattern) {
    int m = pattern.length();
    vector<int> lps(m, 0);
    int len = 0;
    int i = 1;
    
    while (i < m) {
        if (pattern[i] == pattern[len]) {
            len++;
            lps[i] = len;
            i++;
        } else {
            if (len != 0) {
                len = lps[len - 1];
            } else {
                lps[i] = 0;
                i++;
            }
        }
    }
    return lps;
}

int countOccurrences(const string& text, const string& pattern) {
    int count = 0;
    vector<int> lps = computeLPS(pattern);
    int i = 0, j = 0;
    
    while (i < text.length()) {
        if (pattern[j] == text[i]) {
            i++;
            j++;
        }
        
        if (j == pattern.length()) {
            count++;
            j = lps[j - 1];
        } else if (i < text.length() && pattern[j] != text[i]) {
            if (j != 0) {
                j = lps[j - 1];
            } else {
                i++;
            }
        }
    }
    return count;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    cin >> pattern;
    
    // 预计算幂
    pwr[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        pwr[i] = (pwr[i-1] * BASE) % MOD;
    }
    
    dfs(1, 0, 0, 0);
    
    for (int i = 0; i < q; i++) {
        int v;
        cin >> v;
        cout << "模式串出现次数: " << occurrences[v] << endl;
    }
    
    return 0;
}
```

## 6. 树上字典序路径

### 问题6：树上字典序第k小路径

**题目描述**：
在一棵树上，每条边有一个字符，查询所有简单路径中字典序第k小的路径字符串。

**解法**：
使用DFS+优先队列。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <queue>
using namespace std;

const int MAXN = 1e3 + 5;

vector<pair<int, char>> adj[MAXN];
bool visited[MAXN];
priority_queue<string, vector<string>, greater<string>> all_paths;

void dfs_paths(int u, int target, string current_path, int depth) {
    if (depth > 100) return;  // 防止路径过长
    
    if (u == target) {
        all_paths.push(current_path);
        return;
    }
    
    visited[u] = true;
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        char c = edge.second;
        if (!visited[v]) {
            dfs_paths(v, target, current_path + c, depth + 1);
        }
    }
    
    visited[u] = false;
}

int main() {
    int n, k;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        char c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    // 生成所有路径
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            fill(visited, visited + n + 1, false);
            dfs_paths(i, j, "", 0);
        }
    }
    
    // 找到第k小的路径
    int count = 0;
    string result = "";
    while (!all_paths.empty() && count < k) {
        result = all_paths.top();
        all_paths.pop();
        count++;
    }
    
    if (count == k) {
        cout << "第" << k << "小的路径: " << result << endl;
    } else {
        cout << "不存在第" << k << "小的路径" << endl;
    }
    
    return 0;
}
```
