# 树上复杂图论难题

## 1. 树上匹配问题

### 问题1：树上最大匹配计数

**题目描述**：
在一棵树上，计算所有最大匹配的数量。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中，u点是否与父节点匹配的最大匹配数
long long cnt[MAXN][2]; // cnt[u][0/1] 表示对应方案数

void dfs(int u, int parent) {
    dp[u][0] = 0;  // u不与父节点匹配
    dp[u][1] = 0;  // u与父节点匹配
    cnt[u][0] = 1;
    cnt[u][1] = 1;
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不与父节点匹配的情况
            long long old_dp0 = dp[u][0];
            long long old_cnt0 = cnt[u][0];
            
            // 情况1：u不匹配，v也不匹配
            dp[u][0] = old_dp0 + dp[v][0];
            cnt[u][0] = (old_cnt0 * cnt[v][0]) % MOD;
            
            // 情况2：u不匹配，v匹配（通过连接u-v）
            if (old_dp0 + 1 + dp[v][1] > dp[u][0]) {
                dp[u][0] = old_dp0 + 1 + dp[v][1];
                cnt[u][0] = (old_cnt0 * cnt[v][1]) % MOD;
            } else if (old_dp0 + 1 + dp[v][1] == dp[u][0]) {
                cnt[u][0] = (cnt[u][0] + old_cnt0 * cnt[v][1]) % MOD;
            }
            
            // u与父节点匹配的情况
            long long old_dp1 = dp[u][1];
            long long old_cnt1 = cnt[u][1];
            
            // 这种情况下u必须选择一个子节点进行匹配
            long long best = old_dp1 + dp[v][0];
            long long ways = (old_cnt1 * cnt[v][0]) % MOD;
            
            // 尝试与v匹配
            if (1 + dp[v][1] + (old_dp1 - 0) > best) {
                best = 1 + dp[v][1] + (old_dp1 - 0);
                ways = (cnt[u][1] * cnt[v][1]) % MOD;
            } else if (1 + dp[v][1] + (old_dp1 - 0) == best) {
                ways = (ways + cnt[u][1] * cnt[v][1]) % MOD;
            }
            
            dp[u][1] = best;
            cnt[u][1] = ways;
        }
    }
}

pair<long long, long long> solveMaxMatching(int n) {
    dfs(1, 0);
    
    if (dp[1][0] > dp[1][1]) {
        return {dp[1][0], cnt[1][0]};
    } else if (dp[1][0] < dp[1][1]) {
        return {dp[1][1], cnt[1][1]};
    } else {
        return {dp[1][0], (cnt[1][0] + cnt[1][1]) % MOD};
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    auto result = solveMaxMatching(n);
    cout << "最大匹配数: " << result.first << endl;
    cout << "方案数: " << result.second << endl;
    
    return 0;
}
```

## 2. 树上染色问题

### 问题2：树上合法染色方案数

**题目描述**：
在一棵树上，给定k种颜色，要求相邻节点颜色不同，计算合法染色方案数。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
int k;  // 颜色数

long long dfs(int u, int parent, int color) {
    // 计算以u为根的子树中，u染color颜色的方案数
    long long result = 1;
    
    for (int v : adj[u]) {
        if (v != parent) {
            long long ways = 0;
            // v可以染k-1种颜色（除了u的颜色）
            for (int c = 1; c <= k; c++) {
                if (c != color) {
                    ways = (ways + dfs(v, u, c)) % MOD;
                }
            }
            result = (result * ways) % MOD;
        }
    }
    
    return result;
}

long long countColorings(int n) {
    // 以任意节点为根，计算染色方案数
    // 对于树来说，从任意节点开始都可以
    
    // 优化：直接计算
    // 根节点有k种选择，每个其他节点有k-1种选择
    long long result = k;
    for (int i = 2; i <= n; i++) {
        result = (result * (k - 1)) % MOD;
    }
    
    return result;
}

// 更复杂的版本：有颜色限制
long long dp[MAXN][105];  // dp[u][c] 表示u节点染颜色c的方案数

void dfs_with_constraints(int u, int parent, vector<bool>& allowed_colors) {
    for (int c = 1; c <= k; c++) {
        if (allowed_colors[c]) {
            dp[u][c] = 1;
        } else {
            dp[u][c] = 0;
        }
    }
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_with_constraints(v, u, allowed_colors);
            
            for (int c = 1; c <= k; c++) {
                if (allowed_colors[c]) {
                    long long ways = 0;
                    for (int nc = 1; nc <= k; nc++) {
                        if (nc != c) {
                            ways = (ways + dp[v][nc]) % MOD;
                        }
                    }
                    dp[u][c] = (dp[u][c] * ways) % MOD;
                }
            }
        }
    }
}

int main() {
    int n;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "合法染色方案数: " << countColorings(n) << endl;
    
    return 0;
}
```

## 3. 树上流网络

### 问题3：树上最大流

**题目描述**：
在一棵树上，每条边有容量，求从源点s到汇点t的最大流。

**解法**：
利用树的性质简化最大流算法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;

vector<pair<int, int>> adj[MAXN];  // (邻接点, 容量)
int parent[MAXN], capacity[MAXN];  // 从节点到其父节点的边容量

void dfs(int u, int parent_node) {
    for (auto& edge : adj[u]) {
        int v = edge.first, cap = edge.second;
        if (v != parent_node) {
            parent[v] = u;
            capacity[v] = cap;
            dfs(v, u);
        }
    }
}

// 计算从s到t路径上的最小容量（瓶颈）
int min_capacity_on_path(int s, int t) {
    int min_cap = INT_MAX;
    
    // 先将s和t提到同一深度
    int temp_s = s, temp_t = t;
    
    // 获取s到t路径上的所有边容量
    vector<int> path_caps;
    
    // 将深度更深的点上移
    while (temp_s != temp_t) {
        if (depth[temp_s] > depth[temp_t]) {
            path_caps.push_back(capacity[temp_s]);
            temp_s = parent[temp_s];
        } else {
            path_caps.push_back(capacity[temp_t]);
            temp_t = parent[temp_t];
        }
    }
    
    if (path_caps.empty()) return INT_MAX;
    
    for (int cap : path_caps) {
        min_cap = min(min_cap, cap);
    }
    
    return min_cap;
}

// 由于是树，任意两点间只有一条简单路径
int max_flow_tree(int s, int t) {
    return min_capacity_on_path(s, t);
}

int depth[MAXN];

void dfs_depth(int u, int parent_node, int d) {
    depth[u] = d;
    for (auto& edge : adj[u]) {
        int v = edge.first;
        if (v != parent_node) {
            dfs_depth(v, u, d + 1);
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v, c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    dfs_depth(1, 0, 1);
    dfs(1, 0);
    
    int q;
    cin >> q;
    
    for (int i = 0; i < q; i++) {
        int s, t;
        cin >> s >> t;
        cout << "最大流: " << max_flow_tree(s, t) << endl;
    }
    
    return 0;
}
```

## 4. 树上覆盖问题

### 问题4：树上最小顶点覆盖

**题目描述**：
在一棵树上，选择最少的节点，使得每条边至少有一个端点被选中。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int dp[MAXN][2];  // dp[u][0/1] 表示u子树中，u节点是否被选中的最小顶点覆盖数

void dfs(int u, int parent) {
    dp[u][0] = 0;  // u不选
    dp[u][1] = 1;  // u选
    
    bool has_child = false;
    
    for (int v : adj[u]) {
        if (v != parent) {
            has_child = true;
            dfs(v, u);
            
            // u不选：v必须选
            dp[u][0] += dp[v][1];
            
            // u选：v可选可不选，取较小值
            dp[u][1] += min(dp[v][0], dp[v][1]);
        }
    }
    
    // 如果是叶子节点，u不选的话覆盖数为0
    if (!has_child) {
        dp[u][0] = 0;
    }
}

int minVertexCover(int n) {
    dfs(1, 0);
    return min(dp[1][0], dp[1][1]);
}

// 扩展：带权最小顶点覆盖
int weighted_dp[MAXN][2];
int weights[MAXN];

void dfs_weighted(int u, int parent) {
    weighted_dp[u][0] = 0;  // u不选
    weighted_dp[u][1] = weights[u];  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_weighted(v, u);
            
            // u不选：v必须选
            weighted_dp[u][0] += weighted_dp[v][1];
            
            // u选：v可选可不选
            weighted_dp[u][1] += min(weighted_dp[v][0], weighted_dp[v][1]);
        }
    }
}

int minWeightedVertexCover(int n) {
    dfs_weighted(1, 0);
    return min(weighted_dp[1][0], weighted_dp[1][1]);
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> weights[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "最小顶点覆盖数: " << minVertexCover(n) << endl;
    cout << "最小带权顶点覆盖: " << minWeightedVertexCover(n) << endl;
    
    return 0;
}
```

## 5. 树上独立集问题

### 问题5：树上最大独立集变种

**题目描述**：
在一棵树上，每个节点有权值，选择节点使得权值和最大，且任意两个选择的节点距离至少为3。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int weights[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示u子树中，u节点是否被选中的最大权值和

void dfs(int u, int parent) {
    dp[u][0] = 0;  // u不选
    dp[u][1] = weights[u];  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不选：v可选可不选
            dp[u][0] += max(dp[v][0], dp[v][1]);
            
            // u选：v和v的所有邻居都不能选
            // 但在这个变种中，只需要保证距离为3，所以v不能选
            dp[u][1] += dp[v][0];
        }
    }
}

long long maxWeightIndependentSet(int n) {
    dfs(1, 0);
    return max(dp[1][0], dp[1][1]);
}

// 更严格的变种：距离至少为d
long long dp_d[MAXN][2];

void dfs_distance_d(int u, int parent, int d) {
    dp_d[u][0] = 0;
    dp_d[u][1] = weights[u];
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_distance_d(v, u, d);
            
            dp_d[u][0] += max(dp_d[v][0], dp_d[v][1]);
            
            // 如果选了u，那么距离u不超过d的所有节点都不能选
            // 这里需要更复杂的处理
            dp_d[u][1] += dp_d[v][0];
        }
    }
}

// 考虑距离为2的约束（邻居不能选）
long long dp_dist2[MAXN][3];  // 0:不选，1:选，2:父节点被选

void dfs_distance2(int u, int parent) {
    dp_dist2[u][0] = 0;  // u不选
    dp_dist2[u][1] = weights[u];  // u选
    dp_dist2[u][2] = 0;  // 父节点被选，u不能选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_distance2(v, u);
            
            // u不选
            long long temp0 = dp_dist2[u][0];
            dp_dist2[u][0] = temp0 + max({dp_dist2[v][0], dp_dist2[v][1], dp_dist2[v][2]});
            
            // u选：v不能选
            long long temp1 = dp_dist2[u][1];
            dp_dist2[u][1] = temp1 + max(dp_dist2[v][0], dp_dist2[v][2]);
            
            // 父节点被选，u不能选
            long long temp2 = dp_dist2[u][2];
            dp_dist2[u][2] = temp2 + max({dp_dist2[v][0], dp_dist2[v][1], dp_dist2[v][2]});
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> weights[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "最大独立集权值和: " << maxWeightIndependentSet(n) << endl;
    
    dfs_distance2(1, 0);
    cout << "距离2约束的最大权值和: " << max({dp_dist2[1][0], dp_dist2[1][1], dp_dist2[1][2]}) << endl;
    
    return 0;
}
```

## 6. 树上路径问题

### 问题6：树上路径边权最值

**题目描述**：
在一棵树上，多次查询u到v路径上边权的最大值或最小值。

**解法**：
使用倍增法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;
const int LOG = 20;

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边权)
int parent[MAXN], depth[MAXN];
int up[MAXN][LOG];           // 倍增祖先
int max_up[MAXN][LOG];       // 到祖先路径上的最大边权
int min_up[MAXN][LOG];       // 到祖先路径上的最小边权

void dfs(int u, int p, int p_weight) {
    parent[u] = p;
    depth[u] = depth[p] + 1;
    up[u][0] = p;
    max_up[u][0] = p_weight;
    min_up[u][0] = p_weight;
    
    for (int k = 1; k < LOG; k++) {
        if (up[u][k-1] != 0) {
            up[u][k] = up[up[u][k-1]][k-1];
            max_up[u][k] = max(max_up[u][k-1], max_up[up[u][k-1]][k-1]);
            min_up[u][k] = min(min_up[u][k-1], min_up[up[u][k-1]][k-1]);
        } else {
            up[u][k] = 0;
            max_up[u][k] = INT_MIN;
            min_up[u][k] = INT_MAX;
        }
    }
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (v != p) {
            dfs(v, u, w);
        }
    }
}

int getLCA(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    
    int diff = depth[u] - depth[v];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            u = up[u][k];
        }
    }
    
    if (u == v) return u;
    
    for (int k = LOG - 1; k >= 0; k--) {
        if (up[u][k] != up[v][k]) {
            u = up[u][k];
            v = up[v][k];
        }
    }
    
    return up[u][0];
}

// 查询u到v路径上的最大边权
int queryMaxEdge(int u, int v) {
    int lca = getLCA(u, v);
    int result = INT_MIN;
    
    // 查询u到LCA的路径
    int temp = u;
    int diff = depth[u] - depth[lca];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            result = max(result, max_up[temp][k]);
            temp = up[temp][k];
        }
    }
    
    // 查询v到LCA的路径
    temp = v;
    diff = depth[v] - depth[lca];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            result = max(result, max_up[temp][k]);
            temp = up[temp][k];
        }
    }
    
    return result;
}

// 查询u到v路径上的最小边权
int queryMinEdge(int u, int v) {
    int lca = getLCA(u, v);
    int result = INT_MAX;
    
    // 查询u到LCA的路径
    int temp = u;
    int diff = depth[u] - depth[lca];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            result = min(result, min_up[temp][k]);
            temp = up[temp][k];
        }
    }
    
    // 查询v到LCA的路径
    temp = v;
    diff = depth[v] - depth[lca];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            result = min(result, min_up[temp][k]);
            temp = up[temp][k];
        }
    }
    
    return result;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    
    dfs(1, 0, 0);
    
    int q;
    cin >> q;
    
    for (int i = 0; i < q; i++) {
        int op, u, v;
        cin >> op >> u >> v;
        
        if (op == 1) {  // 查询最大边权
            cout << "最大边权: " << queryMaxEdge(u, v) << endl;
        } else {  // 查询最小边权
            cout << "最小边权: " << queryMinEdge(u, v) << endl;
        }
    }
    
    return 0;
}
```
