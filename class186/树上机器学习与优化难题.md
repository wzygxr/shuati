# 树上机器学习与优化难题

## 1. 树上动态规划优化

### 问题1：树上背包问题的优化

**题目描述**：
在一棵树上，每个节点有一个权值和体积，选择节点使得总价值最大，但选择的节点必须形成连通子图，且总体积不超过V。

**解法**：
使用树上背包DP+优化。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e3 + 5;
const int MAXV = 1e3 + 5;

vector<int> adj[MAXN];
int w[MAXN], v[MAXN];  // 权值和体积
int dp[MAXN][MAXV];    // dp[u][j] 表示以u为根的子树中，体积为j的最大价值

void dfs(int u, int parent) {
    // 初始化当前节点
    for (int j = 0; j <= V; j++) {
        dp[u][j] = (j >= v[u]) ? w[u] : 0;  // 可以选择当前节点
    }
    
    // 遍历子节点
    for (int child : adj[u]) {
        if (child != parent) {
            dfs(child, u);
            
            // 合并子树信息 - 树上背包
            for (int j = V; j >= v[u]; j--) {  // 注意逆序枚举
                for (int k = 0; k <= j - v[u]; k++) {
                    if (dp[child][k] > 0 || k == 0) {  // k=0表示不选择子树任何节点
                        dp[u][j] = max(dp[u][j], dp[u][j-k] + dp[child][k]);
                    }
                }
            }
        }
    }
}

// 优化版本：长链剖分优化
int hson[MAXN], sz[MAXN];

void getHson(int u, int parent) {
    sz[u] = 1;
    hson[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            getHson(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[hson[u]]) {
                hson[u] = v;
            }
        }
    }
}

vector<int> temp_dp;

void dfs_optimized(int u, int parent) {
    if (hson[u]) {
        dfs_optimized(hson[u], u);
        // 直接继承重儿子的信息
        for (int j = 0; j <= V; j++) {
            dp[u][j] = dp[hson[u]][j];
        }
    } else {
        // 叶子节点
        for (int j = 0; j <= V; j++) {
            dp[u][j] = (j >= v[u]) ? w[u] : 0;
        }
    }
    
    // 处理轻儿子
    for (int child : adj[u]) {
        if (child != parent && child != hson[u]) {
            dfs_optimized(child, u);
            
            // 合并轻儿子信息
            for (int j = V; j >= v[u]; j--) {
                for (int k = 0; k <= j - v[u]; k++) {
                    if (dp[child][k] > 0 || k == 0) {
                        dp[u][j] = max(dp[u][j], dp[u][j-k] + dp[child][k]);
                    }
                }
            }
        }
    }
    
    // 考虑选择当前节点
    if (v[u] <= V) {
        for (int j = V; j >= v[u]; j--) {
            dp[u][j] = max(dp[u][j], dp[u][j - v[u]] + w[u]);
        }
    }
}

int main() {
    int n, V;
    cin >> n >> V;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i] >> v[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    getHson(1, 0);
    dfs_optimized(1, 0);
    
    int result = 0;
    for (int j = 0; j <= V; j++) {
        result = max(result, dp[1][j]);
    }
    
    cout << "最大价值: " << result << endl;
    
    return 0;
}
```

## 2. 树上启发式算法

### 问题2：树上众数路径

**题目描述**：
在一棵树上，每个节点有一个颜色，找出出现次数最多的颜色的路径。

**解法**：
使用启发式合并（DSU on Tree）。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int color[MAXN];
int sz[MAXN], son[MAXN];
int cnt[MAXN], max_freq = 0, max_color = 0;
long long total_paths = 0;

void dfs_size(int u, int parent) {
    sz[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_size(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[son[u]]) {
                son[u] = v;
            }
        }
    }
}

unordered_map<int, int> *count, temp_count;

void add(int u, int parent, int sign) {
    if (sign == 1) {
        cnt[color[u]]++;
        if (cnt[color[u]] > max_freq) {
            max_freq = cnt[color[u]];
            max_color = color[u];
            total_paths = 1;
        } else if (cnt[color[u]] == max_freq) {
            total_paths++;
        }
    } else {
        if (cnt[color[u]] == max_freq && color[u] == max_color) {
            total_paths--;
            if (total_paths == 0) {
                max_freq--;
            }
        }
        cnt[color[u]]--;
    }
    
    for (int v : adj[u]) {
        if (v != parent) {
            add(v, u, sign);
        }
    }
}

void dfs_hld(int u, int parent, bool keep) {
    // 先处理轻儿子
    for (int v : adj[u]) {
        if (v != parent && v != son[u]) {
            dfs_hld(v, u, 0);
        }
    }
    
    // 处理重儿子
    if (son[u]) {
        dfs_hld(son[u], u, 1);
    }
    
    // 计算u的贡献
    add(u, parent, 1);
    
    // 输出当前子树的众数信息
    cout << "以" << u << "为根的子树中，众数为: " << max_color << "，出现次数: " << max_freq << endl;
    
    // 如果不是保留节点，清空信息
    if (!keep) {
        add(u, parent, -1);
        max_freq = max_color = 0;
        total_paths = 0;
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> color[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_size(1, 0);
    dfs_hld(1, 0, 0);
    
    return 0;
}
```

## 3. 树上概率DP

### 问题3：树上随机游走期望

**题目描述**：
在一棵树上，从节点s开始随机游走，求到达节点t的期望步数。

**解法**：
建立线性方程组，使用高斯消元求解。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const int MAXN = 1e3 + 5;
const double EPS = 1e-9;

vector<int> adj[MAXN];
double matrix[MAXN][MAXN];
int degree[MAXN];

void gaussian_elimination(int n) {
    for (int i = 0; i < n; i++) {
        // 找主元
        int max_row = i;
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > abs(matrix[max_row][i])) {
                max_row = j;
            }
        }
        
        if (max_row != i) {
            for (int j = 0; j <= n; j++) {
                swap(matrix[i][j], matrix[max_row][j]);
            }
        }
        
        // 消元
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > EPS) {
                double ratio = matrix[j][i] / matrix[i][i];
                for (int k = i; k <= n; k++) {
                    matrix[j][k] -= ratio * matrix[i][k];
                }
            }
        }
    }
    
    // 回代
    vector<double> result(n);
    for (int i = n - 1; i >= 0; i--) {
        result[i] = matrix[i][n] / matrix[i][i];
        for (int j = i - 1; j >= 0; j--) {
            matrix[j][n] -= matrix[j][i] * result[i];
        }
    }
}

int main() {
    int n, s, t;
    cin >> n >> s >> t;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        degree[u]++;
        degree[v]++;
    }
    
    // 建立方程组
    // E[u] = 1 + (1/degree[u]) * sum(E[v]) 对所有邻接点v
    for (int u = 1; u <= n; u++) {
        if (u == t) {
            // 目标节点的期望为0
            for (int j = 0; j <= n; j++) {
                matrix[u-1][j] = 0;
            }
            matrix[u-1][u-1] = 1;
            matrix[u-1][n] = 0;
        } else {
            matrix[u-1][u-1] = 1;
            for (int v : adj[u]) {
                matrix[u-1][v-1] = -1.0 / degree[u];
            }
            matrix[u-1][n] = 1;  // 常数项为1
        }
    }
    
    gaussian_elimination(n);
    
    cout << "从" << s << "到" << t << "的期望步数: " << matrix[s-1][n] / matrix[s-1][s-1] << endl;
    
    return 0;
}
```

## 4. 树上博弈问题

### 问题4：树上取石子游戏

**题目描述**：
在一棵树上，每个节点有一定数量的石子，两个玩家轮流操作，每次可以从一个节点取走任意个石子，或把石子移动到相邻节点，不能操作者输。

**解法**：
使用SG函数。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_set>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int stones[MAXN];
int sg[MAXN];

int calculateSG(int u, int parent) {
    if (sg[u] != -1) return sg[u];
    
    if (adj[u].size() == 1 && parent != 0) {
        // 叶子节点
        return sg[u] = stones[u] % 2;  // 叶子节点的SG值
    }
    
    // 计算当前状态的所有后继状态的SG值
    unordered_set<int> next_sg;
    
    // 操作1：取走石子
    for (int take = 1; take <= stones[u]; take++) {
        next_sg.insert((stones[u] - take) % 2);
    }
    
    // 操作2：移动石子到子节点
    if (!adj[u].empty()) {
        for (int child : adj[u]) {
            if (child != parent) {
                int child_sg = calculateSG(child, u);
                // 这里需要更复杂的计算，考虑移动石子后的整体SG值
                // 简化版本：移动k个石子到SG值为x的子树
                for (int move = 1; move <= stones[u]; move++) {
                    // 实际上需要考虑整个子树的状态变化
                    // 这是一个复杂的博弈问题，需要更深入的分析
                }
            }
        }
    }
    
    // 计算mex
    int mex = 0;
    while (next_sg.count(mex)) mex++;
    
    return sg[u] = mex;
}

// 更标准的树上Nim游戏
int subtree_sg[MAXN];

int calculateTreeSG(int u, int parent) {
    int result = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            result ^= calculateTreeSG(v, u);
        }
    }
    
    // 当前节点的贡献
    result ^= stones[u];
    
    return result;
}

    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> stones[i];
        sg[i] = -1;
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    int root_sg = calculateTreeSG(1, 0);
    
    if (root_sg != 0) {
        cout << "先手必胜" << endl;
    } else {
        cout << "后手必胜" << endl;
    }
    
    return 0;
}
```

## 5. 树上机器学习应用

### 问题5：树上聚类问题

**题目描述**：
在一棵树上，每个节点有特征向量，将节点分成k个类，使得类内距离最小，类间距离最大。

**解法**：
树上K-means变种。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <random>
using namespace std;

const int MAXN = 1e3 + 5;
const int MAXD = 10;  // 特征维度
const int MAXK = 20;
const int MAX_ITER = 100;
const double EPS = 1e-6;

vector<int> adj[MAXN];
vector<double> features[MAXN];  // 每个节点的特征向量
int n, k, d;
int cluster[MAXN];  // 每个节点的聚类标签
vector<double> centroids[MAXK];  // 聚类中心

double euclidean_distance(const vector<double>& a, const vector<double>& b) {
    double sum = 0;
    for (int i = 0; i < d; i++) {
        sum += (a[i] - b[i]) * (a[i] - b[i]);
    }
    return sqrt(sum);
}

// 计算树上距离（最短路径长度）
int tree_distance[MAXN][MAXN];
bool visited[MAXN];

void bfs_distance(int start) {
    fill(visited, visited + n + 1, false);
    vector<int> queue = {start};
    visited[start] = true;
    tree_distance[start][start] = 0;
    
    while (!queue.empty()) {
        int u = queue[0];
        queue.erase(queue.begin());
        
        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                tree_distance[start][v] = tree_distance[start][u] + 1;
                queue.push_back(v);
            }
        }
    }
}

void initialize_centroids() {
    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<> dis(1, n);
    
    for (int i = 0; i < k; i++) {
        int node = dis(gen);
        centroids[i] = features[node];
    }
}

double assign_clusters() {
    double total_cost = 0;
    
    for (int i = 1; i <= n; i++) {
        double min_dist = 1e20;
        int best_cluster = 0;
        
        for (int j = 0; j < k; j++) {
            double dist = euclidean_distance(features[i], centroids[j]);
            if (dist < min_dist) {
                min_dist = dist;
                best_cluster = j;
            }
        }
        
        cluster[i] = best_cluster;
        total_cost += min_dist;
    }
    
    return total_cost;
}

void update_centroids() {
    vector<int> cluster_size(k, 0);
    vector<vector<double>> new_centroids(k, vector<double>(d, 0));
    
    for (int i = 1; i <= n; i++) {
        int c = cluster[i];
        cluster_size[c]++;
        for (int j = 0; j < d; j++) {
            new_centroids[c][j] += features[i][j];
        }
    }
    
    for (int i = 0; i < k; i++) {
        if (cluster_size[i] > 0) {
            for (int j = 0; j < d; j++) {
                centroids[i][j] = new_centroids[i][j] / cluster_size[i];
            }
        }
    }
}

double tree_k_means() {
    initialize_centroids();
    
    double prev_cost = 1e20;
    for (int iter = 0; iter < MAX_ITER; iter++) {
        double cost = assign_clusters();
        update_centroids();
        
        if (abs(prev_cost - cost) < EPS) {
            break;
        }
        prev_cost = cost;
    }
    
    return prev_cost;
}

// 树约束的聚类：同一聚类的节点在树上应连通
bool is_connected(int cluster_id) {
    vector<int> nodes_in_cluster;
    for (int i = 1; i <= n; i++) {
        if (cluster[i] == cluster_id) {
            nodes_in_cluster.push_back(i);
        }
    }
    
    if (nodes_in_cluster.size() <= 1) return true;
    
    // 检查这些节点在原树中是否连通
    vector<bool> included(n + 1, false);
    for (int node : nodes_in_cluster) {
        included[node] = true;
    }
    
    // BFS检查连通性
    vector<bool> visited(n + 1, false);
    int start = nodes_in_cluster[0];
    visited[start] = true;
    vector<int> queue = {start};
    
    while (!queue.empty()) {
        int u = queue[0];
        queue.erase(queue.begin());
        
        for (int v : adj[u]) {
            if (included[v] && !visited[v]) {
                visited[v] = true;
                queue.push_back(v);
            }
        }
    }
    
    for (int node : nodes_in_cluster) {
        if (!visited[node]) {
            return false;
        }
    }
    
    return true;
}

int main() {
    cin >> n >> k >> d;
    
    for (int i = 1; i <= n; i++) {
        features[i].resize(d);
        for (int j = 0; j < d; j++) {
            cin >> features[i][j];
        }
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    // 预计算树上距离
    for (int i = 1; i <= n; i++) {
        bfs_distance(i);
    }
    
    double cost = tree_k_means();
    
    cout << "聚类成本: " << cost << endl;
    cout << "聚类分配: ";
    for (int i = 1; i <= n; i++) {
        cout << "节点" << i << "属于聚类" << cluster[i] << " ";
    }
    cout << endl;
    
    // 验证连通性
    for (int i = 0; i < k; i++) {
        if (is_connected(i)) {
            cout << "聚类" << i << "在树上是连通的" << endl;
        } else {
            cout << "聚类" << i << "在树上不连通" << endl;
        }
    }
    
    return 0;
}
```

## 6. 树上优化问题

### 问题6：树上设施选址

**题目描述**：
在一棵树上选择k个节点作为服务中心，使得所有节点到最近服务中心的距离和最小。

**解法**：
树形DP+决策单调性优化。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 3e2 + 5;
const int MAXK = 1e2 + 5;
const long long INF = 1e18;

vector<int> adj[MAXN];
int sz[MAXN];
long long dp[MAXN][MAXK][2];  // dp[u][j][0/1] 表示以u为根的子树选j个点，u是否被选中的最小代价

void dfs_size(int u, int parent) {
    sz[u] = 1;
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_size(v, u);
            sz[u] += sz[v];
        }
    }
}

void dfs(int u, int parent) {
    sz[u] = 1;
    
    // 初始化
    for (int j = 0; j < MAXK; j++) {
        dp[u][j][0] = dp[u][j][1] = INF;
    }
    
    // 叶子节点
    if (adj[u].size() == 1 && parent != 0) {
        dp[u][0][0] = 0;
        dp[u][1][1] = 0;
        return;
    }
    
    // 处理子树
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // 合并子树信息
            vector<long long> new_dp0(MAXK, INF);
            vector<long long> new_dp1(MAXK, INF);
            
            for (int j = 0; j < MAXK; j++) {
                new_dp0[j] = dp[u][j][0];
                new_dp1[j] = dp[u][j][1];
            }
            
            for (int j = 0; j < MAXK; j++) {
                for (int k = 0; k < MAXK && j + k < MAXK; k++) {
                    // u未被选中，v可选可不选
                    if (dp[u][j][0] != INF && dp[v][k][0] != INF) {
                        new_dp0[j + k] = min(new_dp0[j + k], dp[u][j][0] + dp[v][k][0]);
                    }
                    if (dp[u][j][0] != INF && dp[v][k][1] != INF) {
                        new_dp0[j + k] = min(new_dp0[j + k], dp[u][j][0] + dp[v][k][1]);
                    }
                    
                    // u被选中，v可选可不选
                    if (dp[u][j][1] != INF && dp[v][k][0] != INF) {
                        new_dp1[j + k] = min(new_dp1[j + k], dp[u][j][1] + dp[v][k][0]);
                    }
                    if (dp[u][j][1] != INF && dp[v][k][1] != INF) {
                        new_dp1[j + k] = min(new_dp1[j + k], dp[u][j][1] + dp[v][k][1]);
                    }
                }
            }
            
            for (int j = 0; j < MAXK; j++) {
                dp[u][j][0] = new_dp0[j];
                dp[u][j][1] = new_dp1[j];
            }
            
            sz[u] += sz[v];
        }
    }
    
    // 考虑u到父节点的距离
    if (parent != 0) {
        for (int j = MAXK - 1; j >= 1; j--) {
            if (dp[u][j - 1][1] != INF) {
                dp[u][j][1] = min(dp[u][j][1], dp[u][j - 1][1]);
            }
        }
    }
}

int main() {
    int n, k;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_size(1, 0);
    dfs(1, 0);
    
    long long result = min(dp[1][k][0], dp[1][k][1]);
    cout << "最小总距离: " << result << endl;
```
