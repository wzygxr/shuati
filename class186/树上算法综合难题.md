# 树上算法综合难题

## 1. 树链剖分与LCA结合

### 问题1：树链剖分路径查询

**题目描述**：
在一棵树上支持以下操作：
1. `update_path u v val`：将u到v路径上所有点的权值增加val
2. `query_path u v`：查询u到v路径上所有点的权值和
3. `update_subtree u val`：将以u为根的子树所有点的权值增加val
4. `query_subtree u`：查询以u为根的子树权值和

**解法**：
使用树链剖分结合线段树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];  // top:链头，id:新编号，rev:新编号对应原节点
int w[MAXN], wt[MAXN];  // 点权，新编号对应的点权
int cnt = 0;

struct SegmentTree {
    vector<long long> tree, lazy;
    int n;
    
    SegmentTree(int _n) : n(_n) {
        tree.resize(4 * n, 0);
        lazy.resize(4 * n, 0);
    }
    
    void push_down(int node, int start, int end) {
        if (lazy[node] != 0) {
            tree[node] += lazy[node] * (end - start + 1);
            if (start != end) {
                lazy[node * 2] += lazy[node];
                lazy[node * 2 + 1] += lazy[node];
            }
            lazy[node] = 0;
        }
    }
    
    void update(int node, int start, int end, int l, int r, long long val) {
        push_down(node, start, end);
        if (start > r || end < l) return;
        if (l <= start && end <= r) {
            lazy[node] += val;
            push_down(node, start, end);
            return;
        }
        int mid = (start + end) / 2;
        update(node * 2, start, mid, l, r, val);
        update(node * 2 + 1, mid + 1, end, l, r, val);
        push_up(node, start, end);
    }
    
    void push_up(int node, int start, int end) {
        push_down(node * 2, start, (start + end) / 2);
        push_down(node * 2 + 1, (start + end) / 2 + 1, end);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    
    long long query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return 0;
        push_down(node, start, end);
        if (l <= start && end <= r) return tree[node];
        int mid = (start + end) / 2;
        return query(node * 2, start, mid, l, r) + 
               query(node * 2 + 1, mid + 1, end, l, r);
    }
    
    void update_range(int l, int r, long long val) {
        if (l > r) return;
        update(1, 1, n, l, r, val);
    }
    
    long long query_range(int l, int r) {
        if (l > r) return 0;
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int fa, int d) {  // 第一遍dfs，求出size、重儿子、深度
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {  // 第二遍dfs，求出top、id
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    wt[cnt] = w[u];
    
    if (son[u]) dfs2(son[u], t);  // 先遍历重儿子，保证重链在线段树中连续
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);  // 轻儿子的链头是自己
        }
    }
}

SegmentTree segtree(MAXN);

// 查询路径u到v的权值和
long long query_path(int u, int v) {
    long long res = 0;
    while (top[u] != top[v]) {  // 当两个点不在同一条重链上
        if (depth[top[u]] < depth[top[v]]) swap(u, v);  // 让u所在重链的链头深度更深
        res += segtree.query_range(id[top[u]], id[u]);  // 查询u到其链头的权值和
        u = parent[top[u]];  // u跳到链头的父节点
    }
    if (depth[u] > depth[v]) swap(u, v);
    res += segtree.query_range(id[u], id[v]);  // 查询最后一条重链上的值
    return res;
}

// 更新路径u到v的权值
void update_path(int u, int v, long long val) {
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        segtree.update_range(id[top[u]], id[u], val);
        u = parent[top[u]];
    }
    if (depth[u] > depth[v]) swap(u, v);
    segtree.update_range(id[u], id[v], val);
}

int main() {
    int n, m, r;
    cin >> n >> m >> r;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(r, 0, 1);
    dfs2(r, r);
    
    // 初始化线段树
    for (int i = 1; i <= n; i++) {
        segtree.update_range(id[i], id[i], wt[i]);
    }
    
    for (int i = 0; i < m; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // 路径加
            int u, v;
            long long val;
            cin >> u >> v >> val;
            update_path(u, v, val);
        } else if (op == 2) {  // 路径查询
            int u, v;
            cin >> u >> v;
            cout << query_path(u, v) << endl;
        } else if (op == 3) {  // 子树加
            int u;
            long long val;
            cin >> u >> val;
            // 子树在dfs序中是连续的
            segtree.update_range(id[u], id[u] + size[u] - 1, val);
        } else if (op == 4) {  // 子树查询
            int u;
            cin >> u;
            cout << segtree.query_range(id[u], id[u] + size[u] - 1) << endl;
        }
    }
    
    return 0;
}
```

## 2. 虚树（Virtual Tree）问题

### 问题2：虚树构建与应用

**题目描述**：
给定一棵树和一个关键点集合，构建虚树（只保留关键点和它们之间的LCA），然后在虚树上解决问题。

**解法**：
通过DFS序排序关键点，然后利用LCA构建虚树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <stack>
#include <set>
using namespace std;

const int MAXN = 2e5 + 5;
const int LOG = 20;

vector<int> adj[MAXN];
int depth[MAXN], fa[MAXN][LOG], dfn[MAXN], timer = 0;

void dfs(int u, int parent) {
    dfn[u] = ++timer;
    depth[u] = depth[parent] + 1;
    fa[u][0] = parent;
    
    for (int k = 1; k < LOG; k++) {
        fa[u][k] = fa[fa[u][k-1]][k-1];
    }
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
        }
    }
}

int getLCA(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    
    int diff = depth[u] - depth[v];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            u = fa[u][k];
        }
    }
    
    if (u == v) return u;
    
    for (int k = LOG - 1; k >= 0; k--) {
        if (fa[u][k] != fa[v][k]) {
            u = fa[u][k];
            v = fa[v][k];
        }
    }
    
    return fa[u][0];
}

// 构建虚树
vector<int> virtual_tree(vector<int> nodes) {
    // 按DFS序排序
    sort(nodes.begin(), nodes.end(), [](int a, int b) {
        return dfn[a] < dfn[b];
    });
    
    // 去重
    nodes.erase(unique(nodes.begin(), nodes.end()), nodes.end());
    
    // 添加必要的LCA节点
    vector<int> temp = nodes;
    for (int i = 1; i < temp.size(); i++) {
        int lca = getLCA(temp[i-1], temp[i]);
        nodes.push_back(lca);
    }
    
    // 再次排序并去重
    sort(nodes.begin(), nodes.end(), [](int a, int b) {
        return dfn[a] < dfn[b];
    });
    nodes.erase(unique(nodes.begin(), nodes.end()), nodes.end());
    
    // 构建虚树的边
    stack<int> stk;
    vector<pair<int, int>> virtual_edges;  // 虚树的边
    
    for (int u : nodes) {
        while (!stk.empty()) {
            int v = stk.top();
            if (depth[v] <= depth[getLCA(u, v)]) break;
            stk.pop();
            if (!stk.empty()) {
                int w = stk.top();
                if (depth[w] < depth[getLCA(u, v)]) {
                    virtual_edges.push_back({w, v});
                }
            } else {
                virtual_edges.push_back({getLCA(u, v), v});
            }
        }
        stk.push(u);
    }
    
    while (stk.size() > 1) {
        int v = stk.top(); stk.pop();
        virtual_edges.push_back({stk.top(), v});
    }
    
    return nodes;  // 返回虚树中的节点
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0);
    
    int k;
    cin >> k;
    vector<int> important_nodes(k);
    for (int i = 0; i < k; i++) {
        cin >> important_nodes[i];
    }
    
    vector<int> vt_nodes = virtual_tree(important_nodes);
    
    cout << "虚树中的节点: ";
    for (int node : vt_nodes) {
        cout << node << " ";
    }
    cout << endl;
    
    return 0;
}
```

## 3. 点分治问题

### 问题3：树的重心与点分治

**题目描述**：
在一棵树上，对于每个节点，计算包含该节点的路径数量，使得路径上所有点的权值异或和为给定值。

**解法**：
使用点分治算法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int w[MAXN];  // 节点权值
bool vis[MAXN];
int sz[MAXN];  // 子树大小
int max_part[MAXN];  // 删除该点后最大子树大小
int root, tot;  // 分治中心，总节点数
long long result[MAXN];

void getRoot(int u, int parent) {  // 找树的重心
    sz[u] = 1;
    max_part[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getRoot(v, u);
            sz[u] += sz[v];
            max_part[u] = max(max_part[u], sz[v]);
        }
    }
    
    max_part[u] = max(max_part[u], tot - sz[u]);
    if (max_part[u] < max_part[root]) root = u;
}

unordered_map<int, int> cnt;  // 记录路径异或值的出现次数
vector<int> path_xor;  // 当前路径的异或值
int target_xor;  // 目标异或值

void getDis(int u, int parent, int cur_xor) {
    path_xor.push_back(cur_xor ^ w[u]);
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getDis(v, u, cur_xor ^ w[u]);
        }
    }
}

long long calc(int u, int init_xor) {
    path_xor.clear();
    getDis(u, 0, init_xor);
    
    long long res = 0;
    for (int xor_val : path_xor) {
        res += cnt[target_xor ^ xor_val];
    }
    
    for (int xor_val : path_xor) {
        cnt[xor_val]++;
    }
    
    return res;
}

void solve(int u) {
    cnt.clear();
    cnt[0] = 1;  // 空路径异或值为0
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            result[u] += calc(v, w[u]);
            
            // 将子树v中的路径加入到cnt中
            path_xor.clear();
            getDis(v, u, w[u]);
            for (int xor_val : path_xor) {
                cnt[xor_val]++;
            }
        }
    }
    
    // 递归处理子树
    vis[u] = true;
    for (int v : adj[u]) {
        if (!vis[v]) {
            tot = sz[v];
            root = 0;
            getRoot(v, 0);
            solve(root);
        }
    }
}

int main() {
    int n;
    cin >> n >> target_xor;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    max_part[0] = tot = n;
    getRoot(1, 0);
    solve(root);
    
    for (int i = 1; i <= n; i++) {
        cout << "节点" << i << "的满足条件路径数: " << result[i] << endl;
    }
    
    return 0;
}
```

## 4. 在线算法难题

### 问题4：动态LCA问题

**题目描述**：
在一棵动态树上支持加边和LCA查询操作。

**解法**：
使用Link-Cut Tree（LCT）。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

struct LCT {
    int ch[MAXN][2], fa[MAXN], rev[MAXN];
    
    bool isRoot(int x) {
        return ch[fa[x]][0] != x && ch[fa[x]][1] != x;
    }
    
    void pushDown(int x) {
        if (rev[x]) {
            rev[x] = 0;
            rev[ch[x][0]] ^= 1;
            rev[ch[x][1]] ^= 1;
            swap(ch[x][0], ch[x][1]);
        }
    }
    
    void rotate(int x) {
        int y = fa[x], z = fa[y], k = ch[y][1] == x;
        if (!isRoot(y)) ch[z][ch[z][1] == y] = x;
        ch[y][k] = ch[x][k^1], fa[ch[x][k^1]] = y;
        ch[x][k^1] = y, fa[y] = x, fa[x] = z;
    }
    
    int st[MAXN], top;
    void splay(int x) {
        top = 0;
        st[++top] = x;
        for (int i = x; !isRoot(i); i = fa[i]) st[++top] = fa[i];
        while (top) pushDown(st[top--]);
        
        while (!isRoot(x)) {
            int y = fa[x], z = fa[y];
            if (!isRoot(y)) rotate((ch[y][0] == x) ^ (ch[z][0] == y) ? x : y);
            rotate(x);
        }
    }
    
    void access(int x) {
        for (int y = 0; x; y = x, x = fa[x]) {
            splay(x);
            ch[x][1] = y;
        }
    }
    
    void makeRoot(int x) {
        access(x);
        splay(x);
        rev[x] ^= 1;
    }
    
    int findRoot(int x) {
        access(x);
        splay(x);
        while (ch[x][0]) {
            pushDown(x);
            x = ch[x][0];
        }
        splay(x);
        return x;
    }
    
    void split(int x, int y) {
        makeRoot(x);
        access(y);
        splay(y);
    }
    
    void link(int x, int y) {
        makeRoot(x);
        if (findRoot(y) != x) fa[x] = y;
    }
    
    void cut(int x, int y) {
        split(x, y);
        if (ch[y][0] == x && !ch[x][1]) {
            ch[y][0] = fa[x] = 0;
        }
    }
    
    // 查询x和y的LCA
    int getLCA(int x, int y) {
        if (findRoot(x) != findRoot(y)) return -1;  // 不在同一棵树
        
        access(x);
        splay(x);
        int last = 0;
        for (int now = x; now; now = ch[now][0]) {
            pushDown(now);
            last = now;
        }
        
        access(y);
        splay(y);
        last = 0;
        for (int now = y; now; now = ch[now][0]) {
            pushDown(now);
            last = now;
        }
        
        // 实际的LCA获取需要更复杂的实现
        // 这里是简化版本，实际应用中需要更仔细的处理
        return last;
    }
} lct;

int main() {
    int n, q;
    cin >> n >> q;
    
    // 初始化：开始时每个节点是独立的
    for (int i = 1; i <= n; i++) {
        lct.rev[i] = 0;
    }
    
    for (int i = 0; i < q; i++) {
        int op, x, y;
        cin >> op >> x >> y;
        
        if (op == 1) {  // 加边
            lct.link(x, y);
        } else if (op == 2) {  // 断边
            lct.cut(x, y);
        } else if (op == 3) {  // 查询LCA
            int lca = lct.getLCA(x, y);
            if (lca == -1) {
                cout << "节点" << x << "和" << y << "不在同一棵树中" << endl;
            } else {
                cout << "LCA(" << x << "," << y << ") = " << lca << endl;
            }
        }
    }
    
    return 0;
}
```

## 5. 综合应用：树上游戏问题

### 问题5：树上博弈问题

**题目描述**：
在一棵树上，两个玩家轮流选择节点，每次选择的节点必须与之前选择的节点相邻。不能选择时输。求先手是否必胜。

**解法**：
使用树形DP和SG函数。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e4 + 5;

vector<int> adj[MAXN];
int sg[MAXN];  // SG函数值
bool visited[MAXN];

int calculateSG(int u, int parent) {
    if (sg[u] != -1) return sg[u];
    
    vector<int> subSG;
    for (int v : adj[u]) {
        if (v != parent) {
            subSG.push_back(calculateSG(v, u));
        }
    }
    
    // 计算mex值
    sort(subSG.begin(), subSG.end());
    subSG.erase(unique(subSG.begin(), subSG.end()), subSG.end());
    
    int mex = 0;
    for (int val : subSG) {
        if (val == mex) mex++;
        else if (val > mex) break;
    }
    
    // 对于树上的移动游戏，当前状态的SG值是所有子树SG值的异或
    int result = 0;
    for (int val : subSG) {
        result ^= (val + 1);  // 子树的SG值+1后异或
    }
    
    return sg[u] = result;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    fill(sg, sg + MAXN, -1);
    
    int root_sg = calculateSG(1, 0);
    
    if (root_sg != 0) {
        cout << "先手必胜" << endl;
    } else {
        cout << "后手必胜" << endl;
    }
    
    return 0;
}
```
