# 树上动态规划高级优化

## 1. 树上背包问题优化

### 问题1：树上分组背包优化

**题目描述**：
在一棵树上，每个节点属于一个分组，每个分组只能选择一个节点，选择节点需要花费代价，求在总代价不超过V的情况下，获得的最大价值。

**解法**：
使用树上分组背包+长链剖分优化。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXV = 1e3 + 5;

vector<int> adj[MAXN];
int w[MAXN], v[MAXN], group[MAXN];  // 重量、价值、分组
int dp[MAXN][MAXV];  // dp[u][j] 表示以u为根的子树中，容量为j的最大价值
int sz[MAXN], hson[MAXN];  // 子树大小、重儿子

// 存储每个分组的节点信息
map<int, vector<pair<int, int>>> group_info[MAXN];  // {组号 -> {(cost, value)}}

void dfs_size(int u, int parent) {
    sz[u] = 1;
    hson[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_size(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[hson[u]]) {
                hson[u] = v;
            }
        }
    }
}

void merge_dp(int u, int v, int capacity) {
    // 将v子树的信息合并到u
    for (int j = capacity; j >= 0; j--) {
        for (int k = 0; k <= j; k++) {
            dp[u][j] = max(dp[u][j], dp[u][j-k] + dp[v][k]);
        }
    }
}

void dfs(int u, int parent, int capacity) {
    // 初始化当前节点
    for (int j = 0; j <= capacity; j++) {
        dp[u][j] = 0;
    }
    
    // 先处理重儿子
    if (hson[u]) {
        dfs(hson[u], u, capacity);
        // 直接继承重儿子的DP值
        for (int j = 0; j <= capacity; j++) {
            dp[u][j] = dp[hson[u]][j];
        }
    }
    
    // 处理轻儿子
    for (int v : adj[u]) {
        if (v != parent && v != hson[u]) {
            dfs(v, u, capacity);
            merge_dp(u, v, capacity);
        }
    }
    
    // 考虑当前节点的选择
    if (v[u] <= capacity) {
        // 当前节点的组是否已经选择
        bool group_selected = false;
        for (auto& item : group_info[parent][group[u]]) {
            if (item.first <= v[u]) {
                group_selected = true;
                break;
            }
        }
        
        if (!group_selected) {
            for (int j = capacity; j >= v[u]; j--) {
                dp[u][j] = max(dp[u][j], dp[u][j - v[u]] + w[u]);
            }
        }
    }
    
    // 处理分组约束
    for (auto& g : group_info[u]) {
        int group_id = g.first;
        vector<pair<int, int>>& items = g.second;
        
        // 分组背包
        for (int j = capacity; j >= 0; j--) {
            for (auto& item : items) {
                int cost = item.first, val = item.second;
                if (j >= cost) {
                    dp[u][j] = max(dp[u][j], dp[u][j - cost] + val);
                }
            }
        }
    }
}

int main() {
    int n, V;
    cin >> n >> V;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i] >> v[i] >> group[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    // 构建分组信息
    for (int i = 1; i <= n; i++) {
        group_info[i][group[i]].push_back({v[i], w[i]});
    }
    
    dfs_size(1, 0);
    dfs(1, 0, V);
    
    cout << "最大价值: " << dp[1][V] << endl;
    
    return 0;
}
```

## 2. 树上斜率优化DP

### 问题2：树上斜率优化背包

**题目描述**：
在一棵树上，每个节点有权值和体积，选择节点使得总价值最大，满足背包约束，使用斜率优化DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <deque>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXV = 1e5 + 5;

vector<int> adj[MAXN];
int w[MAXN], v[MAXN];  // 价值和体积
long long dp[MAXN][2];  // 简化版本的DP数组

// 斜率优化队列
struct Point {
    long long x, y;  // x是体积，y是价值
    Point(long long _x = 0, long long _y = 0) : x(_x), y(_y) {}
    
    Point operator-(const Point& other) const {
        return Point(x - other.x, y - other.y);
    }
    
    long long cross(const Point& other) const {
        return x * other.y - y * other.x;
    }
};

deque<Point> hull;  // 凸包

// 检查三点是否构成下凸包
bool check(Point a, Point b, Point c) {
    return (b - a).cross(c - b) >= 0;
}

// 斜率优化：查询最大值
long long query(long long slope) {
    if (hull.size() < 2) return hull.empty() ? 0 : hull[0].y;
    
    while (hull.size() >= 2 && 
           hull[1].y - hull[0].y <= slope * (hull[1].x - hull[0].x)) {
        hull.pop_front();
    }
    
    return hull[0].y - slope * hull[0].x;
}

// 添加新点到凸包
void add_point(long long x, long long y) {
    Point new_pt(x, y);
    while (hull.size() >= 2 && 
           !check(hull[hull.size()-2], hull[hull.size()-1], new_pt)) {
        hull.pop_back();
    }
    hull.push_back(new_pt);
}

void dfs_slop(int u, int parent) {
    dp[u][0] = 0;  // 不选u
    dp[u][1] = w[u];  // 选u
    
    // 构建凸包
    hull.clear();
    add_point(0, 0);  // 不选的情况
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_slop(v, u);
            
            // 合并子树信息
            long long child_not_taken = dp[v][0];
            long long child_taken = dp[v][1];
            
            // 更新当前节点
            dp[u][0] = max(dp[u][0], child_not_taken);
            dp[u][1] = max(dp[u][1], child_taken + w[u]);
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i] >> v[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_slop(1, 0);
    
    cout << "最大价值: " << max(dp[1][0], dp[1][1]) << endl;
    
    return 0;
}
```

## 3. 树上插头DP

### 问题3：树上连通块计数

**题目描述**：
在一棵树上，计算所有可能的连通块的数量（选择节点使得选择的节点形成连通子图）。

**解法**：
使用树上插头DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中，u是否在连通块中的方案数

void dfs(int u, int parent) {
    dp[u][0] = 1;  // u不选
    dp[u][1] = 1;  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不选：v可以是任何状态
            long long old_dp0 = dp[u][0];
            dp[u][0] = (old_dp0 * (dp[v][0] + dp[v][1])) % MOD;
            
            // u选：v可以选也可以不选
            dp[u][1] = (dp[u][1] * (dp[v][0] + dp[v][1])) % MOD;
        }
    }
}

// 更精确的连通块计数
long long dp_connected[MAXN][2];  // dp[u][0]不选u，dp[u][1]选u且u在某个连通块中

void dfs_connected(int u, int parent) {
    dp_connected[u][0] = 1;  // 不选u
    dp_connected[u][1] = 1;  // 选u，u单独成为一个连通块
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_connected(v, u);
            
            // u不选
            long long temp0 = dp_connected[u][0];
            dp_connected[u][0] = (temp0 * (dp_connected[v][0] + dp_connected[v][1])) % MOD;
            
            // u选，可以选择连接到子树或不连接
            long long temp1 = dp_connected[u][1];
            dp_connected[u][1] = (temp1 * dp_connected[v][0]) % MOD;  // 不连接到v子树
            dp_connected[u][1] = (dp_connected[u][1] + temp1 * dp_connected[v][1]) % MOD;  // 连接到v子树
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_connected(1, 0);
    
    // 总连通块数 = 选根节点的方案数 + 不选根节点的方案数
    long long result = (dp_connected[1][0] + dp_connected[1][1] - 1) % MOD;  // 减1排除空集
    if (result < 0) result += MOD;
    
    cout << "连通块数量: " << result << endl;
    
    return 0;
}
```

## 4. 树上数位DP

### 问题4：树上路径数字构造

**题目描述**：
在一棵树上，每条边有一个数字字符，从根到叶子路径形成数字，求满足特定条件（如能被k整除）的路径数量。

**解法**：
使用树上数位DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXK = 1e2 + 5;

vector<pair<int, int>> adj[MAXN];  // (子节点, 边上的数字)
int dp[MAXN][MAXK];  // dp[u][r] 表示在节点u，当前路径数字模k余r的方案数
int k;  // 整除数

void dfs_digit(int u, int parent, int current_mod) {
    // 初始化当前节点的DP值
    for (int r = 0; r < k; r++) {
        dp[u][r] = 0;
    }
    
    // 更新当前状态
    dp[u][current_mod] = 1;
    
    // 遍历子节点
    for (auto& edge : adj[u]) {
        int v = edge.first;
        int digit = edge.second;
        
        if (v != parent) {
            int new_mod = (current_mod * 10 + digit) % k;
            dfs_digit(v, u, new_mod);
            
            // 合并子树信息
            for (int r = 0; r < k; r++) {
                dp[u][r] = (dp[u][r] + dp[v][r]) % MOD;
            }
        }
    }
}

int main() {
    int n;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v, d;
        cin >> u >> v >> d;
        adj[u].push_back({v, d});
        adj[v].push_back({u, d});
    }
    
    dfs_digit(1, 0, 0);
    
    cout << "能被" << k << "整除的路径数: " << dp[1][0] << endl;
    
    return 0;
}
```

## 5. 树上概率DP优化

### 问题5：树上期望路径长度

**题目描述**：
在一棵树上，从根节点开始随机游走，求期望的路径长度。

**解法**：
使用树上概率DP+高斯消元优化。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const int MAXN = 1e3 + 5;
const double EPS = 1e-9;

vector<int> adj[MAXN];
double prob[MAXN][MAXN];  // 转移概率
double exp_len[MAXN];     // 期望路径长度
int degree[MAXN];         // 节点度数

void dfs_expectation(int u, int parent) {
    if (adj[u].size() == 1 && parent != -1) {
        // 叶子节点，期望长度为0
        exp_len[u] = 0;
        return;
    }
    
    double total = 0;
    int count = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_expectation(v, u);
            total += exp_len[v];
            count++;
        }
    }
    
    if (count > 0) {
        exp_len[u] = 1 + total / count;  // 期望长度 = 1 + 子节点期望平均值
    } else {
        exp_len[u] = 0;
    }
}

// 更精确的高斯消元解法
double matrix[MAXN][MAXN + 1];  // 增广矩阵

void solve_linear_system(int n) {
    for (int i = 0; i < n; i++) {
        // 找主元
        int max_row = i;
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > abs(matrix[max_row][i])) {
                max_row = j;
            }
        }
        
        if (max_row != i) {
            for (int j = 0; j <= n; j++) {
                swap(matrix[i][j], matrix[max_row][j]);
            }
        }
        
        // 消元
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > EPS) {
                double ratio = matrix[j][i] / matrix[i][i];
                for (int k = i; k <= n; k++) {
                    matrix[j][k] -= ratio * matrix[i][k];
                }
            }
        }
    }
    
    // 回代
    vector<double> result(n);
    for (int i = n - 1; i >= 0; i--) {
        result[i] = matrix[i][n];
        for (int j = i + 1; j < n; j++) {
            result[i] -= matrix[i][j] * result[j];
        }
        result[i] /= matrix[i][i];
    }
    
    for (int i = 0; i < n; i++) {
        exp_len[i + 1] = result[i];
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        degree[u]++;
        degree[v]++;
    }
    
    // 建立方程组
    // E[u] = 1 + sum(E[v]) / degree[u] 对于所有子节点v
    for (int u = 1; u <= n; u++) {
        if (adj[u].size() == 1 && u != 1) {  // 叶子节点
            for (int j = 0; j <= n; j++) {
                matrix[u-1][j] = 0;
            }
            matrix[u-1][u-1] = 1;
            matrix[u-1][n] = 0;  // 叶子节点期望为0
        } else {
            // E[u] - sum(E[v] / degree[u]) = 1
            for (int j = 0; j <= n; j++) {
                matrix[u-1][j] = 0;
            }
            matrix[u-1][u-1] = 1;
            
            for (int v : adj[u]) {
                matrix[u-1][v-1] = -1.0 / degree[u];
            }
            matrix[u-1][n] = 1;
        }
    }
    
    solve_linear_system(n);
    
    cout << "根节点期望路径长度: " << exp_len[1] << endl;
    
    return 0;
}
```

## 6. 树上区间DP

### 问题6：树上区间合并

**题目描述**：
在一棵树上，每个节点有一个区间，合并相邻节点的区间，求最优合并顺序的最大收益。

**解法**：
使用树上区间DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e3 + 5;

vector<int> adj[MAXN];
pair<int, int> intervals[MAXN];  // [left, right]
long long dp[MAXN][MAXN];  // 在子树u中，合并后区间为[l,r]的最大收益

// 计算合并两个区间的收益
long long merge_value(pair<int, int> a, pair<int, int> b) {
    return (long long)(a.second - a.first + 1) * (b.second - b.first + 1);
}

long long dfs_interval(int u, int parent) {
    // 叶子节点
    if (adj[u].size() == 1 && parent != -1) {
        return 0;
    }
    
    long long max_profit = 0;
    
    // 遍历所有子节点，尝试合并
    vector<int> children;
    for (int v : adj[u]) {
        if (v != parent) {
            children.push_back(v);
            max_profit += dfs_interval(v, u);  // 子树内部的收益
        }
    }
    
    // 尝试在当前节点合并子树
    if (children.size() >= 2) {
        for (int i = 0; i < children.size(); i++) {
            for (int j = i + 1; j < children.size(); j++) {
                int v1 = children[i], v2 = children[j];
                long long merge_profit = merge_value(intervals[v1], intervals[v2]);
                max_profit = max(max_profit, merge_profit);
            }
        }
    }
    
    return max_profit;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> intervals[i].first >> intervals[i].second;
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    long long result = dfs_interval(1, -1);
    
    cout << "最大合并收益: " << result << endl;
    
    return 0;
}
```
