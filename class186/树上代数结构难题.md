# 树上代数结构难题

## 1. 树上群论应用

### 问题1：树的自同构群计数

**题目描述**：
给定一棵树，计算其自同构群的大小（即树的对称性数量）。

**解法**：
使用树哈希和递归计算。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
vector<long long> subtree_hash;
vector<long long> subtree_count;  // 每个子树的自同构数

// 计算阶乘
long long fact[MAXN];
void init_fact() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        fact[i] = (fact[i-1] * i) % MOD;
    }
}

// 计算乘法逆元
long long pow_mod(long long a, long long b) {
    long long res = 1;
    while (b) {
        if (b & 1) res = (res * a) % MOD;
        a = (a * a) % MOD;
        b >>= 1;
    }
    return res;
}

long long inv(long long a) {
    return pow_mod(a, MOD - 2);
}

// 计算多重集的排列数
long long multi_perm(const vector<int>& counts) {
    int total = 0;
    long long result = 1;
    
    for (int cnt : counts) {
        total += cnt;
        result = (result * inv(fact[cnt])) % MOD;
    }
    
    result = (result * fact[total]) % MOD;
    return result;
}

pair<long long, long long> dfs(int u, int parent) {
    vector<pair<long long, long long>> children_info;  // {hash, count}
    
    for (int v : adj[u]) {
        if (v != parent) {
            auto child_info = dfs(v, u);
            children_info.push_back(child_info);
        }
    }
    
    // 按哈希值排序，便于比较子树结构
    sort(children_info.begin(), children_info.end());
    
    // 计算当前子树的哈希值
    long long hash = 1;
    for (auto& info : children_info) {
        hash = (hash * 31 + info.first + 1) % MOD;
    }
    
    // 计算相同子树的分组
    vector<int> group_sizes;
    if (!children_info.empty()) {
        int current_size = 1;
        for (int i = 1; i < children_info.size(); i++) {
            if (children_info[i].first == children_info[i-1].first) {
                current_size++;
            } else {
                group_sizes.push_back(current_size);
                current_size = 1;
            }
        }
        group_sizes.push_back(current_size);
    }
    
    // 计算当前节点的自同构数
    long long count = 1;
    for (int sz : group_sizes) {
        // 对于sz个相同子树，它们可以任意排列
        count = (count * fact[sz]) % MOD;
    }
    
    // 乘以每个子树内部的自同构数
    for (auto& info : children_info) {
        count = (count * info.second) % MOD;
    }
    
    return {hash, count};
}

long long countTreeAutomorphisms(int n) {
    init_fact();
    auto result = dfs(1, 0);
    return result.second;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "树的自同构数: " << countTreeAutomorphisms(n) << endl;
    
    return 0;
}
```

## 2. 树上多项式插值

### 问题2：树上路径多项式

**题目描述**：
在一棵树上，每个节点有一个值，对于每次查询(u,v)，计算从u到v路径上节点值构成的多项式在给定点x处的值。

**解法**：
使用树链剖分+线段树维护多项式。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e4 + 5;
const int MOD = 1e9 + 7;
const int MAX_DEGREE = 100;  // 多项式最大度数

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int w[MAXN];  // 节点值
int cnt = 0;

// 多项式结构
struct Poly {
    vector<long long> coef;  // 系数数组，coef[i]是x^i的系数
    
    Poly(int degree = 0) : coef(degree + 1, 0) {}
    
    long long evaluate(long long x) {
        long long result = 0, power = 1;
        for (long long c : coef) {
            result = (result + c * power) % MOD;
            power = (power * x) % MOD;
        }
        return result;
    }
    
    // 多项式加法
    Poly operator+(const Poly& other) const {
        Poly result(max(coef.size(), other.coef.size()) - 1);
        for (int i = 0; i < result.coef.size(); i++) {
            long long a = (i < coef.size()) ? coef[i] : 0;
            long long b = (i < other.coef.size()) ? other.coef[i] : 0;
            result.coef[i] = (a + b) % MOD;
        }
        return result;
    }
    
    // 多项式减法
    Poly operator-(const Poly& other) const {
        Poly result(max(coef.size(), other.coef.size()) - 1);
        for (int i = 0; i < result.coef.size(); i++) {
            long long a = (i < coef.size()) ? coef[i] : 0;
            long long b = (i < other.coef.size()) ? other.coef[i] : 0;
            result.coef[i] = ((a - b) % MOD + MOD) % MOD;
        }
        return result;
    }
};

struct SegmentTree {
    vector<Poly> tree;
    int n;
    
    SegmentTree(int _n) : n(_n) {
        tree.resize(4 * n, Poly(0));
    }
    
    void update(int node, int start, int end, int idx, const Poly& val) {
        if (start == end) {
            tree[node] = val;
            return;
        }
        
        int mid = (start + end) / 2;
        if (idx <= mid) {
            update(node * 2, start, mid, idx, val);
        } else {
            update(node * 2 + 1, mid + 1, end, idx, val);
        }
        
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    
    Poly query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return Poly(-1);  // 返回零多项式
        if (l <= start && end <= r) return tree[node];
        
        int mid = (start + end) / 2;
        return query(node * 2, start, mid, l, r) + 
               query(node * 2 + 1, mid + 1, end, l, r);
    }
    
    void update_single(int idx, const Poly& val) {
        update(1, 1, n, idx, val);
    }
    
    Poly query_range(int l, int r) {
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

SegmentTree segtree(MAXN);

// 查询路径上的多项式
Poly query_path_poly(int u, int v) {
    Poly result(0);  // 零多项式
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        result = result + segtree.query_range(id[top[u]], id[u]);
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    result = result + segtree.query_range(id[u], id[v]);
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
        // 将节点值转换为常数多项式
        Poly p(0);
        p.coef[0] = w[i];
        segtree.update_single(id[i], p);
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // 更新节点值
            int u;
            long long val;
            cin >> u >> val;
            Poly p(0);
            p.coef[0] = val;
            segtree.update_single(id[u], p);
        } else if (op == 2) {  // 查询路径多项式在x处的值
            int u, v;
            long long x;
            cin >> u >> v >> x;
            
            Poly poly = query_path_poly(u, v);
            cout << "多项式在x=" << x << "处的值: " << poly.evaluate(x) << endl;
        }
    }
    
    return 0;
}
```

## 3. 树上线性代数

### 问题3：树上基尔霍夫矩阵树定理应用

**题目描述**：
在一棵树上，计算生成树的个数（虽然树的生成树就是它本身，但可以扩展到图）。

**解法**：
使用基尔霍夫矩阵树定理。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 305;
const int MOD = 1e9 + 7;

// 高斯消元计算行列式
long long gauss_elimination(vector<vector<long long>>& matrix, int n) {
    long long result = 1;
    
    for (int i = 0; i < n; i++) {
        // 找主元
        int max_row = i;
        for (int j = i + 1; j < n; j++) {
            if (abs(matrix[j][i]) > abs(matrix[max_row][i])) {
                max_row = j;
            }
        }
        
        if (max_row != i) {
            swap(matrix[i], matrix[max_row]);
            result = -result;  // 交换行改变符号
        }
        
        if (matrix[i][i] == 0) {
            return 0;  // 行列式为0
        }
        
        for (int j = i + 1; j < n; j++) {
            while (matrix[j][i] != 0) {
                long long ratio = matrix[i][i] / matrix[j][i];
                
                for (int k = i; k < n; k++) {
                    matrix[i][k] = ((matrix[i][k] - ratio * matrix[j][k]) % MOD + MOD) % MOD;
                }
                
                swap(matrix[i], matrix[j]);
                result = -result;
            }
        }
        
        result = (result * matrix[i][i]) % MOD;
        if (result < 0) result += MOD;
    }
    
    return (result + MOD) % MOD;
}

// 计算图的生成树个数（矩阵树定理）
long long countSpanningTrees(vector<vector<int>>& graph, int n) {
    // 构造基尔霍夫矩阵
    vector<vector<long long>> laplacian(n, vector<long long>(n, 0));
    
    // 计算度数
    for (int i = 0; i < n; i++) {
        int degree = 0;
        for (int j = 0; j < n; j++) {
            if (graph[i][j] > 0) {
                degree += graph[i][j];
                laplacian[i][j] = -graph[i][j];
            }
        }
        laplacian[i][i] = degree;
    }
    
    // 删除最后一行和最后一列，计算(n-1)×(n-1)子矩阵的行列式
    vector<vector<long long>> minor(n - 1, vector<long long>(n - 1));
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1; j++) {
            minor[i][j] = laplacian[i][j];
        }
    }
    
    return gauss_elimination(minor, n - 1);
}

// 树的特殊情况：树的生成树就是它本身，个数为1
long long countSpanningTreesInTree(int n) {
    return 1;  // 树的生成树个数就是1
}

// 扩展：计算树上每条边在多少个生成树中（对于树来说，每条边都在唯一的生成树中）
vector<int> countEdgeInclusion(int n) {
    vector<int> result(n - 1, 1);  // 对于树，每条边都在生成树中出现1次
    return result;
}

int main() {
    int n;
    cin >> n;
    
    vector<vector<int>> graph(n, vector<int>(n, 0));
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        u--; v--;  // 转换为0基索引
        graph[u][v] = 1;
        graph[v][u] = 1;
    }
    
    cout << "生成树个数: " << countSpanningTreesInTree(n) << endl;
    
    return 0;
}
```

## 4. 树上数论问题

### 问题4：树上GCD路径查询

**题目描述**：
在一棵树上，每个节点有一个权值，多次查询路径上所有节点权值的最大公约数。

**解法**：
使用树链剖分+线段树维护GCD。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int w[MAXN];  // 节点权值
int cnt = 0;

struct SegmentTree {
    vector<int> tree;
    int n;
    
    SegmentTree(int _n) : n(_n) {
        tree.resize(4 * n, 0);
    }
    
    int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }
    
    void update(int node, int start, int end, int idx, int val) {
        if (start == end) {
            tree[node] = val;
            return;
        }
        
        int mid = (start + end) / 2;
        if (idx <= mid) {
            update(node * 2, start, mid, idx, val);
        } else {
            update(node * 2 + 1, mid + 1, end, idx, val);
        }
        
        tree[node] = gcd(tree[node * 2], tree[node * 2 + 1]);
    }
    
    int query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return 0;  // GCD的单位元是0（特殊处理）
        if (l <= start && end <= r) return tree[node];
        
        int mid = (start + end) / 2;
        return gcd(query(node * 2, start, mid, l, r), 
                   query(node * 2 + 1, mid + 1, end, l, r));
    }
    
    void update_single(int idx, int val) {
        update(1, 1, n, idx, val);
    }
    
    int query_range(int l, int r) {
        if (l > r) return 0;
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

SegmentTree segtree(MAXN);

int query_path_gcd(int u, int v) {
    int result = 0;  // GCD的初始值
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        result = result == 0 ? segtree.query_range(id[top[u]], id[u]) : 
                             gcd(result, segtree.query_range(id[top[u]], id[u]));
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    result = result == 0 ? segtree.query_range(id[u], id[v]) : 
                         gcd(result, segtree.query_range(id[u], id[v]));
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
        segtree.update_single(id[i], w[i]);
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    // 重新初始化线段树
    for (int i = 1; i <= n; i++) {
        segtree.update_single(id[i], w[i]);
    }
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // 更新节点值
            int u, val;
            cin >> u >> val;
            segtree.update_single(id[u], val);
        } else if (op == 2) {  // 查询路径GCD
            int u, v;
            cin >> u >> v;
            cout << "路径GCD: " << query_path_gcd(u, v) << endl;
        }
    }
    
    return 0;
}
```

## 5. 树上组合数学

### 问题5：树上独立集计数

**题目描述**：
计算一棵树上所有独立集的数量（独立集是节点的子集，其中任意两个节点不相邻）。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MOD = 1e9 + 7;

vector<int> adj[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中，u节点不选/选的独立集数量

void dfs(int u, int parent) {
    dp[u][0] = 1;  // u不选
    dp[u][1] = 1;  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不选：v可以选也可以不选
            long long temp0 = dp[u][0];
            dp[u][0] = (temp0 * (dp[v][0] + dp[v][1])) % MOD;
            
            // u选：v必须不选
            long long temp1 = dp[u][1];
            dp[u][1] = (temp1 * dp[v][0]) % MOD;
        }
    }
}

long long countIndependentSets(int n) {
    dfs(1, 0);
    return (dp[1][0] + dp[1][1]) % MOD;
}

// 扩展：计算恰好包含k个节点的独立集数量
long long dp2[MAXN][2][MAXN/100];  // dp[u][selected][count] 

void dfs_count_exact(int u, int parent, int max_k) {
    // 初始化
    for (int i = 0; i <= max_k; i++) {
        dp2[u][0][i] = 0;
        dp2[u][1][i] = 0;
    }
    
    dp2[u][0][0] = 1;  // u不选，当前独立集大小为0
    dp2[u][1][1] = 1;  // u选，当前独立集大小为1
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_count_exact(v, u, max_k);
            
            // 卷积合并
            vector<long long> new_dp0(max_k + 1, 0);
            vector<long long> new_dp1(max_k + 1, 0);
            
            for (int i = 0; i <= max_k; i++) {
                for (int j = 0; j <= max_k - i; j++) {
                    // u不选：v可以选也可以不选
                    new_dp0[i + j] = (new_dp0[i + j] + 
                                     dp2[u][0][i] * (dp2[v][0][j] + dp2[v][1][j])) % MOD;
                    
                    // u选：v必须不选
                    if (i + j <= max_k) {
                        new_dp1[i + j] = (new_dp1[i + j] + 
                                         dp2[u][1][i] * dp2[v][0][j]) % MOD;
                    }
                }
            }
            
            for (int i = 0; i <= max_k; i++) {
                dp2[u][0][i] = new_dp0[i];
                dp2[u][1][i] = new_dp1[i];
            }
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "独立集总数: " << countIndependentSets(n) << endl;
    
    return 0;
}
```
