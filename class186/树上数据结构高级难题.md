# 树上数据结构高级难题

## 1. 线段树合并在树上的应用

### 问题1：子树信息合并查询

**题目描述**：
在一棵树上，每个节点有一个集合，支持以下操作：
1. `add u val`：向节点u的集合中添加值val
2. `query u v`：查询节点u到节点v路径上所有节点集合的并集中第k小的元素

**解法**：
使用线段树合并+LCA。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 1e7 + 5;  // 线段树节点数量
const int LOG = 20;
const int MAX_VAL = 1e9;

vector<int> adj[MAXN];
int depth[MAXN], fa[MAXN][LOG];
int root[MAXN];  // 每个节点的权值线段树根

// 权值线段树节点
struct Node {
    int l, r, sum;
} tree[MAXM];

int cnt = 0;

// 创建新节点
int newNode() {
    cnt++;
    tree[cnt].l = tree[cnt].r = tree[cnt].sum = 0;
    return cnt;
}

// 更新线段树
int update(int pre, int l, int r, int pos, int val) {
    if (!pre) pre = newNode();
    int cur = newNode(), ret = cur;
    tree[cur] = tree[pre];
    tree[cur].sum += val;
    
    if (l == r) return ret;
    
    int mid = (l + r) >> 1;
    if (pos <= mid) {
        tree[cur].l = update(tree[pre].l, l, mid, pos, val);
    } else {
        tree[cur].r = update(tree[pre].r, mid + 1, r, pos, val);
    }
    
    return ret;
}

// 查询第k小
int queryKth(int u_root, int v_root, int lca_root, int pp_root, int l, int r, int k) {
    if (l == r) return l;
    
    int mid = (l + r) >> 1;
    int left_sum = tree[tree[u_root].l].sum + tree[tree[v_root].l].sum - 
                   tree[tree[lca_root].l].sum - tree[tree[pp_root].l].sum;
    
    if (k <= left_sum) {
        return queryKth(tree[u_root].l, tree[v_root].l, tree[lca_root].l, tree[pp_root].l, l, mid, k);
    } else {
        return queryKth(tree[u_root].r, tree[v_root].r, tree[lca_root].r, tree[pp_root].r, mid + 1, r, k - left_sum);
    }
}

void dfs(int u, int parent) {
    depth[u] = depth[parent] + 1;
    fa[u][0] = parent;
    
    for (int k = 1; k < LOG; k++) {
        fa[u][k] = fa[fa[u][k-1]][k-1];
    }
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            // 合并子树的权值线段树
            root[u] = update(root[u], 1, MAX_VAL, v, 1);  // 示例：添加子节点编号
        }
    }
}

int getLCA(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    
    int diff = depth[u] - depth[v];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) {
            u = fa[u][k];
        }
    }
    
    if (u == v) return u;
    
    for (int k = LOG - 1; k >= 0; k--) {
        if (fa[u][k] != fa[v][k]) {
            u = fa[u][k];
            v = fa[v][k];
        }
    }
    
    return fa[u][0];
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0);
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // add
            int u, val;
            cin >> u >> val;
            root[u] = update(root[u], 1, MAX_VAL, val, 1);
        } else if (op == 2) {  // query
            int u, v, k;
            cin >> u >> v >> k;
            
            int lca = getLCA(u, v);
            int pp = fa[lca][0];  // LCA的父节点
            
            int result = queryKth(root[u], root[v], root[lca], 
                                  pp ? root[pp] : 0, 1, MAX_VAL, k);
            cout << "第" << k << "小的元素: " << result << endl;
        }
    }
    
    return 0;
}
```

## 2. 可持久化数据结构在树上的应用

### 问题2：历史版本LCA查询

**题目描述**：
在一棵树上支持以下操作：
1. `add_edge u v`：添加一条边连接u和v
2. `remove_edge u v`：删除边u-v
3. `query u v t`：查询在时间t时u和v的LCA

**解法**：
使用可持久化并查集或可持久化树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXT = 2e5 + 5;

struct PersistentUnionFind {
    struct Node {
        int parent, rank, size;
    };
    
    vector<vector<Node>> nodes;  // 每个版本的节点信息
    vector<int> roots;           // 每个版本的根节点
    
    PersistentUnionFind(int n) {
        nodes.resize(1);
        nodes[0].resize(n + 1);
        roots.resize(1);
        
        for (int i = 1; i <= n; i++) {
            nodes[0][i] = {i, 0, 1};
        }
        roots[0] = 0;
    }
    
    int copyVersion(int version) {
        int new_version = nodes.size();
        nodes.push_back(nodes[version]);
        roots.push_back(roots[version]);
        return new_version;
    }
    
    int find(int version, int x) {
        int p = nodes[version][x].parent;
        if (p == x) return x;
        return find(version, p);
    }
    
    bool unite(int version, int x, int y) {
        x = find(version, x);
        y = find(version, y);
        
        if (x == y) return false;
        
        int new_version = copyVersion(version);
        
        if (nodes[new_version][x].rank < nodes[new_version][y].rank) {
            swap(x, y);
        }
        
        nodes[new_version][y].parent = x;
        nodes[new_version][x].size += nodes[new_version][y].size;
        
        if (nodes[new_version][x].rank == nodes[new_version][y].rank) {
            nodes[new_version][x].rank++;
        }
        
        roots[new_version] = new_version;
        return true;
    }
};

// 由于历史版本LCA查询较为复杂，这里给出概念性的框架
class HistoricalLCASolver {
private:
    vector<pair<int, int>> edges;  // 所有边
    vector<int> edge_time_add;     // 边的添加时间
    vector<int> edge_time_remove;  // 边的删除时间
    vector<int> versions;          // 版本时间点
    
public:
    void addEdge(int u, int v, int time) {
        edges.push_back({u, v});
        edge_time_add.push_back(time);
        edge_time_remove.push_back(-1);  // -1表示未删除
        versions.push_back(time);
    }
    
    void removeEdge(int u, int v, int time) {
        for (int i = 0; i < edges.size(); i++) {
            if ((edges[i].first == u && edges[i].second == v) ||
                (edges[i].first == v && edges[i].second == u)) {
                edge_time_remove[i] = time;
                versions.push_back(time);
                break;
            }
        }
    }
    
    // 在时间t查询u和v的LCA
    int queryLCA(int u, int v, int t) {
        // 需要重构时间t的树结构，然后计算LCA
        // 这是一个复杂的问题，通常使用时间分治或整体二分
        return -1;  // 简化返回
    }
};
```

## 3. 树分治高级应用

### 问题3：树上路径统计（点分治）

**题目描述**：
在一棵树上，多次查询距离某个节点不超过k的节点数量。

**解法**：
使用点分治。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
bool vis[MAXN];
int sz[MAXN], max_part[MAXN];
int root, tot;
vector<int> dist[MAXN];  // 每个重心到子树节点的距离
unordered_map<int, int> cnt[MAXN];  // 每个重心的子树距离计数

void getRoot(int u, int parent) {
    sz[u] = 1;
    max_part[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getRoot(v, u);
            sz[u] += sz[v];
            max_part[u] = max(max_part[u], sz[v]);
        }
    }
    
    max_part[u] = max(max_part[u], tot - sz[u]);
    if (max_part[u] < max_part[root]) root = u;
}

void getDist(int u, int parent, int d, vector<int>& tmp_dist) {
    tmp_dist.push_back(d);
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getDist(v, u, d + 1, tmp_dist);
        }
    }
}

void calc(int u) {
    vector<int> tmp;
    getDist(u, 0, 0, tmp);
    
    sort(tmp.begin(), tmp.end());
    
    // 更新以u为重心的子树距离信息
    for (int d : tmp) {
        cnt[u][d]++;
    }
    
    dist[u] = tmp;
}

void solve(int u) {
    vis[u] = true;
    calc(u);  // 计算以u为重心的子树信息
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            // 删除u-v边后，v所在子树的信息
            vector<int> tmp;
            getDist(v, u, 1, tmp);
            sort(tmp.begin(), tmp.end());
            
            // 从u的计数中减去v子树的贡献
            for (int d : tmp) {
                cnt[u][d]--;
                if (cnt[u][d] == 0) cnt[u].erase(d);
            }
            
            // 递归处理v子树
            tot = sz[v];
            root = 0;
            getRoot(v, 0);
            solve(root);
            
            // 恢复u的计数
            for (int d : tmp) {
                cnt[u][d]++;
            }
        }
    }
}

int query(int u, int k) {
    // 查询距离u不超过k的节点数
    int result = 0;
    int cur = u;
    
    while (true) {
        // 在当前重心的子树中查找
        auto it = cnt[cur].upper_bound(k);
        if (it != cnt[cur].begin()) {
            --it;
            result += it->second;
        }
        
        if (cur == root) break;  // 到达总根
        // 移动到父重心
        // 这里需要额外的数据结构来维护重心树的父子关系
        break;  // 简化处理
    }
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    max_part[0] = tot = n;
    getRoot(1, 0);
    solve(root);
    
    for (int i = 0; i < q; i++) {
        int u, k;
        cin >> u >> k;
        cout << "距离节点" << u << "不超过" << k << "的节点数: " 
             << query(u, k) << endl;
    }
    
    return 0;
}
```

## 4. 树上莫队算法

### 问题4：树上区间众数查询

**题目描述**：
在一棵树上，多次查询路径上出现次数最多的值（众数）。

**解法**：
使用欧拉序+莫队算法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int values[MAXN];
int in[MAXN], out[MAXN], timer = 0;
vector<int> euler;  // 欧拉序
int block_size;

void eulerTour(int u, int parent) {
    in[u] = euler.size();
    euler.push_back(values[u]);
    
    for (int v : adj[u]) {
        if (v != parent) {
            eulerTour(v, u);
        }
    }
    
    out[u] = euler.size();
    euler.push_back(values[u]);  // 回溯时再次添加
}

struct Query {
    int l, r, id, u, v;
    bool operator<(const Query& other) const {
        int block_l = l / block_size;
        int block_r = r / block_size;
        if (block_l != block_r) return block_l < block_r;
        return (block_r & 1) ? r < other.r : r > other.r;  // 奇偶优化
    }
};

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    eulerTour(1, 0);
    block_size = max(1, (int)sqrt(euler.size()));
    
    vector<Query> queries(q);
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        int l = in[u], r = in[v];
        if (l > r) swap(l, r);
        queries[i] = {l, r, i, u, v};
    }
    
    sort(queries.begin(), queries.end());
    
    vector<int> results(q);
    int current_l = 0, current_r = -1;
    unordered_map<int, int> count;  // 值的出现次数
    int max_freq = 0;              // 最大频次
    int mode = 0;                  // 众数值
    
    auto add = [&](int idx) {
        int val = euler[idx];
        count[val]++;
        if (count[val] > max_freq) {
            max_freq = count[val];
            mode = val;
        }
    };
    
    auto remove = [&](int idx) {
        int val = euler[idx];
        if (count[val] == max_freq && val == mode) {
            // 需要重新找到众数
            bool found_new_mode = false;
            count[val]--;
            if (count[val] == 0) count.erase(val);
            
            // 简化处理：这里需要更复杂的更新逻辑
            max_freq = 0;
            for (auto& p : count) {
                if (p.second > max_freq) {
                    max_freq = p.second;
                    mode = p.first;
                    found_new_mode = true;
                }
            }
            if (!found_new_mode) mode = 0;
        } else {
            count[val]--;
            if (count[val] == 0) count.erase(val);
        }
    };
    
    for (auto& query : queries) {
        int l = query.l, r = query.r;
        
        while (current_l > l) {
            current_l--;
            add(current_l);
        }
        while (current_r < r) {
            current_r++;
            add(current_r);
        }
        while (current_l < l) {
            remove(current_l);
            current_l++;
        }
        while (current_r > r) {
            remove(current_r);
            current_r--;
        }
        
        results[query.id] = mode;
    }
    
    for (int result : results) {
        cout << "路径众数: " << result << endl;
    }
    
    return 0;
}
```

## 5. 树哈希与同构判断

### 问题5：树的同构判断

**题目描述**：
判断两棵树是否同构（结构相同）。

**解法**：
使用树哈希。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_set>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;
const int BASE = 131;
const int MOD = 1e9 + 7;

vector<int> adj1[MAXN], adj2[MAXN];
int n;

// 计算子树哈希值
long long computeHash(vector<int> children_hashes) {
    sort(children_hashes.begin(), children_hashes.end());
    
    long long hash = 1;
    for (long long child_hash : children_hashes) {
        hash = (hash * BASE + child_hash) % MOD;
    }
    
    return hash;
}

long long dfsHash(vector<vector<int>>& adj, int u, int parent) {
    vector<long long> child_hashes;
    
    for (int v : adj[u]) {
        if (v != parent) {
            child_hashes.push_back(dfsHash(adj, v, u));
        }
    }
    
    return computeHash(child_hashes);
}

// 重新标记树的根以处理无根树同构
long long getTreeHash(vector<vector<int>>& adj, int n, int root) {
    return dfsHash(adj, root, -1);
}

// 找树的重心
void findCentroids(vector<vector<int>>& adj, int n, int u, int parent, 
                   vector<int>& centroids, vector<int>& sz, vector<int>& max_subtree) {
    sz[u] = 1;
    max_subtree[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent) {
            findCentroids(adj, n, v, u, centroids, sz, max_subtree);
            sz[u] += sz[v];
            max_subtree[u] = max(max_subtree[u], sz[v]);
        }
    }
    
    max_subtree[u] = max(max_subtree[u], n - sz[u]);
    if (max_subtree[u] <= n / 2) {
        centroids.push_back(u);
    }
}

bool areTreesIsomorphic() {
    // 找到两棵树的重心
    vector<int> centroids1, centroids2;
    vector<int> sz1(MAXN), max_subtree1(MAXN);
    vector<int> sz2(MAXN), max_subtree2(MAXN);
    
    findCentroids(adj1, n, 1, -1, centroids1, sz1, max_subtree1);
    findCentroids(adj2, n, 1, -1, centroids2, sz2, max_subtree2);
    
    // 获取第一棵树以重心为根的哈希值
    long long hash1 = getTreeHash(adj1, n, centroids1[0]);
    
    // 检查第二棵树是否与第一棵树同构
    for (int centroid : centroids2) {
        if (getTreeHash(adj2, n, centroid) == hash1) {
            return true;
        }
    }
    
    // 如果第一棵树有两个重心，也需要检查第二棵树以两个重心为根的情况
    if (centroids1.size() > 1) {
        long long hash1_alt = getTreeHash(adj1, n, centroids1[1]);
        for (int centroid : centroids2) {
            if (getTreeHash(adj2, n, centroid) == hash1_alt) {
                return true;
            }
        }
    }
    
    return false;
}

int main() {
    cin >> n;
    
    // 输入第一棵树
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj1[u].push_back(v);
        adj1[v].push_back(u);
    }
    
    // 输入第二棵树
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj2[u].push_back(v);
        adj2[v].push_back(u);
    }
    
    if (areTreesIsomorphic()) {
        cout << "两棵树同构" << endl;
    } else {
        cout << "两棵树不同构" << endl;
    }
    
    return 0;
}
```

## 6. 树上启发式算法

### 问题6：树上最大独立集变种

**题目描述**：
在一棵树上，每个节点有权值，选择节点使得权值和最大，且任意两个选择的节点距离至少为3。

**解法**：
使用树形DP+启发式合并思想。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int weight[MAXN];
int dp[MAXN][2][2];  // [节点][当前节点是否选][父节点是否选][祖父节点是否选]

void dfs(int u, int parent) {
    // 初始化：当前节点不选
    dp[u][0][0] = 0;  // 父节点不选
    dp[u][0][1] = 0;  // 父节点选
    
    // 初始化：当前节点选
    dp[u][1][0] = weight[u];  // 父节点不选
    dp[u][1][1] = weight[u];  // 父节点选（这种情况在初始时设为权值，后面会调整）
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // 合并子树信息
            // 当前节点不选的情况
            int new_dp00 = dp[u][0][0] + max(dp[v][0][0], dp[v][1][0]);
            int new_dp01 = dp[u][0][1] + max(dp[v][0][0], dp[v][1][0]);
            
            // 当前节点选的情况
            int new_dp10 = dp[u][1][0] + dp[v][0][0];  // 如果当前节点选，则子节点不能选
            int new_dp11 = dp[u][1][1] + dp[v][0][0];  // 同上
            
            dp[u][0][0] = new_dp00;
            dp[u][0][1] = new_dp01;
            dp[u][1][0] = new_dp10;
            dp[u][1][1] = new_dp11;
        }
    }
    
    // 如果当前节点被选择，则其子节点不能被选择
    dp[u][1][1] = dp[u][1][0];  // 父节点被选择时，当前节点不能选
}

int solveMaxIndependentSet() {
    dfs(1, 0);
    return max(dp[1][0][0], dp[1][1][0]);
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> weight[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    cout << "最大独立集权值和: " << solveMaxIndependentSet() << endl;
    
    return 0;
}
```
