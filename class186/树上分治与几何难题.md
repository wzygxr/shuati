# 树上分治与几何难题

## 1. 树上CDQ分治

### 问题1：树上三维偏序

**题目描述**：
在一棵树上，每个节点有三个属性(a,b,c)，多次查询满足条件的节点对数量：u到v路径上节点中，有多少节点的属性满足 a≤A, b≤B, c≤C。

**解法**：
使用树链剖分+CDQ分治。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <tuple>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int cnt = 0;

struct Node {
    int a, b, c, id;
    bool operator<(const Node& other) const {
        if (a != other.a) return a < other.a;
        if (b != other.b) return b < other.b;
        return c < other.c;
    }
};

vector<Node> nodes;

// 树状数组用于二维数点
struct BIT2D {
    vector<vector<int>> tree;
    int n, m;
    
    BIT2D(int _n, int _m) : n(_n), m(_m) {
        tree.resize(n + 1, vector<int>(m + 1, 0));
    }
    
    void update(int x, int y, int val) {
        for (int i = x; i <= n; i += i & (-i)) {
            for (int j = y; j <= m; j += j & (-j)) {
                tree[i][j] += val;
            }
        }
    }
    
    int query(int x, int y) {
        if (x <= 0 || y <= 0) return 0;
        int result = 0;
        for (int i = x; i > 0; i -= i & (-i)) {
            for (int j = y; j > 0; j -= j & (-j)) {
                result += tree[i][j];
            }
        }
        return result;
    }
    
    int query_range(int x1, int y1, int x2, int y2) {
        return query(x2, y2) - query(x1-1, y2) - query(x2, y1-1) + query(x1-1, y1-1);
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

// CDQ分治解决三维偏序
int cdq_3d(vector<Node>& arr, int A, int B, int C) {
    if (arr.size() <= 1) {
        return (arr.size() == 1 && arr[0].a <= A && arr[0].b <= B && arr[0].c <= C) ? 1 : 0;
    }
    
    int mid = arr.size() / 2;
    vector<Node> left, right;
    
    for (int i = 0; i < mid; i++) left.push_back(arr[i]);
    for (int i = mid; i < arr.size(); i++) right.push_back(arr[i]);
    
    int result = cdq_3d(left, A, B, C) + cdq_3d(right, A, B, C);
    
    // 处理跨越分割点的贡献
    sort(right.begin(), right.end(), [](const Node& x, const Node& y) {
        return x.b < y.b;
    });
    
    BIT2D bit(100005, 100005);
    int r_idx = 0;
    
    sort(left.begin(), left.end(), [](const Node& x, const Node& y) {
        return x.b < y.b;
    });
    
    for (const Node& l_node : left) {
        while (r_idx < right.size() && right[r_idx].b <= l_node.b) {
            if (right[r_idx].a <= A) {
                bit.update(right[r_idx].b, right[r_idx].c, 1);
            }
            r_idx++;
        }
        
        if (l_node.a <= A) {
            result += bit.query(B, C);
        }
    }
    
    return result;
}

int query_path_3d(int u, int v, int A, int B, int C) {
    vector<Node> path_nodes;
    
    // 获取路径上的节点
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        for (int i = id[top[u]]; i <= id[u]; i++) {
            if (rev[i]) path_nodes.push_back(nodes[rev[i]]);
        }
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    for (int i = id[u]; i <= id[v]; i++) {
        if (rev[i]) path_nodes.push_back(nodes[rev[i]]);
    }
    
    return cdq_3d(path_nodes, A, B, C);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        nodes.push_back({a, b, c, i});
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    for (int i = 0; i < q; i++) {
        int u, v, A, B, C;
        cin >> u >> v >> A >> B >> C;
        cout << "满足条件的节点数: " << query_path_3d(u, v, A, B, C) << endl;
    }
    
    return 0;
}
```

## 2. 树上整体二分

### 问题2：树上第K大路径权值

**题目描述**：
在一棵树上，每条边有权值，多次查询u到v路径上第K大的边权。

**解法**：
使用树链剖分+整体二分。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边权)
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int edge_weights[MAXN];  // 从节点到其父节点的边权
int cnt = 0;

struct Query {
    int u, v, k, id;
    int result;
};

vector<Query> queries;
vector<int> results;

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (auto& e : adj[u]) {
        int v = e.first, w = e.second;
        if (v != fa) {
            edge_weights[v] = w;
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (auto& e : adj[u]) {
        int v = e.first;
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

// 树状数组用于统计边权数量
struct BIT {
    vector<int> tree;
    int n;
    
    BIT(int _n) : n(_n) {
        tree.resize(n + 1, 0);
    }
    
    void update(int idx, int val) {
        for (int i = idx; i <= n; i += i & (-i)) {
            tree[i] += val;
        }
    }
    
    int query(int idx) {
        int result = 0;
        for (int i = idx; i > 0; i -= i & (-i)) {
            result += tree[i];
        }
        return result;
    }
    
    int range_query(int l, int r) {
        return query(r) - query(l - 1);
    }
};

// 查询路径上权值小于等于val的边数
int query_path_count(int u, int v, int val, BIT& bit) {
    int result = 0;
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        
        // 查询重链上的边
        for (int i = id[top[u]]; i <= id[u]; i++) {
            if (i > 1) {  // 非根节点
                int node = rev[i];
                if (depth[node] > depth[parent[node]]) {
                    int edge_w = edge_weights[node];
                    if (edge_w <= val) {
                        result += bit.query(id[node]) - bit.query(id[node] - 1);
                    }
                }
            }
        }
        
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    
    // 查询u到v的路径
    for (int i = id[u] + 1; i <= id[v]; i++) {
        int node = rev[i];
        int edge_w = edge_weights[node];
        if (edge_w <= val) {
            result += bit.query(id[node]) - bit.query(id[node] - 1);
        }
    }
    
    return result;
}

// 整体二分求解
void solve_binary(int l, int r, vector<int>& query_ids, vector<int>& possible_values) {
    if (query_ids.empty() || l > r) return;
    
    if (l == r) {
        for (int qid : query_ids) {
            queries[qid].result = possible_values[l];
        }
        return;
    }
    
    int mid = (l + r) / 2;
    int mid_val = possible_values[mid];
    
    BIT bit(n);
    // 更新权值小于等于mid_val的边
    for (int i = 1; i <= n; i++) {
        if (edge_weights[i] <= mid_val) {
            bit.update(id[i], 1);
        }
    }
    
    vector<int> left_queries, right_queries;
    
    for (int qid : query_ids) {
        int u = queries[qid].u;
        int v = queries[qid].v;
        int k = queries[qid].k;
        
        int count = query_path_count(u, v, mid_val, bit);
        
        if (count >= k) {
            left_queries.push_back(qid);
        } else {
            right_queries.push_back(qid);
        }
    }
    
    solve_binary(l, mid, left_queries, possible_values);
    solve_binary(mid + 1, r, right_queries, possible_values);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    vector<int> possible_values;
    for (int i = 1; i <= n; i++) {
        if (parent[i] != 0) {  // 非根节点
            possible_values.push_back(edge_weights[i]);
        }
    }
    sort(possible_values.begin(), possible_values.end());
    possible_values.erase(unique(possible_values.begin(), possible_values.end()), possible_values.end());
    
    for (int i = 0; i < q; i++) {
        int u, v, k;
        cin >> u >> v >> k;
        queries.push_back({u, v, k, i, 0});
    }
    
    vector<int> query_ids;
    for (int i = 0; i < q; i++) {
        query_ids.push_back(i);
    }
    
    solve_binary(0, possible_values.size() - 1, query_ids, possible_values);
    
    for (int i = 0; i < q; i++) {
        cout << "第" << queries[i].k << "大边权: " << queries[i].result << endl;
    }
    
    return 0;
}
```

## 3. 树上几何问题

### 问题3：树上最远点对距离

**题目描述**：
在一棵树上，多次查询，每次查询距离节点u最远的节点及其距离。

**解法**：
使用点分治。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
bool vis[MAXN];
int sz[MAXN], max_part[MAXN];
int root, tot;
int max_dist[MAXN];      // 每个点的最远距离
int farthest_node[MAXN]; // 每个点最远的节点

void getRoot(int u, int parent) {
    sz[u] = 1;
    max_part[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getRoot(v, u);
            sz[u] += sz[v];
            max_part[u] = max(max_part[u], sz[v]);
        }
    }
    
    max_part[u] = max(max_part[u], tot - sz[u]);
    if (max_part[u] < max_part[root]) root = u;
}

// 从u开始，更新子树中所有点的最远距离信息
void updateDist(int u, int parent, int dist, int src, int& max_d, int& far_node) {
    if (dist > max_d) {
        max_d = dist;
        far_node = src;
    }
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            updateDist(v, u, dist + 1, src, max_d, far_node);
        }
    }
}

void calc(int u) {
    // 以u为重心，更新距离信息
    vector<int> nodes;
    vector<int> distances;
    
    // 收集子树节点
    function<void(int, int, int)> collect = [&](int x, int p, int d) {
        nodes.push_back(x);
        distances.push_back(d);
        for (int v : adj[x]) {
            if (v != p && !vis[v]) {
                collect(v, x, d + 1);
            }
        }
    };
    
    collect(u, 0, 0);
    
    // 对于每个节点，计算其到其他节点的最大距离
    for (int i = 0; i < nodes.size(); i++) {
        int node = nodes[i];
        int dist = distances[i];
        
        // 更新通过重心u的路径
        int max_d = dist;
        int far_node = u;
        
        for (int j = 0; j < nodes.size(); j++) {
            if (i != j) {
                int path_dist = distances[i] + distances[j];
                if (path_dist > max_d) {
                    max_d = path_dist;
                    // 这里需要更精确地找到最远点
                }
            }
        }
        
        if (max_d > max_dist[node]) {
            max_dist[node] = max_d;
            farthest_node[node] = far_node;
        }
    }
    
    // 考虑每个子树内部的最大距离
    for (int v : adj[u]) {
        if (!vis[v]) {
            // 暂时移除边u-v，计算v子树内部的最大距离
            vector<int> v_nodes, v_dist;
            function<void(int, int, int)> get_v = [&](int x, int p, int d) {
                v_nodes.push_back(x);
                v_dist.push_back(d);
                for (int w : adj[x]) {
                    if (w != p && !vis[w] && w != u) {
                        get_v(w, x, d + 1);
                    }
                }
            };
            
            get_v(v, u, 1);
            
            // 计算v子树内部的直径
            int sub_max_dist = 0;
            int sub_farthest = v_nodes[0];
            
            for (int node : v_nodes) {
                int max_d = 0;
                int far_node = node;
                updateDist(node, parent[node], 0, node, max_d, far_node);
                
                if (max_d > sub_max_dist) {
                    sub_max_dist = max_d;
                    sub_farthest = far_node;
                }
            }
        }
    }
}

void solve(int u) {
    vis[u] = true;
    calc(u);
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            tot = sz[v];
            root = 0;
            getRoot(v, 0);
            solve(root);
        }
    }
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    max_part[0] = tot = n;
    getRoot(1, 0);
    solve(root);
    
    for (int i = 0; i < q; i++) {
        int u;
        cin >> u;
        cout << "节点" << u << "最远的节点: " << farthest_node[u] 
             << "，距离: " << max_dist[u] << endl;
    }
    
    return 0;
}
```

## 4. 树上计算几何

### 问题4：树上凸包问题

**题目描述**：
在一棵树上，每个节点有权值，表示在二维平面上的点，查询路径上点的凸包。

**解法**：
使用树链剖分+线段树维护凸包。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const int MAXN = 1e4 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
pair<int, int> points[MAXN];  // 每个节点对应的二维坐标
int cnt = 0;

struct Point {
    long long x, y;
    Point(long long x = 0, long long y = 0) : x(x), y(y) {}
    Point operator-(const Point& other) const {
        return Point(x - other.x, y - other.y);
    }
    long long cross(const Point& other) const {
        return x * other.y - y * other.x;
    }
    bool operator<(const Point& other) const {
        if (x != other.x) return x < other.x;
        return y < other.y;
    }
};

// 凸包
vector<Point> convexHull(vector<Point> pts) {
    int n = pts.size();
    if (n <= 1) return pts;
    
    sort(pts.begin(), pts.end());
    
    vector<Point> hull(2 * n);
    int sz = 0;
    
    // 下凸包
    for (int i = 0; i < n; i++) {
        while (sz > 1 && (hull[sz-1] - hull[sz-2]).cross(pts[i] - hull[sz-1]) <= 0) {
            sz--;
        }
        hull[sz++] = pts[i];
    }
    
    // 上凸包
    int start = sz;
    for (int i = n - 2; i >= 0; i--) {
        while (sz > start && (hull[sz-1] - hull[sz-2]).cross(pts[i] - hull[sz-1]) <= 0) {
            sz--;
        }
        hull[sz++] = pts[i];
    }
    
    hull.resize(sz - (sz > 1 ? 1 : 0));
    return hull;
}

struct SegTreeNode {
    vector<Point> hull;  // 该区间上的凸包
    vector<Point> points; // 该区间上的所有点
    
    SegTreeNode() {}
};

struct SegmentTree {
    vector<SegTreeNode> tree;
    int n;
    
    SegmentTree(int _n) : n(_n) {
        tree.resize(4 * n);
    }
    
    void pushUp(int node) {
        // 合并左右子树的点集
        tree[node].points = tree[node * 2].points;
        tree[node].points.insert(tree[node].points.end(), 
                                tree[node * 2 + 1].points.begin(), 
                                tree[node * 2 + 1].points.end());
        
        // 计算合并后的凸包
        tree[node].hull = convexHull(tree[node].points);
    }
    
    void build(int node, int start, int end) {
        if (start == end) {
            tree[node].points = {Point(points[rev[start]].first, points[rev[start]].second)};
            tree[node].hull = tree[node].points;
            return;
        }
        
        int mid = (start + end) / 2;
        build(node * 2, start, mid);
        build(node * 2 + 1, mid + 1, end);
        pushUp(node);
    }
    
    vector<Point> query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return {};
        if (l <= start && end <= r) return tree[node].hull;
        
        int mid = (start + end) / 2;
        vector<Point> left_hull = query(node * 2, start, mid, l, r);
        vector<Point> right_hull = query(node * 2 + 1, mid + 1, end, l, r);
        
        // 合并两个凸包
        vector<Point> all_points = left_hull;
        all_points.insert(all_points.end(), right_hull.begin(), right_hull.end());
        return convexHull(all_points);
    }
    
    vector<Point> query_range(int l, int r) {
        if (l > r) return {};
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

SegmentTree segtree(MAXN);

vector<Point> query_path_hull(int u, int v) {
    vector<Point> all_points;
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        vector<Point> segment_points = segtree.query_range(id[top[u]], id[u]);
        all_points.insert(all_points.end(), segment_points.begin(), segment_points.end());
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    vector<Point> final_segment = segtree.query_range(id[u], id[v]);
    all_points.insert(all_points.end(), final_segment.begin(), final_segment.end());
    
    return convexHull(all_points);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        int x, y;
        cin >> x >> y;
        points[i] = {x, y};
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    segtree.build(1, 1, n);
    
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        vector<Point> hull = query_path_hull(u, v);
        cout << "路径凸包点数: " << hull.size() << endl;
    }
    
    return 0;
}
```

## 5. 树上离线算法

### 问题5：树上区间众数动态查询

**题目描述**：
在一棵树上，支持加点操作和查询路径上出现次数最多的元素。

**解法**：
使用欧拉序+莫队算法+回滚莫队。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int values[MAXN];
int in[MAXN], out[MAXN], timer = 0;
vector<int> euler;  // 欧拉序
int block_size;

void eulerTour(int u, int parent) {
    in[u] = euler.size();
    euler.push_back(values[u]);
    
    for (int v : adj[u]) {
        if (v != parent) {
            eulerTour(v, u);
        }
    }
    
    out[u] = euler.size();
    euler.push_back(values[u]);
}

struct Query {
    int l, r, id, u, v;
    bool operator<(const Query& other) const {
        int block_l = l / block_size;
        int block_r = r / block_size;
        if (block_l != block_r) return block_l < block_r;
        return (block_r & 1) ? r < other.r : r > other.r;
    }
};

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    eulerTour(1, 0);
    block_size = max(1, (int)sqrt(euler.size()));
    
    vector<Query> queries(q);
    for (int i = 0; i < q; i++) {
        int u, v;
        cin >> u >> v;
        int l = in[u], r = in[v];
        if (l > r) swap(l, r);
        queries[i] = {l, r, i, u, v};
    }
    
    sort(queries.begin(), queries.end());
    
    vector<int> results(q);
    int current_l = 0, current_r = -1;
    unordered_map<int, int> count;  // 值的出现次数
    int max_freq = 0;              // 最大频次
    int mode = 0;                  // 众数值
    
    auto add = [&](int idx) {
        int val = euler[idx];
        count[val]++;
        if (count[val] >= max_freq) {
            max_freq = count[val];
            mode = val;
        }
    };
    
    auto remove = [&](int idx) {
        int val = euler[idx];
        count[val]--;
        if (val == mode && count[val] < max_freq) {
            // 需要重新找到众数
            max_freq = 0;
            for (auto& p : count) {
                if (p.second > max_freq) {
                    max_freq = p.second;
                    mode = p.first;
                }
            }
        }
    };
    
    for (auto& query : queries) {
        int l = query.l, r = query.r;
        
        while (current_l > l) {
            current_l--;
            add(current_l);
        }
        while (current_r < r) {
            current_r++;
            add(current_r);
        }
        while (current_l < l) {
            remove(current_l);
            current_l++;
        }
        while (current_r > r) {
            remove(current_r);
            current_r--;
        }
        
        results[query.id] = mode;
    }
    
    for (int result : results) {
        cout << "路径众数: " << result << endl;
    }
    
    return 0;
}
```
