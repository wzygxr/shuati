# 树上网络流与匹配难题

## 1. 树上最大流问题

### 问题1：树上多源多汇最大流

**题目描述**：
在一棵树上，有多个源点和多个汇点，每条边有容量限制，求从所有源点到所有汇点的最大流。

**解法**：
利用树的性质简化网络流算法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;

vector<pair<int, int>> adj[MAXN];  // (邻接点, 容量)
int parent[MAXN], depth[MAXN];
int capacity[MAXN];  // 从节点到父节点的边容量
bool is_source[MAXN], is_sink[MAXN];
int flow[MAXN];  // 通过每个节点的流量

void dfs_tree(int u, int parent_node, int d) {
    parent[u] = parent_node;
    depth[u] = d;
    
    for (auto& edge : adj[u]) {
        int v = edge.first, cap = edge.second;
        if (v != parent_node) {
            capacity[v] = cap;
            dfs_tree(v, u, d + 1);
        }
    }
}

// 找到两点路径上的最小容量（瓶颈）
int min_capacity_path(int u, int v) {
    int min_cap = INT_MAX;
    
    // 将u和v提到同一深度
    while (depth[u] > depth[v]) {
        min_cap = min(min_cap, capacity[u]);
        u = parent[u];
    }
    
    while (depth[v] > depth[u]) {
        min_cap = min(min_cap, capacity[v]);
        v = parent[v];
    }
    
    // 现在u和v在同一深度
    while (u != v) {
        min_cap = min(min_cap, min(capacity[u], capacity[v]));
        u = parent[u];
        v = parent[v];
    }
    
    return min_cap;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v, c;
        cin >> u >> v >> c;
        adj[u].push_back({v, c});
        adj[v].push_back({u, c});
    }
    
    dfs_tree(1, 0, 1);
    
    int num_sources, num_sinks;
    cin >> num_sources;
    for (int i = 0; i < num_sources; i++) {
        int s;
        cin >> s;
        is_source[s] = true;
    }
    
    cin >> num_sinks;
    for (int i = 0; i < num_sinks; i++) {
        int t;
        cin >> t;
        is_sink[t] = true;
    }
    
    // 对于树，任意两个点之间只有一条路径
    // 所以多源多汇最大流问题可以转换为：找到所有源点到汇点的路径，并分配流量
    int total_flow = 0;
    
    // 简化版本：计算每条边的最大可能流量
    vector<pair<int, int>> edges;  // (容量, 边的两端)
    for (int i = 1; i <= n; i++) {
        if (parent[i] != 0) {
            edges.push_back({capacity[i], i});
        }
    }
    
    // 实际的多源多汇流需要更复杂的处理
    // 这里我们考虑一个简化的版本：计算所有源点到所有汇点的总流量
    for (int i = 1; i <= n; i++) {
        if (is_source[i]) {
            for (int j = 1; j <= n; j++) {
                if (is_sink[j]) {
                    int path_flow = min_capacity_path(i, j);
                    total_flow += path_flow;
                    // 实际上需要考虑流量守恒，这里只是示例
                }
            }
        }
    }
    
    cout << "最大流: " << total_flow << endl;
    
    return 0;
}
```

## 2. 树上最大匹配

### 问题2：树上最大加权匹配

**题目描述**：
在一棵树上，每条边有权值，求权值和最大的匹配（选择边使得没有公共顶点）。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边权)
long long dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中，u节点是否与父节点匹配的最大权值

void dfs_matching(int u, int parent_node) {
    dp[u][0] = 0;  // u不与父节点匹配
    dp[u][1] = 0;  // u与父节点匹配
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (v != parent_node) {
            dfs_matching(v, u);
            
            // u不匹配：v可以匹配或不匹配
            dp[u][0] += max(dp[v][0], dp[v][1]);
            
            // u匹配：需要选择一个子节点进行匹配
            dp[u][1] = max(dp[u][1], dp[v][0] + w);
        }
    }
    
    // 如果u匹配了，那么它的子节点都不能匹配
    if (dp[u][1] > 0) {
        dp[u][0] = max(dp[u][0], dp[u][1]);
    }
}

// 更精确的实现
long long dp_match[MAXN][2];  // dp[u][0] = u不匹配，dp[u][1] = u匹配

void dfs_precise(int u, int parent_node) {
    dp_match[u][0] = 0;  // u不匹配
    dp_match[u][1] = 0;  // u匹配
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (v != parent_node) {
            dfs_precise(v, u);
            
            // u不匹配：v可匹配可不匹配
            dp_match[u][0] += max(dp_match[v][0], dp_match[v][1]);
            
            // u匹配：考虑与v匹配的情况
            dp_match[u][1] = max(dp_match[u][1], 
                               dp_match[u][0] - max(dp_match[v][0], dp_match[v][1]) 
                               + dp_match[v][0] + w);
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    
    dfs_precise(1, 0);
    
    cout << "最大加权匹配权值: " << max(dp_match[1][0], dp_match[1][1]) << endl;
    
    return 0;
}
```

## 3. 树上最小顶点覆盖

### 问题3：树上加权最小顶点覆盖

**题目描述**：
在一棵树上，每个节点有权值，选择节点使得每条边至少有一个端点被选中，且选择节点的权值和最小。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int weight[MAXN];
long long dp[MAXN][2];  // dp[u][0/1] 表示u子树中，u节点是否被选中的最小权值和

void dfs_vertex_cover(int u, int parent_node) {
    dp[u][0] = 0;  // u不选
    dp[u][1] = weight[u];  // u选
    
    bool has_child = false;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            has_child = true;
            dfs_vertex_cover(v, u);
            
            // u不选：v必须选（因为边(u,v)需要被覆盖）
            dp[u][0] += dp[v][1];
            
            // u选：v可选可不选，取较小值
            dp[u][1] += min(dp[v][0], dp[v][1]);
        }
    }
    
    // 如果是叶子节点，且没有父节点（即整棵树只有一个节点）
    if (!has_child) {
        dp[u][0] = 0;  // 叶子节点不选，但这种情况需要特殊考虑
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> weight[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_vertex_cover(1, 0);
    
    cout << "最小顶点覆盖权值和: " << min(dp[1][0], dp[1][1]) << endl;
    
    return 0;
}
```

## 4. 树上二分图匹配

### 问题4：树上最大独立集与匹配的关系

**题目描述**：
在一棵树上，利用二分图性质求最大独立集和最小顶点覆盖，并验证König定理。

**解法**：
利用树是二分图的性质。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int color[MAXN];  // 二分图染色
long long dp[MAXN][2];  // dp[u][0/1] 表示u子树中u染色为0/1时的最大独立集大小

void dfs_color(int u, int parent_node, int c) {
    color[u] = c;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            dfs_color(v, u, 1 - c);  // 交替染色
        }
    }
}

void dfs_independent_set(int u, int parent_node) {
    dp[u][0] = (color[u] == 0) ? 1 : 0;  // u选择且u为0色
    dp[u][1] = (color[u] == 1) ? 1 : 0;  // u选择且u为1色
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            dfs_independent_set(v, u);
            
            // u选择0色：子节点选择1色
            dp[u][0] += dp[v][1];
            
            // u选择1色：子节点选择0色
            dp[u][1] += dp[v][0];
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_color(1, 0, 0);
    dfs_independent_set(1, 0);
    
    long long max_independent_set = max(dp[1][0], dp[1][1]);
    long long min_vertex_cover = n - max_independent_set;
    
    cout << "最大独立集大小: " << max_independent_set << endl;
    cout << "最小顶点覆盖大小: " << min_vertex_cover << endl;
    cout << "验证König定理: 最大匹配大小 = " << min_vertex_cover << endl;
    
    return 0;
}
```

## 5. 树上费用流

### 问题5：树上最小费用最大流

**题目描述**：
在一棵树上，每条边有容量和单位费用，求在最大流下的最小费用。

**解法**：
利用树的特殊性质。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
#include <climits>
using namespace std;

const int MAXN = 1e4 + 5;

struct Edge {
    int to, capacity, cost, flow, rev;  // 反向边索引
};

vector<vector<Edge>> graph;
int parent[MAXN], prev_edge[MAXN];
long long dist[MAXN];

void add_edge(int from, int to, int cap, int cost) {
    int from_size = graph[from].size();
    int to_size = graph[to].size();
    
    graph[from].push_back({to, cap, cost, 0, to_size});
    graph[to].push_back({from, 0, -cost, 0, from_size});
}

bool spfa(int source, int sink, int n) {
    fill(dist, dist + n + 1, LLONG_MAX);
    fill(parent, parent + n + 1, -1);
    fill(prev_edge, prev_edge + n + 1, -1);
    
    queue<int> q;
    vector<bool> in_queue(n + 1, false);
    
    dist[source] = 0;
    q.push(source);
    in_queue[source] = true;
    
    while (!q.empty()) {
        int u = q.front(); q.pop();
        in_queue[u] = false;
        
        for (int i = 0; i < graph[u].size(); i++) {
            Edge& e = graph[u][i];
            if (e.capacity > e.flow && dist[u] + e.cost < dist[e.to]) {
                dist[e.to] = dist[u] + e.cost;
                parent[e.to] = u;
                prev_edge[e.to] = i;
                
                if (!in_queue[e.to]) {
                    q.push(e.to);
                    in_queue[e.to] = true;
                }
            }
        }
    }
    
    return dist[sink] != LLONG_MAX;
}

pair<int, long long> min_cost_max_flow(int source, int sink, int n) {
    int max_flow = 0;
    long long min_cost = 0;
    
    while (spfa(source, sink, n)) {
        int flow = INT_MAX;
        
        // 找到增广路径上的最小残量
        for (int u = sink; u != source; u = parent[u]) {
            Edge& e = graph[parent[u]][prev_edge[u]];
            flow = min(flow, e.capacity - e.flow);
        }
        
        // 更新边的流量和费用
        for (int u = sink; u != source; u = parent[u]) {
            Edge& e = graph[parent[u]][prev_edge[u]];
            Edge& rev_e = graph[u][e.rev];
            
            e.flow += flow;
            rev_e.flow -= flow;
            min_cost += (long long)flow * e.cost;
        }
        
        max_flow += flow;
    }
    
    return {max_flow, min_cost};
}

int main() {
    int n;
    cin >> n;
    
    graph.resize(n + 1);
    
    for (int i = 1; i < n; i++) {
        int u, v, cap, cost;
        cin >> u >> v >> cap >> cost;
        add_edge(u, v, cap, cost);
        add_edge(v, u, cap, cost);  // 树上的边可以双向流
    }
    
    int source, sink;
    cin >> source >> sink;
    
    auto result = min_cost_max_flow(source, sink, n);
    
    cout << "最大流: " << result.first << endl;
    cout << "最小费用: " << result.second << endl;
    
    return 0;
}
```

## 6. 树上稳定匹配

### 问题6：树上稳定婚姻问题

**题目描述**：
在一棵树上，节点分为两组，每组节点对另一组节点有偏好列表，求稳定匹配。

**解法**：
使用改进的Gale-Shapley算法。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
using namespace std;

const int MAXN = 1e3 + 5;

vector<int> adj[MAXN];
int color[MAXN];  // 二分图染色，0或1
vector<int> preference[MAXN];  // 每个节点的偏好列表
int partner[MAXN];  // 匹配的伙伴
int next_propose[MAXN];  // 下一个要求婚的对象索引
int rank[MAXN][MAXN];  // rank[u][v] 表示u对v的排名

void dfs_bipartite(int u, int parent_node, int c) {
    color[u] = c;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            dfs_bipartite(v, u, 1 - c);
        }
    }
}

int stable_matching(int n) {
    fill(partner, partner + n + 1, -1);
    fill(next_propose, next_propose + n + 1, 0);
    
    // 找到所有0色节点（求婚者）
    queue<int> free_men;
    for (int i = 1; i <= n; i++) {
        if (color[i] == 0) {
            free_men.push(i);
        }
    }
    
    while (!free_men.empty()) {
        int man = free_men.front();
        free_men.pop();
        
        if (next_propose[man] >= preference[man].size()) {
            continue;  // man已经向所有喜欢的woman求婚过了
        }
        
        int woman = preference[man][next_propose[man]];
        next_propose[man]++;
        
        if (partner[woman] == -1) {
            // woman未婚，接受求婚
            partner[man] = woman;
            partner[woman] = man;
        } else {
            // woman已婚，比较当前求婚者和现任丈夫
            int current_husband = partner[woman];
            if (rank[woman][man] < rank[woman][current_husband]) {
                // woman更喜欢man，离婚并和man结婚
                partner[man] = woman;
                partner[woman] = man;
                partner[current_husband] = -1;
                free_men.push(current_husband);
            } else {
                // woman更喜欢现任丈夫，拒绝man
                free_men.push(man);
            }
        }
    }
    
    int matching_size = 0;
    for (int i = 1; i <= n; i++) {
        if (partner[i] != -1) {
            matching_size++;
        }
    }
    
    return matching_size / 2;  // 每对匹配被计算了两次
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs_bipartite(1, 0, 0);
    
    // 输入偏好列表
    for (int i = 1; i <= n; i++) {
        int pref_count;
        cin >> pref_count;
        preference[i].resize(pref_count);
        for (int j = 0; j < pref_count; j++) {
            cin >> preference[i][j];
        }
        
        // 建立排名表
        for (int j = 0; j < pref_count; j++) {
            rank[i][preference[i][j]] = j;
        }
    }
    
    int result = stable_matching(n);
    
    cout << "稳定匹配大小: " << result << endl;
    
    // 输出匹配结果
    for (int i = 1; i <= n; i++) {
        if (partner[i] != -1 && i < partner[i]) {
            cout << "匹配: " << i << " - " << partner[i] << endl;
        }
    }
    
    return 0;
}
```
