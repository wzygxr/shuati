# 树上高级数据结构难题

## 1. 树上可持久化数据结构

### 问题1：可持久化树上路径查询

**题目描述**：
在一棵树上，支持以下操作：
1. `update_version u val t`：在版本t时将节点u的值更新为val
2. `query_path u v t`：查询在版本t时u到v路径上的某种统计信息（如和、最值等）

**解法**：
使用可持久化线段树+树链剖分。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 2e6 + 5;  // 可持久化线段树节点数量

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int cnt = 0;

// 可持久化线段树节点
struct PSTNode {
    int l, r;
    long long sum, mx, mn;
} tree[MAXM];

int root[MAXN];  // 每个版本的线段树根
int idx = 0;

// 创建新节点
int newNode() {
    idx++;
    tree[idx].l = tree[idx].r = 0;
    tree[idx].sum = tree[idx].mx = tree[idx].mn = 0;
    return idx;
}

// 构建初始线段树
int build(int l, int r) {
    int node = newNode();
    if (l == r) {
        tree[node].sum = tree[node].mx = tree[node].mn = 0;
        return node;
    }
    int mid = (l + r) / 2;
    tree[node].l = build(l, mid);
    tree[node].r = build(mid + 1, r);
    return node;
}

// 更新操作
int update(int pre, int l, int r, int pos, long long val) {
    if (!pre) pre = build(1, cnt);
    
    int cur = newNode(), ret = cur;
    tree[cur] = tree[pre];
    
    if (l == r) {
        tree[cur].sum = tree[cur].mx = tree[cur].mn = val;
        return ret;
    }
    
    int mid = (l + r) / 2;
    if (pos <= mid) {
        tree[cur].l = update(tree[pre].l, l, mid, pos, val);
    } else {
        tree[cur].r = update(tree[pre].r, mid + 1, r, pos, val);
    }
    
    // 更新信息
    tree[cur].sum = tree[tree[cur].l].sum + tree[tree[cur].r].sum;
    tree[cur].mx = max(tree[tree[cur].l].mx, tree[tree[cur].r].mx);
    tree[cur].mn = min(tree[tree[cur].l].mn, tree[tree[cur].r].mn);
    
    return ret;
}

// 查询操作
long long query(int node, int l, int r, int ql, int qr) {
    if (!node || qr < l || ql > r) return 0;
    if (ql <= l && r <= qr) return tree[node].sum;
    
    int mid = (l + r) / 2;
    return query(tree[node].l, l, mid, ql, qr) + 
           query(tree[node].r, mid + 1, r, ql, qr);
}

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

// 版本管理
vector<int> version_roots;  // 各个版本的根

// 查询路径和
long long query_path(int u, int v, int version) {
    long long result = 0;
    int ver_root = version_roots[version];
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        result += query(ver_root, 1, cnt, id[top[u]], id[u]);
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    result += query(ver_root, 1, cnt, id[u], id[v]);
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    // 初始化版本0
    version_roots.push_back(build(1, cnt));
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // update
            int u, val, version;
            cin >> u >> val >> version;
            
            // 创建新版本
            int new_root = update(version_roots[version], 1, cnt, id[u], val);
            if (version_roots.size() <= version + 1) {
                version_roots.resize(version + 2);
            }
            version_roots[version + 1] = new_root;
        } else if (op == 2) {  // query path
            int u, v, version;
            cin >> u >> v >> version;
            cout << "路径和: " << query_path(u, v, version) << endl;
        }
    }
    
    return 0;
}
```

## 2. 树上分块算法

### 问题2：树上带修莫队

**题目描述**：
在一棵树上，支持单点修改和路径查询操作。

**解法**：
使用树分块+带修莫队。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <unordered_map>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int values[MAXN];
int in[MAXN], out[MAXN], timer = 0;
vector<int> euler;  // 欧拉序
int block_size;

void eulerTour(int u, int parent) {
    in[u] = euler.size();
    euler.push_back(values[u]);
    
    for (int v : adj[u]) {
        if (v != parent) {
            eulerTour(v, u);
        }
    }
    
    out[u] = euler.size();
    euler.push_back(values[u]);
}

struct Query {
    int l, r, time, id, u, v;
    bool operator<(const Query& other) const {
        int block_l = l / block_size;
        int block_r = r / block_size;
        if (block_l != block_r) return block_l < block_r;
        if (block_r != (int)(other.r / block_size)) return block_r < (int)(other.r / block_size);
        return time < other.time;
    }
};

struct Update {
    int pos, old_val, new_val;
};

int main() {
    int n, m;
    cin >> n >> m;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    eulerTour(1, 0);
    block_size = cbrt(euler.size());  // 立方根分块
    
    vector<Query> queries;
    vector<Update> updates;
    
    int query_id = 0;
    for (int i = 0; i < m; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // update
            int pos, val;
            cin >> pos >> val;
            updates.push_back({in[pos], values[pos], val});
            values[pos] = val;
        } else {  // query
            int u, v;
            cin >> u >> v;
            int l = in[u], r = in[v];
            if (l > r) swap(l, r);
            queries.push_back({l, r, (int)updates.size(), query_id++, u, v});
        }
    }
    
    sort(queries.begin(), queries.end());
    
    vector<long long> results(queries.size());
    int current_l = 0, current_r = -1, current_time = 0;
    long long current_answer = 0;
    unordered_map<int, int> count;
    
    auto add = [&](int pos) {
        int val = euler[pos];
        count[val]++;
        if (count[val] == 1) current_answer += val;
    };
    
    auto remove = [&](int pos) {
        int val = euler[pos];
        count[val]--;
        if (count[val] == 0) current_answer -= val;
    };
    
    auto update_version = [&](int time) {
        if (time > current_time) {
            for (int i = current_time; i < time; i++) {
                // 应用更新
                if (updates[i].pos >= current_l && updates[i].pos <= current_r) {
                    remove(updates[i].pos);
                    euler[updates[i].pos] = updates[i].new_val;
                    add(updates[i].pos);
                } else {
                    euler[updates[i].pos] = updates[i].new_val;
                }
            }
        } else {
            for (int i = current_time - 1; i >= time; i--) {
                // 撤销更新
                if (updates[i].pos >= current_l && updates[i].pos <= current_r) {
                    remove(updates[i].pos);
                    euler[updates[i].pos] = updates[i].old_val;
                    add(updates[i].pos);
                } else {
                    euler[updates[i].pos] = updates[i].old_val;
                }
            }
        }
        current_time = time;
    };
    
    for (auto& query : queries) {
        update_version(query.time);
        
        while (current_l > query.l) {
            current_l--;
            add(current_l);
        }
        while (current_r < query.r) {
            current_r++;
            add(current_r);
        }
        while (current_l < query.l) {
            remove(current_l);
            current_l++;
        }
        while (current_r > query.r) {
            remove(current_r);
            current_r--;
        }
        
        results[query.id] = current_answer;
    }
    
    for (long long result : results) {
        cout << "查询结果: " << result << endl;
    }
    
    return 0;
}
```

## 3. 树上平衡树

### 问题3：树上区间最值操作

**题目描述**：
在一棵树上，支持以下操作：
1. `update_path u v val`：将u到v路径上所有节点的值与val取最大值
2. `query_path u v`：查询u到v路径上的最值

**解法**：
使用树链剖分+线段树维护区间最值。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 1e5 + 5;
const int INF = 1e9;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], size[MAXN], son[MAXN];
int top[MAXN], id[MAXN], rev[MAXN];
int w[MAXN];  // 节点权值
int cnt = 0;

struct LazySegmentTree {
    vector<int> tree, lazy;
    int n;
    
    LazySegmentTree(int _n) : n(_n) {
        tree.resize(4 * n, -INF);
        lazy.resize(4 * n, -INF);
    }
    
    void push_down(int node, int start, int end) {
        if (lazy[node] != -INF) {
            tree[node] = max(tree[node], lazy[node]);
            
            if (start != end) {
                lazy[node * 2] = max(lazy[node * 2], lazy[node]);
                lazy[node * 2 + 1] = max(lazy[node * 2 + 1], lazy[node]);
            }
            
            lazy[node] = -INF;
        }
    }
    
    void update(int node, int start, int end, int l, int r, int val) {
        push_down(node, start, end);
        
        if (start > r || end < l) return;
        
        if (l <= start && end <= r) {
            lazy[node] = max(lazy[node], val);
            push_down(node, start, end);
            return;
        }
        
        int mid = (start + end) / 2;
        update(node * 2, start, mid, l, r, val);
        update(node * 2 + 1, mid + 1, end, l, r, val);
        
        push_up(node, start, end);
    }
    
    void push_up(int node, int start, int end) {
        push_down(node * 2, start, (start + end) / 2);
        push_down(node * 2 + 1, (start + end) / 2 + 1, end);
        tree[node] = max(tree[node * 2], tree[node * 2 + 1]);
    }
    
    int query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return -INF;
        push_down(node, start, end);
        
        if (l <= start && end <= r) return tree[node];
        
        int mid = (start + end) / 2;
        return max(query(node * 2, start, mid, l, r),
                  query(node * 2 + 1, mid + 1, end, l, r));
    }
    
    void update_range(int l, int r, int val) {
        if (l > r) return;
        update(1, 1, n, l, r, val);
    }
    
    int query_range(int l, int r) {
        if (l > r) return -INF;
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int fa, int d) {
    parent[u] = fa;
    depth[u] = d;
    size[u] = 1;
    son[u] = 0;
    
    for (int v : adj[u]) {
        if (v != fa) {
            dfs1(v, u, d + 1);
            size[u] += size[v];
            if (size[v] > size[son[u]]) {
                son[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (son[u]) dfs2(son[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != son[u]) {
            dfs2(v, v);
        }
    }
}

LazySegmentTree segtree(MAXN);

int query_path_max(int u, int v) {
    int result = -INF;
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        result = max(result, segtree.query_range(id[top[u]], id[u]));
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    result = max(result, segtree.query_range(id[u], id[v]));
    
    return result;
}

void update_path_max(int u, int v, int val) {
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        segtree.update_range(id[top[u]], id[u], val);
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    segtree.update_range(id[u], id[v], val);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0, 1);
    dfs2(1, 1);
    
    // 初始化线段树
    for (int i = 1; i <= n; i++) {
        segtree.update_range(id[i], id[i], w[i]);
    }
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // update path
            int u, v, val;
            cin >> u >> v >> val;
            update_path_max(u, v, val);
        } else {  // query path
            int u, v;
            cin >> u >> v;
            cout << "路径最值: " << query_path_max(u, v) << endl;
        }
    }
    
    return 0;
}
```

## 4. 树上字典树

### 问题4：树上路径异或最大值

**题目描述**：
在一棵树上，每条边有权值，多次查询从根到某节点路径上边权异或值的最大子路径异或值。

**解法**：
使用字典树+DFS。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAX_NODES = 3e6 + 5;  // 字典树节点数量
const int B = 20;  // 二进制位数

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边权)
int trie[MAX_NODES][2], cnt_nodes = 1;
int cnt_end[MAX_NODES];

// 插入数字到字典树
void insert(int num) {
    int node = 1;
    for (int i = B - 1; i >= 0; i--) {
        int bit = (num >> i) & 1;
        if (!trie[node][bit]) {
            trie[node][bit] = ++cnt_nodes;
        }
        node = trie[node][bit];
        cnt_end[node]++;
    }
}

// 查询与给定数字异或的最大值
int query_max_xor(int num) {
    int node = 1, result = 0;
    for (int i = B - 1; i >= 0; i--) {
        int bit = (num >> i) & 1;
        int target_bit = 1 - bit;  // 希望找到相反的位
        
        if (trie[node][target_bit]) {
            result |= (1 << i);
            node = trie[node][target_bit];
        } else {
            node = trie[node][bit];
        }
    }
    return result;
}

int path_xor[MAXN];  // 从根到各节点的路径异或值
int result[MAXN];    // 各查询的结果

void dfs(int u, int parent, int current_xor) {
    path_xor[u] = current_xor;
    
    // 将当前路径异或值插入字典树，并查询最大异或值
    insert(current_xor);
    result[u] = query_max_xor(current_xor);
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (v != parent) {
            dfs(v, u, current_xor ^ w);
        }
    }
    
    // 回溯时从字典树中移除当前值
    // 这里为了简化，实际应用中可能需要更复杂的节点计数
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    
    dfs(1, 0, 0);
    
    for (int i = 0; i < q; i++) {
        int v;
        cin >> v;
        cout << "最大异或值: " << result[v] << endl;
    }
    
    return 0;
}
```

## 5. 树上线段树合并

### 问题5：树上子树统计

**题目描述**：
在一棵树上，每个节点有一个权值，多次查询以某节点为根的子树中，权值在给定区间内的节点数量。

**解法**：
使用线段树合并。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 2e6 + 5;
const int MAX_VAL = 1e6;

vector<int> adj[MAXN];
int values[MAXN];
int root[MAXN];  // 每个节点的权值线段树根

struct Node {
    int l, r, sum;
} tree[MAXM];

int cnt = 0;

int newNode() {
    cnt++;
    tree[cnt].l = tree[cnt].r = tree[cnt].sum = 0;
    return cnt;
}

int update(int pre, int l, int r, int pos, int val) {
    if (!pre) pre = newNode();
    
    int cur = newNode(), ret = cur, p = pre;
    int start = l, end = r;
    
    tree[cur] = tree[pre];
    tree[cur].sum += val;
    
    while (start < end) {
        int mid = (start + end) >> 1;
        if (pos <= mid) {
            tree[cur].l = newNode();
            tree[tree[cur].l] = tree[tree[p].l];
            tree[tree[cur].l].sum += val;
            cur = tree[cur].l;
            p = tree[p].l;
            end = mid;
        } else {
            tree[cur].r = newNode();
            tree[tree[cur].r] = tree[tree[p].r];
            tree[tree[cur].r].sum += val;
            cur = tree[cur].r;
            p = tree[p].r;
            start = mid + 1;
        }
    }
    
    return ret;
}

int query(int node, int l, int r, int ql, int qr) {
    if (!node || qr < l || ql > r) return 0;
    if (ql <= l && r <= qr) return tree[node].sum;
    
    int mid = (l + r) >> 1;
    return query(tree[node].l, l, mid, ql, qr) + 
           query(tree[node].r, mid + 1, r, ql, qr);
}

int merge(int u, int v, int l, int r) {
    if (!u) return v;
    if (!v) return u;
    
    if (l == r) {
        tree[u].sum += tree[v].sum;
        return u;
    }
    
    int mid = (l + r) >> 1;
    tree[u].l = merge(tree[u].l, tree[v].l, l, mid);
    tree[u].r = merge(tree[u].r, tree[v].r, mid + 1, r);
    tree[u].sum = tree[tree[u].l].sum + tree[tree[u].r].sum;
    
    return u;
}

void dfs(int u, int parent) {
    root[u] = update(0, 1, MAX_VAL, values[u], 1);
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            root[u] = merge(root[u], root[v], 1, MAX_VAL);
        }
    }
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0);
    
    for (int i = 0; i < q; i++) {
        int u, l, r;
        cin >> u >> l >> r;
        cout << "子树中[" << l << "," << r << "]区间节点数: " 
             << query(root[u], 1, MAX_VAL, l, r) << endl;
    }
    
    return 0;
}
```
