# 树上分治高级应用

## 1. 点分治高级应用

### 问题1：树上路径统计（距离为k的点对）

**题目描述**：
在一棵树上，求距离恰好为k的点对数量。

**解法**：
使用点分治+FFT。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <complex>
#include <cmath>
using namespace std;

const int MAXN = 1e5 + 5;
const double PI = acos(-1);

vector<int> adj[MAXN];
bool vis[MAXN];
int sz[MAXN], mx[MAXN];
int root, tot;
vector<int> dist[MAXN];  // 从重心出发到各点的距离
long long count[MAXN * 2];  // 距离计数
long long result[MAXN * 2]; // 最终结果

struct Complex {
    double x, y;
    Complex(double _x = 0, double _y = 0) : x(_x), y(_y) {}
    Complex operator+(const Complex& other) const {
        return Complex(x + other.x, y + other.y);
    }
    Complex operator-(const Complex& other) const {
        return Complex(x - other.x, y - other.y);
    }
    Complex operator*(const Complex& other) const {
        return Complex(x * other.x - y * other.y, x * other.y + y * other.x);
    }
};

void fft(vector<Complex>& a, bool invert) {
    int n = a.size();
    
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    
    for (int len = 2; len <= n; len <<= 1) {
        double angle = 2 * PI / len * (invert ? -1 : 1);
        Complex wlen(cos(angle), sin(angle));
        
        for (int i = 0; i < n; i += len) {
            Complex w(1, 0);
            for (int j = 0; j < len / 2; j++) {
                Complex u = a[i + j];
                Complex v = a[i + j + len / 2] * w;
                a[i + j] = u + v;
                a[i + j + len / 2] = u - v;
                w = w * wlen;
            }
        }
    }
    
    if (invert) {
        for (int i = 0; i < n; i++) {
            a[i].x /= n;
        }
    }
}

vector<Complex> multiply(vector<long long> a, vector<long long> b) {
    vector<Complex> fa, fb;
    
    int n = 1;
    while (n < a.size() + b.size()) n <<= 1;
    
    fa.resize(n);
    fb.resize(n);
    
    for (int i = 0; i < a.size(); i++) fa[i] = Complex(a[i], 0);
    for (int i = 0; i < b.size(); i++) fb[i] = Complex(b[i], 0);
    
    fft(fa, false);
    fft(fb, false);
    
    for (int i = 0; i < n; i++) {
        fa[i] = fa[i] * fb[i];
    }
    
    fft(fa, true);
    
    vector<Complex> result(n);
    for (int i = 0; i < n; i++) {
        result[i] = (long long)(fa[i].x + 0.5);
    }
    
    return result;
}

void getRoot(int u, int parent) {
    sz[u] = 1;
    mx[u] = 0;
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getRoot(v, u);
            sz[u] += sz[v];
            mx[u] = max(mx[u], sz[v]);
        }
    }
    
    mx[u] = max(mx[u], tot - sz[u]);
    if (mx[u] < mx[root]) root = u;
}

void getDist(int u, int parent, int d, vector<int>& distances) {
    distances.push_back(d);
    
    for (int v : adj[u]) {
        if (v != parent && !vis[v]) {
            getDist(v, u, d + 1, distances);
        }
    }
}

void calc(int u) {
    vector<int> temp_dist = {0};  // 包含重心本身
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            vector<int> distances;
            getDist(v, u, 1, distances);
            
            // 统计经过u的路径
            vector<long long> poly1(MAXN * 2, 0), poly2(MAXN * 2, 0);
            
            for (int d : temp_dist) {
                poly1[d]++;
            }
            for (int d : distances) {
                poly2[d]++;
            }
            
            // 卷积计算
            vector<Complex> prod = multiply(poly1, poly2);
            for (int i = 0; i < prod.size(); i++) {
                result[i] += (long long)(prod[i].x + 0.5);
            }
            
            // 将当前子树的距离加入temp_dist
            for (int d : distances) {
                temp_dist.push_back(d);
            }
        }
    }
}

void solve(int u) {
    vis[u] = true;
    calc(u);
    
    for (int v : adj[u]) {
        if (!vis[v]) {
            tot = sz[v];
            root = 0;
            mx[0] = MAXN;
            getRoot(v, u);
            solve(root);
        }
    }
}

int main() {
    int n, k;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    mx[0] = tot = n;
    getRoot(1, 0);
    solve(root);
    
    cout << "距离为" << k << "的点对数量: " << result[k] << endl;
    
    return 0;
}
```

## 2. 边分治应用

### 问题2：树上最大路径异或和

**题目描述**：
在一棵树上，每条边有权值，求路径上边权异或和的最大值。

**解法**：
使用边分治+字典树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int B = 30;  // 二进制位数

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边权)
bool vis[MAXN];
int sz[MAXN];
int root, tot;
vector<int> path_xor;  // 从分治中心到各点的路径异或值

struct TrieNode {
    int children[2];
    int cnt;
    TrieNode() : cnt(0) {
        children[0] = children[1] = 0;
    }
};

vector<TrieNode> trie;
int trie_cnt;

int new_node() {
    trie.push_back(TrieNode());
    return trie_cnt++;
}

void insert(int val) {
    int node = 0;
    for (int i = B - 1; i >= 0; i--) {
        int bit = (val >> i) & 1;
        if (!trie[node].children[bit]) {
            trie[node].children[bit] = new_node();
        }
        node = trie[node].children[bit];
        trie[node].cnt++;
    }
}

int query_max_xor(int val) {
    int node = 0, result = 0;
    for (int i = B - 1; i >= 0; i--) {
        int bit = (val >> i) & 1;
        int target_bit = 1 - bit;
        
        if (trie[node].children[target_bit] && trie[trie[node].children[target_bit]].cnt > 0) {
            result |= (1 << i);
            node = trie[node].children[target_bit];
        } else {
            node = trie[node].children[bit];
        }
    }
    return result;
}

void getRoot(int u, int parent) {
    sz[u] = 1;
    int mx = 0;
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        if (v != parent && !vis[v]) {
            getRoot(v, u);
            sz[u] += sz[v];
            mx = max(mx, sz[v]);
        }
    }
    
    mx = max(mx, tot - sz[u]);
    if (mx < tot && mx < sz[root]) root = u;
}

void dfs_dist(int u, int parent, int xor_val, vector<int>& xors) {
    xors.push_back(xor_val);
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (v != parent && !vis[v]) {
            dfs_dist(v, u, xor_val ^ w, xors);
        }
    }
}

int max_xor_result = 0;

void calc(int u) {
    // 清空字典树
    trie.clear();
    trie_cnt = 0;
    int root_node = new_node();
    
    vector<int> all_xors = {0};  // 包含分治中心本身
    
    for (auto& edge : adj[u]) {
        int v = edge.first, w = edge.second;
        if (!vis[v]) {
            vector<int> sub_xors;
            dfs_dist(v, u, w, sub_xors);
            
            // 查询最大异或值
            for (int x : sub_xors) {
                max_xor_result = max(max_xor_result, query_max_xor(x));
            }
            
            // 将子树的异或值插入字典树
            for (int x : sub_xors) {
                insert(x);
                all_xors.push_back(x);
            }
        }
    }
}

void solve(int u) {
    vis[u] = true;
    calc(u);
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        if (!vis[v]) {
            tot = sz[v];
            root = 0;
            sz[0] = 0;
            getRoot(v, u);
            solve(root);
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    
    sz[0] = 0;
    tot = n;
    getRoot(1, 0);
    solve(root);
    
    cout << "最大路径异或和: " << max_xor_result << endl;
    
    return 0;
}
```

## 3. 长链剖分应用

### 问题3：树上区间最值查询

**题目描述**：
在一棵树上，支持子树修改和路径查询操作。

**解法**：
使用长链剖分+线段树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], sz[MAXN], hson[MAXN];
int top[MAXN], id[MAXN], rev[MAXN], cnt = 0;

struct SegTree {
    vector<long long> tree, lazy;
    int n;
    
    SegTree(int _n) : n(_n) {
        tree.resize(4 * n, 0);
        lazy.resize(4 * n, 0);
    }
    
    void push_down(int node, int start, int end) {
        if (lazy[node] != 0) {
            tree[node] += lazy[node] * (end - start + 1);
            
            if (start != end) {
                lazy[node * 2] += lazy[node];
                lazy[node * 2 + 1] += lazy[node];
            }
            
            lazy[node] = 0;
        }
    }
    
    void update(int node, int start, int end, int l, int r, long long val) {
        push_down(node, start, end);
        
        if (start > r || end < l) return;
        
        if (l <= start && end <= r) {
            lazy[node] += val;
            push_down(node, start, end);
            return;
        }
        
        int mid = (start + end) / 2;
        update(node * 2, start, mid, l, r, val);
        update(node * 2 + 1, mid + 1, end, l, r, val);
        
        push_up(node, start, end);
    }
    
    void push_up(int node, int start, int end) {
        push_down(node * 2, start, (start + end) / 2);
        push_down(node * 2 + 1, (start + end) / 2 + 1, end);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    
    long long query(int node, int start, int end, int l, int r) {
        if (start > r || end < l) return 0;
        push_down(node, start, end);
        
        if (l <= start && end <= r) return tree[node];
        
        int mid = (start + end) / 2;
        return query(node * 2, start, mid, l, r) +
               query(node * 2 + 1, mid + 1, end, l, r);
    }
    
    void update_range(int l, int r, long long val) {
        if (l > r) return;
        update(1, 1, n, l, r, val);
    }
    
    long long query_range(int l, int r) {
        if (l > r) return 0;
        return query(1, 1, n, l, r);
    }
};

void dfs1(int u, int parent_node) {
    sz[u] = 1;
    hson[u] = 0;
    depth[u] = depth[parent_node] + 1;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            parent[v] = u;
            dfs1(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[hson[u]]) {
                hson[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (hson[u]) dfs2(hson[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != hson[u]) {
            dfs2(v, v);
        }
    }
}

SegTree segtree(MAXN);

// 子树操作
void update_subtree(int u, long long val) {
    segtree.update_range(id[u], id[u] + sz[u] - 1, val);
}

long long query_subtree(int u) {
    return segtree.query_range(id[u], id[u] + sz[u] - 1);
}

// 路径操作
void update_path(int u, int v, long long val) {
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        segtree.update_range(id[top[u]], id[u], val);
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    segtree.update_range(id[u], id[v], val);
}

long long query_path(int u, int v) {
    long long result = 0;
    
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        result += segtree.query_range(id[top[u]], id[u]);
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    result += segtree.query_range(id[u], id[v]);
    
    return result;
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0);
    dfs2(1, 1);
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // 子树修改
            int u;
            long long val;
            cin >> u >> val;
            update_subtree(u, val);
        } else if (op == 2) {  // 路径查询
            int u, v;
            cin >> u >> v;
            cout << "路径和: " << query_path(u, v) << endl;
        } else if (op == 3) {  // 子树查询
            int u;
            cin >> u;
            cout << "子树和: " << query_subtree(u) << endl;
        }
    }
    
    return 0;
}
```

## 4. 重链剖分高级应用

### 问题4：树上区间第K大查询

**题目描述**：
在一棵树上，支持路径修改和路径第K大查询。

**解法**：
使用重链剖分+主席树。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 2e6 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], sz[MAXN], hson[MAXN];
int top[MAXN], id[MAXN], rev[MAXN], cnt = 0;
int w[MAXN];  // 节点权值

struct主席树Node {
    int l, r, sum;
} tree[MAXM];

int root[MAXN], idx = 0;

int newNode() {
    idx++;
    tree[idx].l = tree[idx].r = tree[idx].sum = 0;
    return idx;
}

int update(int pre, int l, int r, int pos, int val) {
    if (!pre) pre = newNode();
    
    int cur = newNode(), ret = cur, p = pre;
    int start = l, end = r;
    
    tree[cur] = tree[pre];
    tree[cur].sum += val;
    
    while (start < end) {
        int mid = (start + end) >> 1;
        if (pos <= mid) {
            tree[cur].l = newNode();
            tree[tree[cur].l] = tree[tree[p].l];
            tree[tree[cur].l].sum += val;
            cur = tree[cur].l;
            p = tree[p].l;
            end = mid;
        } else {
            tree[cur].r = newNode();
            tree[tree[cur].r] = tree[tree[p].r];
            tree[tree[cur].r].sum += val;
            cur = tree[cur].r;
            p = tree[p].r;
            start = mid + 1;
        }
    }
    
    return ret;
}

int query(int u_root, int v_root, int lca_root, int pp_root, int l, int r, int k) {
    if (l == r) return l;
    
    int mid = (l + r) >> 1;
    int left_sum = tree[tree[u_root].l].sum + tree[tree[v_root].l].sum - 
                   tree[tree[lca_root].l].sum - tree[tree[pp_root].l].sum;
    
    if (k <= left_sum) {
        return query(tree[u_root].l, tree[v_root].l, tree[lca_root].l, 
                    tree[pp_root].l, l, mid, k);
    } else {
        return query(tree[u_root].r, tree[v_root].r, tree[lca_root].r, 
                    tree[pp_root].r, mid + 1, r, k - left_sum);
    }
}

void dfs1(int u, int parent_node) {
    sz[u] = 1;
    hson[u] = 0;
    depth[u] = depth[parent_node] + 1;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            parent[v] = u;
            dfs1(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[hson[u]]) {
                hson[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (hson[u]) dfs2(hson[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != hson[u]) {
            dfs2(v, v);
        }
    }
}

// 获取LCA（使用倍增法）
int lca[MAXN][20];

void dfs_lca(int u, int parent_node) {
    lca[u][0] = parent_node;
    for (int k = 1; k < 20; k++) {
        if (lca[u][k-1]) {
            lca[u][k] = lca[lca[u][k-1]][k-1];
        } else {
            lca[u][k] = 0;
        }
    }
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            dfs_lca(v, u);
        }
    }
}

int getLCA(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    
    int diff = depth[u] - depth[v];
    for (int k = 0; k < 20; k++) {
        if (diff & (1 << k)) {
            u = lca[u][k];
        }
    }
    
    if (u == v) return u;
    
    for (int k = 19; k >= 0; k--) {
        if (lca[u][k] != lca[v][k]) {
            u = lca[u][k];
            v = lca[v][k];
        }
    }
    
    return lca[u][0];
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0);
    dfs2(1, 1);
    dfs_lca(1, 0);
    
    // 构建可持久化线段树
    vector<int> sorted_w = {0};  // 添加0作为离散化辅助
    for (int i = 1; i <= n; i++) {
        sorted_w.push_back(w[i]);
    }
    sort(sorted_w.begin(), sorted_w.end());
    sorted_w.erase(unique(sorted_w.begin(), sorted_w.end()), sorted_w.end());
    
    function<int(int)> get_id = [&](int val) {
        return lower_bound(sorted_w.begin(), sorted_w.end(), val) - sorted_w.begin() + 1;
    };
    
    function<void(int, int)> build_dfs = [&](int u, int parent_node) {
        int w_id = get_id(w[u]);
        root[u] = update(root[parent_node], 1, sorted_w.size(), w_id, 1);
        
        for (int v : adj[u]) {
            if (v != parent_node) {
                build_dfs(v, u);
            }
        }
    };
    
    build_dfs(1, 0);
    
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        
        if (op == 1) {  // 路径第K大查询
            int u, v, k;
            cin >> u >> v >> k;
            int lca_uv = getLCA(u, v);
            int pp = parent[lca_uv];  // lca的父节点
            
            int result = query(root[u], root[v], root[lca_uv], 
                              pp ? root[pp] : 0, 1, sorted_w.size(), k);
            cout << "路径第" << k << "大: " << sorted_w[result - 1] << endl;
        }
    }
    
    return 0;
}
```

## 5. 树上CDQ分治

### 问题5：树上三维偏序问题

**题目描述**：
在一棵树上，每个节点有三个属性(a,b,c)，多次查询满足条件的节点对数量。

**解法**：
使用树上CDQ分治。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <tuple>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int parent[MAXN], depth[MAXN], sz[MAXN], hson[MAXN];
int top[MAXN], id[MAXN], rev[MAXN], cnt = 0;

struct Node {
    int a, b, c, id;
    bool operator<(const Node& other) const {
        if (a != other.a) return a < other.a;
        if (b != other.b) return b < other.b;
        return c < other.c;
    }
};

vector<Node> nodes;

// 树状数组用于二维数点
struct BIT2D {
    vector<vector<int>> tree;
    int n, m;
    
    BIT2D(int _n, int _m) : n(_n), m(_m) {
        tree.resize(n + 1, vector<int>(m + 1, 0));
    }
    
    void update(int x, int y, int val) {
        for (int i = x; i <= n; i += i & (-i)) {
            for (int j = y; j <= m; j += j & (-j)) {
                tree[i][j] += val;
            }
        }
    }
    
    int query(int x, int y) {
        if (x <= 0 || y <= 0) return 0;
        int result = 0;
        for (int i = x; i > 0; i -= i & (-i)) {
            for (int j = y; j > 0; j -= j & (-j)) {
                result += tree[i][j];
            }
        }
        return result;
    }
    
    int query_range(int x1, int y1, int x2, int y2) {
        return query(x2, y2) - query(x1-1, y2) - query(x2, y1-1) + query(x1-1, y1-1);
    }
};

void dfs1(int u, int parent_node) {
    sz[u] = 1;
    hson[u] = 0;
    depth[u] = depth[parent_node] + 1;
    
    for (int v : adj[u]) {
        if (v != parent_node) {
            parent[v] = u;
            dfs1(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[hson[u]]) {
                hson[u] = v;
            }
        }
    }
}

void dfs2(int u, int t) {
    top[u] = t;
    id[u] = ++cnt;
    rev[cnt] = u;
    
    if (hson[u]) dfs2(hson[u], t);
    
    for (int v : adj[u]) {
        if (v != parent[u] && v != hson[u]) {
            dfs2(v, v);
        }
    }
}

// CDQ分治解决三维偏序
int cdq_3d(vector<Node>& arr, int A, int B, int C) {
    if (arr.size() <= 1) {
        return (arr.size() == 1 && arr[0].a <= A && arr[0].b <= B && arr[0].c <= C) ? 1 : 0;
    }
    
    int mid = arr.size() / 2;
    vector<Node> left, right;
    
    for (int i = 0; i < mid; i++) left.push_back(arr[i]);
    for (int i = mid; i < arr.size(); i++) right.push_back(arr[i]);
    
    int result = cdq_3d(left, A, B, C) + cdq_3d(right, A, B, C);
    
    // 处理跨越分割点的贡献
    sort(right.begin(), right.end(), [](const Node& x, const Node& y) {
        return x.b < y.b;
    });
    
    BIT2D bit(100005, 100005);
    int r_idx = 0;
    
    sort(left.begin(), left.end(), [](const Node& x, const Node& y) {
        return x.b < y.b;
    });
    
    for (const Node& l_node : left) {
        while (r_idx < right.size() && right[r_idx].b <= l_node.b) {
            if (right[r_idx].a <= A) {
                bit.update(right[r_idx].b, right[r_idx].c, 1);
            }
            r_idx++;
        }
        
        if (l_node.a <= A) {
            result += bit.query(B, C);
        }
    }
    
    return result;
}

int query_path_3d(int u, int v, int A, int B, int C) {
    vector<Node> path_nodes;
    
    // 获取路径上的节点
    while (top[u] != top[v]) {
        if (depth[top[u]] < depth[top[v]]) swap(u, v);
        for (int i = id[top[u]]; i <= id[u]; i++) {
            if (rev[i]) path_nodes.push_back(nodes[rev[i]]);
        }
        u = parent[top[u]];
    }
    
    if (depth[u] > depth[v]) swap(u, v);
    for (int i = id[u]; i <= id[v]; i++) {
        if (rev[i]) path_nodes.push_back(nodes[rev[i]]);
    }
    
    return cdq_3d(path_nodes, A, B, C);
}

int main() {
    int n, q;
    cin >> n >> q;
    
    for (int i = 1; i <= n; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        nodes.push_back({a, b, c, i});
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs1(1, 0);
    dfs2(1, 1);

        int u, v, A, B, C;
        cin >> u >> v >> A >> B >> C;
        cout << "满足条件的节点数: " << query_path_3d(u, v, A, B, C) << endl;
    }
    
    return 0;
}
```
