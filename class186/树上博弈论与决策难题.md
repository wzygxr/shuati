# 树上博弈论与决策难题

## 1. 树上SG函数高级应用

### 问题1：树上组合博弈

**题目描述**：
在一棵树上，有若干个棋子，两个玩家轮流移动棋子（只能向子节点移动），无法移动者输。求先手必胜还是后手必胜。

**解法**：
使用树上SG函数。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_set>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int sg[MAXN];  // 每个节点的SG值
int pieces[MAXN];  // 每个节点上棋子的数量

int calculateSG(int u) {
    if (sg[u] != -1) return sg[u];
    
    if (adj[u].empty()) {
        // 叶子节点，无法移动，SG值为0
        return sg[u] = 0;
    }
    
    unordered_set<int> next_sg;
    for (int v : adj[u]) {
        next_sg.insert(calculateSG(v));
    }
    
    // 计算mex
    int mex = 0;
    while (next_sg.count(mex)) mex++;
    
    return sg[u] = mex;
}

// 对于多个棋子的Nim游戏
int main() {
    int n, m;
    cin >> n >> m;
    
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
    }
    
    fill(sg, sg + n + 1, -1);
    
    for (int i = 1; i <= m; i++) {
        int pos;
        cin >> pos;
        pieces[pos]++;
    }
    
    int game_xor = 0;
    for (int i = 1; i <= n; i++) {
        if (pieces[i] > 0) {
            int node_sg = calculateSG(i);
            // 如果节点上有奇数个棋子，贡献SG值
            if (pieces[i] % 2 == 1) {
                game_xor ^= node_sg;
            }
        }
    }
    
    if (game_xor != 0) {
        cout << "先手必胜" << endl;
    } else {
        cout << "后手必胜" << endl;
    }
    
    return 0;
}
```

## 2. 树上动态博弈

### 问题2：树上删除节点博弈

**题目描述**：
在一棵树上，两个玩家轮流删除叶子节点，删除节点后与其相连的边也被删除，最后无法操作者输。

**解法**：
使用树形DP计算SG值。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <set>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int sg[MAXN];
bool removed[MAXN];

int calculateTreeSG(int u, int parent) {
    if (removed[u]) return 0;
    
    vector<int> child_sg;
    for (int v : adj[u]) {
        if (v != parent && !removed[v]) {
            child_sg.push_back(calculateTreeSG(v, u));
        }
    }
    
    // 将所有子树的SG值异或起来
    int total_sg = 0;
    for (int sg_val : child_sg) {
        total_sg ^= sg_val;
    }
    
    // 当前节点是否为叶子节点
    int children_count = 0;
    for (int v : adj[u]) {
        if (v != parent && !removed[v]) {
            children_count++;
        }
    }
    
    // 如果是叶子节点，SG值为1（可以移除），否则为total_sg
    if (children_count == 0) {
        return 1;  // 叶子节点，可以被移除
    } else {
        return total_sg;
    }
}

// 简化版本：直接计算树的SG值
int calculateTreeSGSimple(int u, int parent) {
    vector<int> child_sg;
    for (int v : adj[u]) {
        if (v != parent) {
            child_sg.push_back(calculateTreeSGSimple(v, u));
        }
    }
    
    int total = 0;
    for (int sg_val : child_sg) {
        total ^= (sg_val + 1);  // 每棵子树贡献sg+1
    }
    
    return total;
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    int root_sg = calculateTreeSGSimple(1, 0);
    
    if (root_sg != 0) {
        cout << "先手必胜" << endl;
    } else {
        cout << "后手必胜" << endl;
    }
    
    return 0;
}
```

## 3. 树上最优化博弈

### 问题3：树上资源分配博弈

**题目描述**：
在一棵树上，两个玩家轮流选择节点获取权值，但选择一个节点后，其父节点和子节点都不能被选择。求先手能获得的最大权值。

**解法**：
使用树形DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int w[MAXN];  // 节点权值
int dp[MAXN][2];  // dp[u][0/1] 表示以u为根的子树中，u节点是否被选择的最大权值

void dfs(int u, int parent) {
    dp[u][0] = 0;  // u不选
    dp[u][1] = w[u];  // u选
    
    for (int v : adj[u]) {
        if (v != parent) {
            dfs(v, u);
            
            // u不选：v可选可不选
            dp[u][0] += max(dp[v][0], dp[v][1]);
            
            // u选：v不能选
            dp[u][1] += dp[v][0];
        }
    }
}

int main() {
    int n;
    cin >> n;
    
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0);
    
    cout << "先手最大权值: " << max(dp[1][0], dp[1][1]) << endl;
    
    return 0;
}
```

## 4. 树上概率博弈

### 问题4：树上随机决策

**题目描述**：
在一棵树上，从根节点开始随机游走，每个节点有概率选择不同的子节点，求到达某个目标集合的期望收益。

**解法**：
使用概率DP。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e3 + 5;

vector<int> adj[MAXN];
double prob[MAXN][MAXN];  // prob[u][v] 表示从u到v的概率
double dp[MAXN];  // dp[u] 表示从u开始的期望收益
bool target[MAXN];  // 是否为目标节点
double value[MAXN];  // 目标节点的收益

void dfs_prob(int u, int parent) {
    if (target[u]) {
        dp[u] = value[u];
        return;
    }
    
    if (adj[u].empty()) {
        dp[u] = 0;
        return;
    }
    
    dp[u] = 0;
    for (int v : adj[u]) {
        if (v != parent) {
            dfs_prob(v, u);
            dp[u] += prob[u][v] * dp[v];
        }
    }
}

int main() {
    int n, k;
    cin >> n >> k;
    
    for (int i = 1; i < n; i++) {
        int u, v;
        double p;
        cin >> u >> v >> p;
        adj[u].push_back(v);
        adj[v].push_back(u);
        prob[u][v] = p;
        prob[v][u] = 1.0 - p;
    }
    
    for (int i = 0; i < k; i++) {
        int node;
        double val;
        cin >> node >> val;
        target[node] = true;
        value[node] = val;
    }
    
    dfs_prob(1, 0);
    
    cout << "期望收益: " << dp[1] << endl;
    
    return 0;
}
```

## 5. 树上多智能体决策

### 问题5：树上多玩家资源竞争

**题目描述**：
在树上有多个玩家，每个玩家控制一个棋子，轮流移动，目标是占领有价值的节点，每个节点只能被一个玩家占领，求最优策略。

**解法**：
使用博弈树搜索+剪枝。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 105;
const int MAXP = 5;  // 最大玩家数

vector<int> adj[MAXN];
int values[MAXN];  // 节点价值
int positions[MAXP];  // 每个玩家的位置
int n, p, depth_limit = 6;  // 玩家数

struct GameState {
    vector<int> player_pos;  // 每个玩家的位置
    vector<bool> occupied;   // 哪些节点被占领
    vector<int> scores;      // 每个玩家的分数
    
    GameState(int players) : player_pos(players), scores(players, 0) {
        occupied.resize(n + 1, false);
    }
    
    // 评估函数
    int evaluate(int player) {
        int score = scores[player];
        // 考虑与有价值节点的距离
        for (int i = 1; i <= n; i++) {
            if (!occupied[i] && values[i] > 0) {
                // 计算到该节点的最短距离
                score += values[i] / (1 + distanceTo(i, player_pos[player]));
            }
        }
        return score;
    }
    
    int distanceTo(int node, int start) {
        // 简化：使用BFS计算距离
        vector<bool> visited(n + 1, false);
        vector<int> dist(n + 1, INT_MAX);
        vector<int> queue = {start};
        visited[start] = true;
        dist[start] = 0;
        
        for (int i = 0; i < queue.size(); i++) {
            int u = queue[i];
            if (u == node) return dist[u];
            
            for (int v : adj[u]) {
                if (!visited[v]) {
                    visited[v] = true;
                    dist[v] = dist[u] + 1;
                    queue.push_back(v);
                }
            }
        }
        return dist[node];
    }
    
    vector<GameState> getNextStates(int player) {
        vector<GameState> next_states;
        int current_pos = player_pos[player];
        
        // 尝试移动到所有相邻节点
        for (int next_pos : adj[current_pos]) {
            GameState new_state = *this;
            new_state.player_pos[player] = next_pos;
            
            // 如果目标节点未被占领，占领它
            if (!new_state.occupied[next_pos]) {
                new_state.occupied[next_pos] = true;
                new_state.scores[player] += values[next_pos];
            }
            
            next_states.push_back(new_state);
        }
        
        // 也可以选择不移动
        next_states.push_back(*this);
        
        return next_states;
    }
};

int minimax(GameState& state, int depth, int player, int alpha, int beta, int total_players) {
    if (depth == 0) {
        return state.evaluate(player);
    }
    
    vector<GameState> next_states = state.getNextStates(player);
    
    if (player == 0) {  // 最大化玩家
        int max_eval = INT_MIN;
        for (GameState& next_state : next_states) {
            int eval = minimax(next_state, depth - 1, (player + 1) % total_players, 
                              alpha, beta, total_players);
            max_eval = max(max_eval, eval);
            alpha = max(alpha, eval);
            if (beta <= alpha) break;  // 剪枝
        }
        return max_eval;
    } else {  // 最小化玩家
        int min_eval = INT_MAX;
        for (GameState& next_state : next_states) {
            int eval = minimax(next_state, depth - 1, (player + 1) % total_players, 
                              alpha, beta, total_players);
            min_eval = min(min_eval, eval);
            beta = min(beta, eval);
            if (beta <= alpha) break;  // 剪枝
        }
        return min_eval;
    }
}

int main() {
    cin >> n >> p;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    for (int i = 0; i < p; i++) {
        cin >> positions[i];
    }
    
    GameState initial_state(p);
    for (int i = 0; i < p; i++) {
        initial_state.player_pos[i] = positions[i];
    }
    
    int best_score = minimax(initial_state, depth_limit, 0, INT_MIN, INT_MAX, p);
    
    cout << "先手玩家最优得分: " << best_score << endl;
    
    return 0;
}
```

## 6. 树上机制设计

### 问题6：树上拍卖机制

**题目描述**：
在树上进行拍卖，每个节点有一个价值，买家从根节点开始移动，每次移动需要支付边的费用，求最大化收益的策略。

**解法**：
使用树形DP+背包思想。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e3 + 5;
const int MAXB = 1e4 + 5;  // 预算上限

vector<pair<int, int>> adj[MAXN];  // (邻接点, 边费用)
int values[MAXN];  // 节点价值
int dp[MAXN][MAXB];  // dp[u][b] 表示在u节点剩余预算为b时的最大收益

void dfs(int u, int parent, int budget) {
    // 初始化当前节点
    for (int b = 0; b <= budget; b++) {
        dp[u][b] = (b >= 0) ? values[u] : 0;
    }
    
    // 遍历所有子节点
    for (auto& edge : adj[u]) {
        int v = edge.first;
        int cost = edge.second;
        
        if (v != parent) {
            // 递归处理子节点
            dfs(v, u, budget - cost);
            
            // 合并子树信息
            for (int b = budget; b >= cost; b--) {
                dp[u][b] = max(dp[u][b], dp[u][b - cost] + dp[v][budget - cost]);
            }
        }
    }
}

// 优化版本：使用树上背包
int maxValue[MAXN][MAXB];

void tree_knapsack(int u, int parent) {
    // 初始化
    for (int b = 0; b < MAXB; b++) {
        maxValue[u][b] = (b >= 0) ? values[u] : -1e9;
    }
    
    for (auto& edge : adj[u]) {
        int v = edge.first;
        int cost = edge.second;
        
        if (v != parent) {
            tree_knapsack(v, u);
            
            // 0-1背包合并
            for (int b = MAXB - 1; b >= cost; b--) {
                maxValue[u][b] = max(maxValue[u][b], 
                                   maxValue[u][b - cost] + maxValue[v][b - cost] - values[v]);
            }
        }
    }
}

int main() {
    int n, budget;
    cin >> n >> budget;
    
    for (int i = 1; i <= n; i++) {
        cin >> values[i];
    }
    
    for (int i = 1; i < n; i++) {
        int u, v, cost;
        cin >> u >> v >> cost;
        adj[u].push_back({v, cost});
        adj[v].push_back({u, cost});
    }
    
    tree_knapsack(1, 0);
    
    int result = 0;
    for (int b = 0; b <= budget; b++) {
        result = max(result, maxValue[1][b]);
    }
    
    cout << "最大收益: " << result << endl;
    
    return 0;
}
```
