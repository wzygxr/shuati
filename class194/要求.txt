     #include <iostream>

     const int MAXN = 4e5 + 5; // 数组最大长度，圆方树节点数最多为原始节点数的2倍（每个点双对应一个圆节点），笔试按数据范围调整，面试需说明取值依据
     const int MAXM = 4e5 + 5; // 边的最大数量，无向图成对存边，适配原始图与圆方树的边存储
     const int LOG = 20; // LCA倍增数组的最大层级，适配4e5节点的深度需求

     // 原始无向图边结构体：存储目标节点、反向边索引、边编号，兼容重边/自环
     struct Edge {
         int to, rev, id;
         Edge(int _to, int _rev, int _id) : to(_to), rev(_rev), id(_id) {}
     };
     vector<Edge> adj[MAXN]; // 原始无向图邻接表，ML中无向图数据常用邻接表标准化存储
     vector<pair<int, int>> tree[MAXN]; // 圆方树邻接表（存邻接节点+边权），核心存储结构

     int dfn[MAXN], low[MAXN], timestamp = 0; // Tarjan核心数组：dfn时间戳、low最早回溯时间戳
     stack<pair<int, int>> edge_stack; // 存储边的栈，用于点双连通分量提取（圆方树构建的基础）
     int node_cnt; // 圆方树总节点数（原始方节点+新增圆节点），初始为原始节点数n
     bool is_cut[MAXN]; // 标记原始节点是否为割点，圆方树中割点对应方节点，且连接多个圆节点
     int fa[MAXN][LOG], depth[MAXN]; // LCA倍增数组、节点深度，圆方树LCA求解核心
     int dp[MAXN]; // 树上DP数组，用于圆方树路径最值/和求解，笔试高频应用

     // 判断节点是否为圆节点：圆节点编号 > 原始节点数n，方节点编号 ≤ n，面试高频判断方法
     bool is_circle(int u, int n) {
         return u > n;
     }

     // Tarjan算法求点双连通分量，同步构建圆方树（核心函数，笔试必写）
     // 核心原理：1. 提取点双分量，每个点双分量对应一个圆节点；2. 圆节点连接其包含的所有方节点（原始节点）
     // ML中可用于将复杂无向图转化为树结构，简化模型输入，提升特征提取效率
     void tarjan(int u, int fa_node, int n) {
         dfn[u] = low[u] = ++timestamp; // 初始化时间戳：发现时间=最早回溯时间
         int child = 0; // 统计根节点子节点数，用于割点判定
         for (auto &e : adj[u]) { // 遍历原始图所有邻接边
             int v = e.to, id = e.id;
             if (!dfn[v]) { // 邻接节点未访问，递归遍历
                 edge_stack.push({u, v}); // 当前边入栈，用于点双提取
                 child++;
                 tarjan(v, u, n);
                 low[u] = min(low[u], low[v]); // 回溯更新low[u]
                 
                 // 割点判定+点双提取+圆方树构建（核心逻辑，面试高频提问）
                 if ((fa_node == -1 && child >= 2) || (fa_node != -1 && low[v] >= dfn[u])) {
                     is_cut[u] = true; // 标记当前节点为割点（方节点）
                     node_cnt++; // 新增圆节点，编号从n+1开始
                     int circle_node = node_cnt; // 记录当前圆节点编号
                     // 弹出边栈中的边，构成当前点双分量，将分量内方节点与圆节点连接
                     while (true) {
                         auto [x, y] = edge_stack.top();
                         edge_stack.pop();
                         // 方节点x与圆节点circle_node连接，边权可根据题目设置（此处设为1，适配路径计数）
                         tree[x].emplace_back(circle_node, 1);
                         tree[circle_node].emplace_back(x, 1);
                         // 方节点y与圆节点circle_node连接（避免重复连接，因无向边成对存储）
                         tree[y].emplace_back(circle_node, 1);
                         tree[circle_node].emplace_back(y, 1);
                         // 弹出到当前边（u,v）为止，当前边是点双分量的边界
                         if (x == u && y == v) break;
                     }
                 }
             } else if (v != fa_node && dfn[v] < dfn[u]) { // 已访问且非父节点，且是更早发现的节点（非树边）
                 edge_stack.push({u, v}); // 非树边入栈，参与点双分量构成
                 low[u] = min(low[u], dfn[v]); // 更新low[u]
             }
         }
     }

     // 处理根节点所在的点双分量（递归结束后，边栈中剩余边构成根节点的点双，需单独构建圆方树）
     // 笔试易遗漏步骤，面试常考；ML中确保圆方树无遗漏，保证图结构简化的完整性
     void handle_root(int root, int n) {
         if (!edge_stack.empty()) { // 边栈非空，说明根节点所在点双未处理
             node_cnt++;
             int circle_node = node_cnt;
             while (!edge_stack.empty()) {
                 auto [x, y] = edge_stack.top();
                 edge_stack.pop();
                 tree[x].emplace_back(circle_node, 1);
                 tree[circle_node].emplace_back(x, 1);
                 tree[y].emplace_back(circle_node, 1);
                 tree[circle_node].emplace_back(y, 1);
             }
         }
     }

     // 圆方树LCA初始化（倍增法）：预处理深度和倍增数组，适配圆方树的树结构
     // 笔试中LCA是圆方树的高频操作，需熟练实现；ML中可用于圆方树路径特征提取
     void lca_init(int u, int father) {
         fa[u][0] = father; // 初始化2^0级祖先（直接父节点）
         depth[u] = depth[father] + 1; // 初始化节点深度
         // 递推倍增数组，fa[u][k] = fa[fa[u][k-1]][k-1]，面试高频推导公式
         for (int k = 1; k < LOG; k++) {
             fa[u][k] = fa[fa[u][k-1]][k-1];
         }
         // 递归遍历圆方树，初始化所有节点
         for (auto &[v, w] : tree[u]) {
             if (v != father) {
                 lca_init(v, u);
             }
         }
     }

     // 圆方树LCA查询（倍增法）：查询两个节点的最近公共祖先，适配圆方树的圆/方节点混合场景
     // 笔试高频操作，核心用于路径查询；ML中可用于圆方树中节点路径特征的定位
     int lca_query(int u, int v) {
         if (depth[u] < depth[v]) swap(u, v); // 确保u深度≥v
         // 将u上跳至与v同深度
         for (int k = LOG - 1; k >= 0; k--) {
             if (depth[u] - (1 << k) >= depth[v]) {
                 u = fa[u][k];
             }
         }
         if (u == v) return u; // 两节点重合，直接返回
         // 两节点同时上跳，直至找到LCA
         for (int k = LOG - 1; k >= 0; k--) {
             if (fa[u][k] != fa[v][k]) {
                 u = fa[u][k];
                 v = fa[v][k];
             }
         }
         return fa[u][0]; // 返回LCA
     }

     // 圆方树树上DP：求树的最长路（直径），笔试高频应用，适配带权/无权圆方树
     // 面试常考两次DFS/DP求树直径的原理；ML中可用于提取圆方树的全局结构特征
     int max_dist = 0, far_node = 0;
     void dfs_dp(int u, int father) {
         for (auto &[v, w] : tree[u]) {
             if (v == father) continue;
             dfs_dp(v, u);
             // DP核心：当前节点的最长路 = 子节点最长路 + 边权，更新全局最长路
             max_dist = max(max_dist, dp[u] + dp[v] + w);
             dp[u] = max(dp[u], dp[v] + w);
         }
     }
     int get_tree_diameter() {
         max_dist = 0;
         dfs_dp(1, 0); // 第一次DFS，找到最远节点
         max_dist = 0;
         memset(dp, 0, sizeof(dp));
         dfs_dp(far_node, 0); // 第二次DFS，找到最长路（直径）
         return max_dist;
     }

     int main() {
         // 初始化核心数组，笔试中必须注意初始化，避免脏数据（如dfn未初始化导致时间戳异常）
         memset(dfn, 0, sizeof(dfn));
         memset(low, 0, sizeof(low));
         memset(is_cut, false, sizeof(is_cut));
         memset(fa, 0, sizeof(fa));
         memset(depth, 0, sizeof(depth));
         memset(dp, 0, sizeof(dp));

         int n, m; // n为原始无向图节点数（方节点），m为边数，笔试需注意输入格式
         cin >> n >> m;
         node_cnt = n; // 圆方树初始节点数 = 原始方节点数，圆节点从n+1开始编号
         for (int i = 1; i <= m; i++) { // 构建原始无向图，成对存储边，分配唯一边编号
             int u, v;
             cin >> u >> v;
             adj[u].emplace_back(v, adj[v].size(), i);
             adj[v].emplace_back(u, adj[u].size() - 1, i);
         }

         // 遍历所有连通块，构建圆方树（处理非连通图场景）
         for (int i = 1; i <= n; i++) {
             if (!dfn[i]) { // 未访问节点作为当前连通块根节点
                 tarjan(i, -1, n); // 根节点父节点设为-1，用于割点判定
                 handle_root(i, n); // 单独处理根节点所在点双，避免遗漏
             }
         }

         // 初始化圆方树LCA（从任意节点开始，此处从1号方节点开始）
         lca_init(1, 0);

         // 圆方树核心操作示例：求树的直径（笔试可根据题目替换为LCA查询、路径和等）
         int tree_diameter = get_tree_diameter();

         // 输出核心结果，笔试需按题目要求格式输出，ML中可作为图的元特征（圆方树节点数、直径等）
         cout << "圆方树总节点数：" << node_cnt << endl;
         cout << "圆节点数：" << node_cnt - n << endl;
         cout << "圆方树的直径：" << tree_diameter << endl;
         // 示例：输出割点数量（圆方树中割点对应连接多个圆节点的方节点）
         int cut_cnt = 0;
         for (int i = 1; i <= n; i++) if (is_cut[i]) cut_cnt++;
         cout << "原始图割点总数：" << cut_cnt << endl;

         return 0;
     }
