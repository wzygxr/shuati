## 三、算法详解

### 1. 多叉树转二叉树
- **目的**：避免重儿子导致的复杂度退化
- **实现方法**：左儿子右兄弟表示法
- **时间复杂度**：O(n)

### 2. 边的重心求解
- **定义**：删除该边后，两子树大小最接近的边
- **求解方法**：
  - 从任意点开始，向大小超过一半的子树移动
  - 直到无法移动为止
  - 在所有未分割边中选择最优边
- **时间复杂度**：O(n)

### 3. 边分治过程
- **分治策略**：递归处理分割后的子树
- **信息统计**：统计经过分割边的路径信息
- **合并方法**：如何合并子问题解

### 4. 边分树构建（如需要）
- **目的**：支持动态查询操作
- **构建方法**：基于边分治结果构建
- **查询原理**：倍增技术快速跳转

// C++代码模板
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

const int MAXN = 2e5 + 5; // 定义数组最大长度，边分治处理多叉树转二叉树后节点数可能翻倍，故设2e5+5，笔试需根据数据范围调整
const int LOG = 20; // 边分树的最大层级（2^20覆盖1e5节点的深度需求），面试需说明取值依据

struct Edge { // 边的结构体（存储目标节点、边权、反向边索引），边分治的核心数据结构
    int to, w, rev;
    Edge(int _to, int _w, int _rev) : to(_to), w(_w), rev(_rev) {}
};

vector<Edge> adj[MAXN]; // 邻接表存储原始树结构，ML中动态树数据常用邻接表预处理
int n, tot; // n为原始树节点数，tot为多叉树转二叉树后的总节点数，笔试中需注意节点编号扩展
int size[MAXN]; // 存储子树大小，用于求解边的重心
bool vis_edge[MAXN << 1]; // 标记边是否被分割（边分治核心标记数组），面试中需说明边标记的作用
int root, max_sub; // root为当前边分治的重心边对应的子树根，max_sub为最大子树大小（用于找边重心）
int depth[MAXN]; // 存储节点在边分树中的深度，边分树查询的辅助数组
int fa[MAXN][LOG]; // 边分树的倍增数组（fa[u][k]表示u在边分树中2^k级祖先）
vector<int> val[MAXN][LOG]; // 边分树的权值存储（记录祖先路径上的边权信息），ML中可作为边特征存储容器

// 多叉树转二叉树：将多叉树转化为二叉树，避免边分治中重儿子导致的复杂度退化
// 笔试中该步骤是边分治的关键预处理，需熟练实现；ML中可用于树结构的标准化处理
void add_edge(int u, int v, int w) {
    adj[u].emplace_back(v, w, adj[v].size());
    adj[v].emplace_back(u, w, adj[u].size());
}

void dfs_convert(int u, int fa_node) { // 递归将多叉树转为二叉树（左儿子右兄弟结构）
    int last = -1; // 记录当前节点的最后一个子节点（用于构建右兄弟指针）
    for (auto &e : adj[u]) { // 遍历当前节点的所有邻接边
        int v = e.to, w = e.w;
        if (v == fa_node) continue; // 排除父节点，避免循环访问
        tot++; // 新节点编号（多叉树转二叉树后节点数增加）
        if (last == -1) { // 第一个子节点作为左儿子
            add_edge(u, tot, w);
        } else { // 后续子节点作为前一个子节点的右兄弟
            add_edge(last, tot, 0); // 兄弟边权设为0（不影响路径权值计算）
        }
        last = tot;
        dfs_convert(v, u); // 递归处理子节点的多叉树转二叉树
    }
}

// 计算子树大小：输入当前节点、父节点，输出子树大小，用于寻找边的重心
// 笔试中该函数是边分治的基础，需快速手写；ML中可用于提取子树规模特征
void get_size(int u, int fa_node) {
    size[u] = 1; // 初始化子树大小为1（包含当前节点）
    for (auto &e : adj[u]) { // 遍历当前节点的所有邻接边
        int v = e.to;
        if (v == fa_node || vis_edge[e.rev]) continue; // 排除父节点和已分割的边
        get_size(v, u); // 递归计算子树大小
        size[u] += size[v]; // 累加子树大小
    }
}

// 寻找边的重心：输入当前节点、父节点、总子树大小，找到使分割后最大子树最小的边
// 面试高频考点：边的重心定义、寻找逻辑，与点的重心的区别；ML中边重心可作为树的关键分割特征
void find_edge_centroid(int u, int fa_node, int total) {
    for (auto &e : adj[u]) { // 遍历当前节点的所有邻接边
        int v = e.to, w = e.w;
        if (v == fa_node || vis_edge[e.rev]) continue; // 排除父节点和已分割的边
        int s = min(size[v], total - size[v]); // 分割该边后，两个子树的较小大小
        if (s > max_sub) { // 寻找最大子树最小的边（边重心）
            max_sub = s;
            root = v; // 记录边重心对应的子树根节点
        }
        find_edge_centroid(v, u, total); // 递归寻找子树中的边重心
    }
}

// 边分治核心处理函数：输入当前节点、总子树大小，递归分割树并处理路径问题
// 笔试中边分治的核心逻辑，需结合具体题目补充路径统计；ML中可用于拆分树结构提取局部边特征
void edge_divide(int u, int total) {
    get_size(u, -1); // 计算当前子树大小
    max_sub = -1; // 初始化最大子树大小为-1
    find_edge_centroid(u, -1, total); // 寻找当前子树的边重心
    int centroid_edge_v = root; // 边重心对应的子树根节点（分割边为u到centroid_edge_v）
    vis_edge[adj[u].back().rev] = true; // 标记该边已被分割（避免重复处理）
    vis_edge[adj[centroid_edge_v].back().rev] = true;

    // 【核心逻辑】此处根据具体题目补充路径统计代码（如经过当前分割边的路径统计）
    // 面试中需说明：边分治为何要分割边而非节点？—— 避免点分治中重儿子导致的复杂度退化
    // ML关联：分割后的子树可作为独立的边特征单元，输入GNN模型进行局部边关系学习

    int left_size = size[centroid_edge_v]; // 分割后左子树大小
    int right_size = total - left_size; // 分割后右子树大小
    edge_divide(u, right_size); // 递归处理右子树（原节点所在子树）
    edge_divide(centroid_edge_v, left_size); // 递归处理左子树（边重心对应的子树）
}

// 构建边分树：基于边分治的分割结果，构建边分树（支持动态路径查询）
// 笔试中边分树是动态查询的基础，需掌握构建逻辑；ML中可用于动态树的层级特征存储
void build_edge_tree(int u, int fa_node, int d) {
    depth[u] = d; // 记录节点在边分树中的深度
    fa[u][0] = fa_node; // 初始化边分树的直接祖先（2^0级）
    for (int k = 1; k < LOG; k++) {
        fa[u][k] = fa[fa[u][k-1]][k-1]; // 边分树倍增数组递推公式，面试高频提问点
        // 合并祖先路径上的边权信息（如前缀和、最大值），适配动态查询
        val[u][k].resize(val[u][k-1].size() + val[fa[u][k-1]][k-1].size());
        merge(val[u][k-1].begin(), val[u][k-1].end(),
              val[fa[u][k-1]][k-1].begin(), val[fa[u][k-1]][k-1].end(),
              val[u][k].begin());
    }
    for (auto &e : adj[u]) { // 遍历当前节点的邻接边
        int v = e.to, w = e.w;
        if (v == fa_node || vis_edge[e.rev]) continue; // 排除父节点和已分割的边
        val[v][0].push_back(w); // 存储当前边的权值（边分树第0层）
        build_edge_tree(v, u, d + 1); // 递归构建边分树
    }
}

// 边分树查询函数：输入两个节点u、v，查询路径上的目标信息（如权值和、最大值）
// 笔试高频动态查询场景，核心是倍增上跳合并信息；ML中可用于动态提取树路径的边特征
int query_edge_tree(int u, int v) {
    int res = 0; // 初始化查询结果（根据题目需求调整，如最大值设为INT_MIN）
    if (depth[u] < depth[v]) swap(u, v); // 确保u的深度≥v
    // 将u上跳至与v同深度，同时合并路径上的边权信息
    for (int k = LOG - 1; k >= 0; k--) {
        if (depth[u] - (1 << k) >= depth[v]) {
            // 合并val[u][k]中的权值信息（此处以求和为例，笔试需根据题目调整）
            for (int w : val[u][k]) res += w;
            u = fa[u][k];
        }
    }
    if (u == v) return res; // 两节点重合，返回当前结果
    // 两节点同时上跳，合并路径上的边权信息，直至找到LCA
    for (int k = LOG - 1; k >= 0; k--) {
        if (fa[u][k] != fa[v][k]) {
            for (int w : val[u][k]) res += w;
            for (int w : val[v][k]) res += w;
            u = fa[u][k];
            v = fa[v][k];
        }
    }
    // 合并最后两段边的权值（u到LCA、v到LCA）
    for (int w : val[u][0]) res += w;
    for (int w : val[v][0]) res += w;
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    tot = n; // 初始化节点数（多叉树转二叉树前）
    for (int i = 1; i < n; i++) { // 输入原始树的边（多叉树）
        int u, v, w;
        cin >> u >> v >> w;
        add_edge(u, v, w);
    }
    dfs_convert(1, -1); // 多叉树转二叉树，避免边分治复杂度退化，笔试中需说明该步骤的必要性
    edge_divide(1, tot); // 执行边分治（处理静态路径问题）
    build_edge_tree(1, 0, 1); // 构建边分树（支持动态查询）

    // 处理动态查询（示例：查询两点路径权值和），笔试中需根据题目要求调整查询逻辑
    int q;
    cin >> q;
    while (q--) {
        int u, v;
        cin >> u >> v;
        cout << "边分树查询路径权值和：" << query_edge_tree(u, v) << endl;
    }
    return 0;
}
