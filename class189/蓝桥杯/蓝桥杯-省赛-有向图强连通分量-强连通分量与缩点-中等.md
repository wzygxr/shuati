## 题目描述
给定一个有向图，图中包含n个顶点和m条有向边。请计算该有向图中强连通分量的个数，并输出每个强连通分量包含的顶点。

强连通分量定义：在有向图中，如果任意两个顶点u和v，都存在从u到v的路径和从v到u的路径，则称这些顶点构成一个强连通分量。

## 输入格式
第一行包含两个整数n和m，分别表示图的顶点数和边数（1 ≤ n ≤ 10000, 1 ≤ m ≤ 100000）。

接下来m行，每行包含两个整数u和v，表示从顶点u到顶点v有一条有向边（1 ≤ u, v ≤ n）。

## 输出格式
第一行输出一个整数，表示强连通分量的个数k。

接下来k行，每行输出一个强连通分量包含的所有顶点，顶点编号按升序排列，顶点之间用一个空格分隔。

## 样例输入
```
6 8
1 2
2 3
3 1
3 4
4 5
5 6
6 4
2 5
```

## 样例输出
```
3
1 2 3
4 5 6
2
```

## 解题思路

### 核心步骤
1. **构建图**：使用邻接表存储有向图
2. **Tarjan算法**：
   - dfn[u]：节点u的访问时间戳
   - low[u]：从u出发能到达的最小时间戳
   - 栈维护当前DFS路径上的节点
   - 当dfn[u]==low[u]时，找到SCC根节点，出栈标记整个SCC
3. **结果排序**：每个SCC内的节点按编号排序

## 完整代码实现

```cpp
#include <iostream>
#include <vector>
#include <stack>
#include <algorithm>
using namespace std;

const int MAXN = 10005; // 最大节点数，根据题目n ≤ 10000设定，蓝桥杯需注意内存限制

vector<int> adj[MAXN]; // 邻接表存储有向图，vector实现灵活高效
int dfn[MAXN], low[MAXN]; // Tarjan算法核心数组：dfn时间戳，low最早回溯时间
int timestamp = 0; // 全局时间戳计数器，每次访问新节点时递增
bool in_stack[MAXN]; // 标记节点是否在栈中，用于判断是否在当前的DFS路径上
stack<int> st; // Tarjan算法用栈，存储当前DFS路径上的节点

int scc_id[MAXN], scc_cnt = 0; // scc_id[i]表示节点i所属强连通分量编号，scc_cnt为分量总数
vector<vector<int>> scc_nodes; // 存储每个强连通分量包含的所有节点

/**
 * Tarjan算法求强连通分量
 * @param u 当前访问的节点编号
 * 蓝桥杯要点：Tarjan是必会模板，需熟练手写
 * ML关联：SCC提取可用于图神经网络的图粗化（Graph Coarsening）
 */
void tarjan(int u) {
    // 步骤1：初始化当前节点的dfn和low值
    dfn[u] = low[u] = ++timestamp; // 当前时间戳加1，作为节点的发现时间
    st.push(u); // 当前节点入栈，加入当前DFS路径
    in_stack[u] = true; // 标记节点已在栈中
    
    // 步骤2：遍历当前节点的所有出边
    for (int v : adj[u]) { // v是u的邻接节点
        if (dfn[v] == 0) { // 如果v未被访问过（dfn为0表示未访问）
            tarjan(v); // 递归访问v，深入DFS
            // 递归返回后，用v的low值更新u的low值
            // 原理：u能通过v到达的最早节点，也是u能到达的最早节点
            low[u] = min(low[u], low[v]);
        } 
        // 如果v已访问且在栈中，说明v是当前DFS路径上的节点，形成环
        else if (in_stack[v]) { 
            // 用v的发现时间更新u的low值
            // 注意这里用dfn[v]而不是low[v]，因为v在栈中，其low还未最终确定
            low[u] = min(low[u], dfn[v]);
        }
    }
    
    // 步骤3：判断当前节点是否为强连通分量的根节点
    // 当dfn[u] == low[u]时，说明u无法通过任何路径到达发现时间更早的节点
    if (dfn[u] == low[u]) {
        scc_cnt++; // 强连通分量计数加1，发现新的SCC
        vector<int> nodes; // 存储当前SCC的所有节点
        int v;
        
        // 从栈中弹出节点，直到弹出u为止，这些节点构成一个SCC
        do {
            v = st.top(); // 获取栈顶节点
            st.pop(); // 栈顶节点出栈
            in_stack[v] = false; // 标记节点已不在栈中
            scc_id[v] = scc_cnt; // 记录节点v属于当前SCC
            nodes.push_back(v); // 将节点加入当前SCC的节点列表
        } while (v != u); // 循环直到弹出u本身，此时一个SCC完整提取
        
        // 对当前SCC的节点按编号排序，满足输出要求
        sort(nodes.begin(), nodes.end());
        scc_nodes.push_back(nodes); // 保存当前SCC的节点列表
    }
}

int main() {
    ios::sync_with_stdio(false); // 关闭iostream与stdio同步，加速输入输出
    cin.tie(nullptr); // 解除cin与cout绑定，进一步提升IO效率
    
    int n, m; // n:顶点数 m:边数
    cin >> n >> m; // 读取顶点数和边数
    
    // 读取m条有向边，构建邻接表
    for (int i = 0; i < m; i++) {
        int u, v; // 边的起点u，终点v
        cin >> u >> v; // 读取一条边
        adj[u].push_back(v); // 在邻接表中添加从u到v的有向边
    }
    
    // 初始化数组为0，表示所有节点未访问
    for (int i = 1; i <= n; i++) {
        dfn[i] = low[i] = 0; // Tarjan数组初始化为0
        in_stack[i] = false; // 栈标记初始化为false
    }
    
    // 对所有未访问的节点启动Tarjan算法
    // 注意：图可能不连通，需要遍历所有节点
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) { // 如果节点i未被访问（dfn为0）
            tarjan(i); // 从节点i开始Tarjan算法
        }
    }
    
    // 输出结果
    cout << scc_cnt << endl; // 第一行输出强连通分量总数
    
    // 对SCC按首元素排序（可选，根据题目要求）
    sort(scc_nodes.begin(), scc_nodes.end());
    
    // 输出每个强连通分量的节点
    for (const auto& nodes : scc_nodes) {
        for (int i = 0; i < nodes.size(); i++) {
            if (i > 0) cout << " "; // 节点之间用空格分隔
            cout << nodes[i]; // 输出节点编号
        }
        cout << endl; // 每个SCC输出后换行
    }
    
    return 0; // 程序正常结束
}
```

## Python实现

```python
import sys
from collections import defaultdict

# 设置递归深度限制，防止大规模图递归溢出
sys.setrecursionlimit(20000)

def tarjan(u, adj, dfn, low, timestamp, stack, in_stack, scc_id, scc_nodes):
    """
    Tarjan算法求强连通分量
    :param u: 当前节点
    :param adj: 邻接表
    :param dfn: 发现时间数组
    :param low: 最早回溯时间数组
    :param timestamp: 时间戳列表（用于传引用）
    :param stack: 节点栈
    :param in_stack: 栈标记数组
    :param scc_id: SCC编号数组
    :param scc_nodes: SCC节点列表
    :return: 当前SCC数量
    """
    timestamp[0] += 1  # 时间戳递增
    dfn[u] = low[u] = timestamp[0]  # 初始化当前节点的dfn和low
    stack.append(u)  # 节点入栈
    in_stack[u] = True  # 标记在栈中
    
    # 遍历所有邻接节点
    for v in adj[u]:
        if dfn[v] == 0:  # 未访问
            tarjan(v, adj, dfn, low, timestamp, stack, in_stack, scc_id, scc_nodes)
            low[u] = min(low[u], low[v])  # 更新low值
        elif in_stack[v]:  # 已访问且在栈中
            low[u] = min(low[u], dfn[v])  # 用dfn更新low
    
    # 发现SCC根节点
    if dfn[u] == low[u]:
        scc_cnt = len(scc_nodes) + 1
        nodes = []
        while True:
            v = stack.pop()  # 出栈
            in_stack[v] = False  # 标记不在栈中
            scc_id[v] = scc_cnt  # 记录SCC编号
            nodes.append(v)  # 加入当前SCC
            if v == u:  # 直到弹出u
                break
        nodes.sort()  # 排序
        scc_nodes.append(nodes)  # 保存SCC
    
    return len(scc_nodes)

def main():
    # 读取输入
    n, m = map(int, input().split())
    
    # 构建邻接表
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
    
    # 初始化数组
    dfn = [0] * (n + 1)  # 发现时间
    low = [0] * (n + 1)  # 最早回溯时间
    scc_id = [0] * (n + 1)  # SCC编号
    in_stack = [False] * (n + 1)  # 栈标记
    stack = []  # 节点栈
    scc_nodes = []  # SCC节点列表
    timestamp = [0]  # 时间戳（用列表传引用）
    
    # 对所有未访问节点执行Tarjan
    for i in range(1, n + 1):
        if dfn[i] == 0:
            tarjan(i, adj, dfn, low, timestamp, stack, in_stack, scc_id, scc_nodes)
    
    # 输出结果
    print(len(scc_nodes))
    scc_nodes.sort()  # 按首元素排序
    for nodes in scc_nodes:
        print(' '.join(map(str, nodes)))

if __name__ == "__main__":
    main()
```

## 时间/空间复杂度分析

### 时间复杂度
- **Tarjan算法**：O(N+M)
  - 每个节点访问一次：O(N)
  - 每条边遍历一次：O(M)
  - 排序操作：O(N log N)（每个SCC内排序）
- **总时间复杂度**：O(N+M+N log N) = O(N log N + M)

### 空间复杂度
- **邻接表存储**：O(N+M)
- **Tarjan数组**：O(N)（dfn, low, in_stack, scc_id）
- **栈空间**：O(N)（最坏情况所有节点在栈中）
- **总空间复杂度**：O(N+M)
